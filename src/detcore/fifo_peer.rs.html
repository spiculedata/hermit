9//! Does a guest hold the other end of a FIFO?
10//!
11//! A FIFO opened by path is a `Regular` descriptor, so a read or write that
12//! must wait blocks in the kernel while the caller holds the scheduler turn.
13//! When the other end belongs to a host process that is right: the host makes
14//! progress on its own, and every guest waits at a fixed point. When it
15//! belongs to another guest, that guest never runs, and the run deadlocks
16//! (<https://github.com/rrnewton/hermit/issues/4200>).
17//!
18//! So the first blocking read or write on such a FIFO asks whether some guest
19//! holds a descriptor for the same FIFO with the access the other end needs
20//! (a writer for a read, a reader for a write). Only then does Detcore make
21//! the FIFO an internally polled pipe; otherwise the call blocks as before.
22//!
23//! The answer is deterministic. The scan runs in the caller's turn, when every
24//! other guest thread is stopped, and a guest's descriptor table changes only
25//! in its own turns. A host process is never counted: the scan covers only the
26//! processes of Hermit's private PID namespace, whose `/proc` lists nothing
27//! else, and is skipped (answering no) when the tracer is not that
28//! namespace's init.
29
30use std::fs;
31use std::os::unix::fs::FileTypeExt;
32use std::os::unix::fs::MetadataExt;
33use std::path::Path;

35/// The end of a FIFO a waiting call needs another holder of.
36#[derive(Clone, Copy, Debug, PartialEq, Eq)]
37pub(crate) enum FifoPeer {
38    /// A descriptor open for writing (`O_WRONLY` or `O_RDWR`), which a read
39    /// waits for.
40    Writer,
41    /// A descriptor open for reading (`O_RDONLY` or `O_RDWR`), which a write
42    /// waits for.
43    Reader,
44}
45
46impl FifoPeer {
47    /// Whether a descriptor with these status flags is such an end. An
48    /// `O_PATH` handle is neither: it reads and writes nothing, and its access
49    /// bits are 0, which would otherwise read as `O_RDONLY`.
50    fn accepts_flags(self, flags: i32) -> bool {
51        flags & libc::O_PATH == 0 && self.accepts(flags & libc::O_ACCMODE)
52    }
53
54    fn accepts(self, access_mode: i32) -> bool {
55        match self {
56            FifoPeer::Writer => access_mode == libc::O_WRONLY || access_mode == libc::O_RDWR,
57            FifoPeer::Reader => access_mode == libc::O_RDONLY || access_mode == libc::O_RDWR,
58        }
59    }
60}

62/// The file status flags of `<table>/fdinfo/<fd>`, as the kernel holds them
63/// now, where `table` is a process's or a thread's `/proc` directory.
64fn fdinfo_flags(table: &Path, fd: &str) -> Option<i32> {
65    let info = fs::read_to_string(table.join("fdinfo").join(fd)).ok()?;
66    let flags = info.lines().find_map(|line| line.strip_prefix("flags:"))?;
67    i32::from_str_radix(flags.trim(), 8).ok()
68}

70/// Whether Detcore can see a guest's descriptors through `/proc/<tid>` on
71/// `backend`, as every check here needs. Only where the tool is a separate
72/// tracer, outside the guest's descriptor table (not DBT, whose tool runs in
73/// the guest), and where a guest thread's ID is a host task's
74/// ([`crate::procfs_inode::guest_tids_name_host_tasks`]; not KVM, where
75/// `/proc/<tid>` is whatever host task has that ID). `physical_tid` is the
76/// guest thread's host ID where the backend supplies one beside its own.
77/// Elsewhere a FIFO keeps blocking, as before.
78pub(crate) fn guest_descriptors_are_visible(
79    backend: &reverie::BackendCapabilities,
80    physical_tid: Option<i32>,
81) -> bool {
82    !backend.tool_shares_guest_descriptor_table
83        && crate::procfs_inode::guest_tids_name_host_tasks(backend, physical_tid)
84}

86/// Whether guest thread `tid`'s descriptor `fd` names a FIFO (a named FIFO,
87/// or a pipe reached by path). Read through the thread, whose descriptor
88/// table may not be its leader's (a thread cloned without CLONE_FILES), and
89/// only where [`guest_descriptors_are_visible`]: elsewhere `/proc/<tid>` is
90/// not the guest's, and the answer is no. Every later check here runs only on
91/// a FIFO this said yes to.
92pub(crate) fn is_fifo(
93    backend: &reverie::BackendCapabilities,
94    physical_tid: Option<i32>,
95    tid: i32,
96    fd: i32,
97) -> bool {
98    guest_descriptors_are_visible(backend, physical_tid)
99        && fs::metadata(format!("/proc/{tid}/fd/{fd}")).is_ok_and(|meta| meta.file_type().is_fifo())
100}

102/// Whether the kernel's open file description behind thread `tid`'s `fd` is
103/// nonblocking now. `None` when procfs cannot say.
104pub(crate) fn kernel_nonblocking(tid: i32, fd: i32) -> Option<bool> {
105    fdinfo_flags(&Path::new("/proc").join(tid.to_string()), &fd.to_string())
106        .map(|flags| flags & libc::O_NONBLOCK != 0)
107}

109/// Whether a guest holds a descriptor for the FIFO behind thread `tid`'s
110/// `fd` with the access `peer` names, other than `fd`'s own open file
111/// description, wherever that is held (a `dup`, or a child that inherited it).
112/// A FIFO the guest opened read-write is its own writer to the kernel, but
113/// whether data ever comes is up to the FIFO's other holders, which may be
114/// host processes only (`exec 3<>fifo`). `fd` is read in `tid`'s own table.
115pub(crate) fn guest_holds_peer(tid: i32, fd: i32, peer: FifoPeer) -> bool {
116    // Outside a private PID namespace `/proc` lists host processes too.
117    let tracer = std::process::id();
118    if tracer != 1 {
119        return false;
120    }
121    let Ok(fifo) = fs::metadata(format!("/proc/{tid}/fd/{fd}")) else {
122        return false;
123    };
124    let proc_root = Path::new("/proc");
125    let Ok(entries) = fs::read_dir(proc_root) else {
126        return false;
127    };
128    let tracer = tracer.to_string();
129    entries
130        .filter_map(|entry| entry.ok()?.file_name().into_string().ok())
131        .filter(|name| name.bytes().all(|b| b.is_ascii_digit()) && *name != tracer)
132        .any(|process| {
133            descriptor_tables(&proc_root.join(process))
134                .iter()
135                .any(|(table, task)| {
136                    holds_peer(
137                        table,
138                        *task,
139                        (fifo.dev(), fifo.ino()),
140                        peer,
141                        Some((tid, fd)),
142                    )
143                })
144        })
145}

147/// Whether the descriptor table at `table` (a task's `/proc` directory, of
148/// task `task`) holds a descriptor for the FIFO `(dev, ino)` with the access
149/// `peer` names, other than the open file description of `own`, a
150/// `(tid, fd)` pair.
151fn holds_peer(
152    table: &Path,
153    task: i32,
154    (dev, ino): (u64, u64),
155    peer: FifoPeer,
156    own: Option<(i32, i32)>,
157) -> bool {
158    let Ok(descriptors) = fs::read_dir(table.join("fd")) else {
159        return false;
160    };
161    descriptors.filter_map(Result::ok).any(|descriptor| {
162        let Some(fd) = descriptor
163            .file_name()
164            .to_str()
165            .and_then(|fd| fd.parse::<i32>().ok())
166        else {
167            return false;
168        };
169        fs::metadata(descriptor.path())
170            .is_ok_and(|meta| meta.file_type().is_fifo() && meta.dev() == dev && meta.ino() == ino)
171            && fdinfo_flags(table, &fd.to_string()).is_some_and(|flags| peer.accepts_flags(flags))
172            && !own.is_some_and(|(tid, own_fd)| same_description(tid, own_fd, task, fd))
173    })
174}

176/// Each distinct descriptor table of a process, as a task's `/proc` directory
177/// and that task's id. Its threads usually share one table, but a thread
178/// cloned without CLONE_FILES has its own, and the leader's `fd` directory is
179/// empty once the leader has exited (`pthread_exit`). Threads with the same
180/// table are listed once (`kcmp(KCMP_FILES)`); when the kernel cannot say,
181/// each is listed, which costs only a second look.
182///
183/// The leader-exit case is untested: on ptrace any leader `pthread_exit`
184/// beside a live thread hangs before a FIFO is read
185/// (<https://github.com/rrnewton/hermit/issues/4100>).
186fn descriptor_tables(process: &Path) -> Vec<(std::path::PathBuf, i32)> {
187    let Ok(tasks) = fs::read_dir(process.join("task")) else {
188        return Vec::new();
189    };
190    let mut tasks: Vec<(std::path::PathBuf, i32)> = tasks
191        .filter_map(|task| {
192            let task = task.ok()?;
193            let id = task.file_name().to_str()?.parse::<i32>().ok()?;
194            Some((task.path(), id))
195        })
196        .collect();
197    tasks.sort_by_key(|(_, id)| *id);
198    let mut tables: Vec<(std::path::PathBuf, i32)> = Vec::new();
199    for (dir, id) in tasks {
200        let listed = fs::read_dir(dir.join("fd")).is_ok_and(|mut entries| entries.next().is_some());
201        if listed && !tables.iter().any(|(_, seen)| same_table(*seen, id)) {
202            tables.push((dir, id));
203        }
204    }
205    tables
206}

208/// Whether tasks `a` and `b` share one descriptor table (`kcmp(KCMP_FILES)`).
209fn same_table(a: i32, b: i32) -> bool {
210    const KCMP_FILES: libc::c_int = 2;
211    // SAFETY: kcmp reads no memory; it compares two tasks' tables.
212    let order = unsafe {
213        libc::syscall(
214            libc::SYS_kcmp,
215            a as libc::pid_t,
216            b as libc::pid_t,
217            KCMP_FILES,
218            0 as libc::c_ulong,
219            0 as libc::c_ulong,
220        )
221    };
222    order == 0
223}
224
225/// Whether `fd1` of task `pid1` and `fd2` of task `pid2` are one open file
226/// description (`kcmp(KCMP_FILE)`). When the kernel cannot say, they are
227/// taken to be the same, so the descriptor is not counted as a peer and the
228/// FIFO keeps blocking as before.
229fn same_description(pid1: i32, fd1: i32, pid2: i32, fd2: i32) -> bool {
230    const KCMP_FILE: libc::c_int = 0;
231    // SAFETY: kcmp reads no memory; it compares two tasks' descriptors.
232    let order = unsafe {
233        libc::syscall(
234            libc::SYS_kcmp,
235            pid1 as libc::pid_t,
236            pid2 as libc::pid_t,
237            KCMP_FILE,
238            fd1 as libc::c_ulong,
239            fd2 as libc::c_ulong,
240        )
241    };
242    !(1..=3).contains(&order)
243}

319    #[test]
320    fn a_peer_is_a_descriptor_for_the_same_fifo_with_the_other_access() {
321        let dir = tempfile::tempdir().unwrap();
322        let path = fifo_in(dir.path());
323        let me = std::process::id().to_string();
324        let proc_root = Path::new("/proc");
325
326        let reader = OpenOptions::new()
327            .read(true)
328            .custom_flags(libc::O_NONBLOCK)
329            .open(&path)
330            .unwrap();
331        let fifo = identity(&reader);
332        assert!(holds_peer(
333            &proc_root.join(&me),
334            me.parse().unwrap(),
335            fifo,
336            FifoPeer::Reader,
337            None
338        ));
339        assert!(
340            !holds_peer(
341                &proc_root.join(&me),
342                me.parse().unwrap(),
343                fifo,
344                FifoPeer::Writer,
345                None
346            ),
347            "a reader is not the writer a read waits for"
348        );
349
350        let writer = OpenOptions::new().write(true).open(&path).unwrap();
351        assert!(holds_peer(
352            &proc_root.join(&me),
353            me.parse().unwrap(),
354            fifo,
355            FifoPeer::Writer,
356            None
357        ));
358
359        // Another FIFO's writer is not a peer.
360        let other_dir = tempfile::tempdir().unwrap();
361        let other = fifo_in(other_dir.path());
362        let _other_reader = OpenOptions::new()
363            .read(true)
364            .custom_flags(libc::O_NONBLOCK)
365            .open(&other)
366            .unwrap();
367        let _other_writer = OpenOptions::new().write(true).open(&other).unwrap();
368        drop(writer);
369        assert!(
370            !holds_peer(
371                &proc_root.join(&me),
372                me.parse().unwrap(),
373                fifo,
374                FifoPeer::Writer,
375                None
376            ),
377            "the other FIFO's writer must not count"
378        );
379    }

410        let dir = tempfile::tempdir().unwrap();
411        let path = fifo_in(dir.path());
