9//! Procfs files that Detcore renders without the backend opening them
10//! (<https://github.com/rrnewton/hermit/issues/4213>).
11//!
12//! For a file whose every byte comes from Detcore's own state
13//! (`ProcfsFile::rendered_path`), Detcore decides on every backend that it
14//! exists, what its metadata is and what it holds. The backend opens no
15//! procfs file. The guest's descriptor is a placeholder: a memfd named
16//! `PLACEHOLDER_NAME_PREFIX` and the file's path, reopened read-only with
17//! the guest's own flags, a kernel object every backend has. Reopened that
18//! way, the kernel itself refuses writes, write locks and size changes, and
19//! reports the guest's status flags and a read-only descriptor link, as it
20//! does for the procfs file. Detcore answers the rest as Linux answers for a
21//! procfs file only global root may write (0444, or 0644 for the fixed
22//! tunables of `procfs::fixed`), to a process that is not global root:
23//!
24//! - opening it, including the flag errors Linux reports;
25//! - reads, from the procfs snapshot, as for every procfs file;
26//! - `stat` of the path and `fstat` of the descriptor, with Detcore's procfs
27//!   device (`procfs_inode::VIRTUAL_PROC_DEVICE`) and an inode that its path
28//!   alone names (`procfs_inode::rendered_inode`), the same on every backend
29//!   and for every spelling that reaches the file;
30//! - the descriptor's link in any process's `/proc/*/fd`, and reopening
31//!   through it;
32//! - its `fdinfo`, `statfs`, and the calls such a procfs file refuses:
33//!   `mmap`, `fsync`, `fchmod`, extended attributes, file `ioctl`s and seals.
34//!
35//! Not under record and replay, whose Replayer stands its own placeholder in
36//! for every open, and not after the process changed its root with `chroot`,
37//! where the kernel decides whether the file exists. `epoll` refuses the
38//! placeholder (`EPERM`), which has no poll method, where it accepts the
39//! procfs file: a known limitation, see
40//! <https://github.com/rrnewton/hermit/issues/4263>.

163/// The rendered file that the kernel's file with metadata `stat` is, if it
164/// is one: the guest reached it through a spelling Detcore did not resolve,
165/// as a symbolic link or a directory listing, under a backend whose procfs
166/// files Detcore does not name by path
167/// (`procfs_inode::guest_tids_name_host_tasks`).
168///
169/// Each candidate costs a stat, so only a regular file with one link and
170/// permissions 0444 or 0644, as every rendered file is in the kernel's
171/// procfs, is checked, and an identity found not to be a rendered file is
172/// remembered (`NOT_RENDERED`): the checks of one procfs file cost their
173/// stats once per tool process.
174pub(crate) fn rendered_path_of_raw(stat: &DetStat) -> Option<&'static Path> {
175    let has = |field| stat.mask.contains(field);
176    if (has(StatxMask::STATX_TYPE) && stat.mode & libc::S_IFMT != libc::S_IFREG)
177        || (has(StatxMask::STATX_NLINK) && stat.nlink != 1)
178        || (has(StatxMask::STATX_MODE) && !matches!(stat.mode & 0o7777, 0o444 | 0o644))
179    {
180        return None;
181    }
182    let identity = (stat.dev, stat.inode);
183    if NOT_RENDERED
184        .lock()
185        .expect("raw identity cache poisoned")
186        .contains(&identity)
187    {
188        return None;
189    }
190    let found = rendered_path_among(stat.dev, stat.inode, rendered_paths());
191    if found.is_none() && proc_device() == Some(stat.dev) {
192        NOT_RENDERED
193            .lock()
194            .expect("raw identity cache poisoned")
195            .insert(identity);
196    }
197    found
198}

215/// The rendered file among `paths` that the kernel's procfs inode
216/// `(dev, ino)` is, if it is one.
217fn rendered_path_among(
218    dev: u64,
219    ino: u64,
220    mut paths: impl Iterator<Item = &'static std::ffi::CStr>,
221) -> Option<&'static Path> {
222    if proc_device() != Some(dev) {
223        return None;
224    }
225    paths.find_map(|path| {
226        let stat = GuestProcfs::stat(path).ok()?;
227        (stat.st_dev == dev && stat.st_ino == ino)
228            .then(|| Path::new(OsStr::from_bytes(path.to_bytes())))
229            .and_then(|path| ProcfsFile::from_path(path)?.rendered_path())
230    })
231}

233/// Facts about the guest's procfs mount that a rendered file shares with the
234/// kernel's procfs files. The tool runs in the guest's mount and user
235/// namespaces on every backend outside record and replay, so its `/proc` is
236/// the guest's. Read with raw system calls: under in-guest backends the tool
237/// shares the guest's C library.
238///
239/// Each fact is read once, when `/proc` is first reachable, and kept: the
240/// container's procfs mount does not change during the run, while the tool's
241/// own view of it can disappear, as when an in-guest tool runs in a process
242/// that later changed its root. A fact that cannot be read is the errno the
243/// guest's call gets, never a failure of the run.
244struct GuestProcfs;
245
246impl GuestProcfs {
247    fn stat(path: &std::ffi::CStr) -> Result<libc::stat, Errno> {
248        let mut stat = std::mem::MaybeUninit::<libc::stat>::zeroed();
249        // SAFETY: the path is NUL-terminated and `stat` is writable.
250        let result = unsafe {
251            raw_syscall(
252                libc::SYS_newfstatat,
253                [
254                    libc::AT_FDCWD as u64,
255                    path.as_ptr() as u64,
256                    stat.as_mut_ptr() as u64,
257                    0,
258                    0,
259                    0,
260                ],
261            )
262        };
263        raw_errno(result)?;
264        // SAFETY: the call succeeded and initialized the structure.
265        Ok(unsafe { stat.assume_init() })
266    }

268    /// The owner Linux reports for the guest's procfs sysctl files, and for
269    /// the other system-wide procfs files (`/proc/version`, `/proc/cmdline`):
270    /// global root as the guest's user namespace sees it.
271    fn sysctl_owner() -> Result<(u32, u32), Errno> {
272        static OWNER: std::sync::OnceLock<(u32, u32)> = std::sync::OnceLock::new();
273        if let Some(owner) = OWNER.get() {
274            return Ok(*owner);
275        }
276        let stat = Self::stat(c"/proc/sys/kernel")?;
277        Ok(*OWNER.get_or_init(|| (stat.st_uid, stat.st_gid)))
278    }

280    /// The guest's procfs statistics, and its mount flags (`f_flags`, which
281    /// the C library's structure keeps private).
282    fn statfs() -> Result<(libc::statfs, u64), Errno> {
283        // The kernel's `struct statfs` on x86_64: fifteen 64-bit words, of
284        // which the eleventh is `f_flags`.
285        const WORDS: usize = 15;
286        const _: () = assert!(std::mem::size_of::<libc::statfs>() == WORDS * 8);
287        static WORDS_READ: std::sync::OnceLock<[u64; WORDS]> = std::sync::OnceLock::new();
288        let words = match WORDS_READ.get() {
289            Some(words) => *words,
290            None => {
291                let mut words = [0_u64; WORDS];
292                // SAFETY: the path is NUL-terminated and `words` is writable
293                // and as large as the kernel's structure.
294                let result = unsafe {
295                    raw_syscall(
296                        libc::SYS_statfs,
297                        [
298                            c"/proc".as_ptr() as u64,
299                            words.as_mut_ptr() as u64,
300                            0,
301                            0,
302                            0,
303                            0,
304                        ],
305                    )
306                };
307                raw_errno(result)?;
308                *WORDS_READ.get_or_init(|| words)
309            }
310        };
311        let flags = words[10];
312        // SAFETY: the two have the same size and layout, of plain integers.
313        let statfs = unsafe { std::mem::transmute::<[u64; WORDS], libc::statfs>(words) };
314        Ok((statfs, flags))
315    }

317    /// The raw mount ID of the guest's procfs mount, as Linux reports it in
318    /// the fdinfo of a procfs file.
319    fn mount_id() -> Result<u64, Error> {
320        static MOUNT_ID: std::sync::OnceLock<u64> = std::sync::OnceLock::new();
321        if let Some(mount_id) = MOUNT_ID.get() {
322            return Ok(*mount_id);
323        }
324        let mut statx = std::mem::MaybeUninit::<libc::statx>::zeroed();
325        // SAFETY: the path is NUL-terminated and `statx` is writable.
326        let result = unsafe {
327            raw_syscall(
328                libc::SYS_statx,
329                [
330                    libc::AT_FDCWD as u64,
331                    c"/proc".as_ptr() as u64,
332                    0,
333                    u64::from(libc::STATX_MNT_ID),
334                    statx.as_mut_ptr() as u64,
335                    0,
336                ],
337            )
338        };
339        raw_errno(result)?;
340        // SAFETY: the call succeeded and initialized the structure.
341        let statx = unsafe { statx.assume_init() };
342        if statx.stx_mask & libc::STATX_MNT_ID == 0 {
343            // A kernel too old to report it: the guest's read fails with
344            // the errno a missing feature gives, not the run.
345            return Err(Errno::EOPNOTSUPP.into());
346        }
347        Ok(*MOUNT_ID.get_or_init(|| statx.stx_mnt_id))
348    }

350    /// Whether the guest's user namespace maps user ID `id` (`gid` false) or
351    /// group ID `id` (`gid` true). -1 never names an ID.
352    fn maps_id(gid: bool, id: u32) -> bool {
353        static UID_MAP: std::sync::OnceLock<Vec<u8>> = std::sync::OnceLock::new();
354        static GID_MAP: std::sync::OnceLock<Vec<u8>> = std::sync::OnceLock::new();
355        if id == u32::MAX {
356            return false;
357        }
358        let (cell, path) = if gid {
359            (&GID_MAP, c"/proc/self/gid_map")
360        } else {
361            (&UID_MAP, c"/proc/self/uid_map")
362        };
363        let map = match cell.get() {
364            Some(map) => map,
365            None => match Self::read_small(path) {
366                Some(map) => cell.get_or_init(|| map),
367                None => return false,
368            },
369        };
370        String::from_utf8_lossy(map).lines().any(|line| {
371            let fields: Vec<u64> = line
372                .split_whitespace()
373                .filter_map(|field| field.parse().ok())
374                .collect();
375            matches!(fields.as_slice(), [inside, _, count]
376                if (*inside..inside.saturating_add(*count)).contains(&u64::from(id)))
377        })
378    }

380    /// The first 4 KiB of the small procfs file at `path`, read with raw
381    /// system calls.
382    fn read_small(path: &std::ffi::CStr) -> Option<Vec<u8>> {
383        // SAFETY: the path is NUL-terminated.
384        let fd = unsafe {
385            raw_syscall(
386                libc::SYS_openat,
387                [
388                    libc::AT_FDCWD as u64,
389                    path.as_ptr() as u64,
390                    (libc::O_RDONLY | libc::O_CLOEXEC) as u64,
391                    0,
392                    0,
393                    0,
394                ],
395            )
396        };
397        let fd = raw_errno(fd).ok()?;
398        let mut buffer = vec![0_u8; 4096];
399        // SAFETY: `buffer` is writable for its length.
400        let length = unsafe {
401            raw_syscall(
402                libc::SYS_read,
403                [fd, buffer.as_mut_ptr() as u64, buffer.len() as u64, 0, 0, 0],
404            )
405        };
406        // SAFETY: closing the descriptor opened above.
407        unsafe { raw_syscall(libc::SYS_close, [fd, 0, 0, 0, 0, 0]) };
408        buffer.truncate(raw_errno(length).ok()? as usize);
409        Some(buffer)
410    }

416        let _ = Self::mount_id();
417        let _ = Self::maps_id(false, 0);
418        let _ = Self::maps_id(true, 0);
419    }

434/// The metadata Linux gives the procfs file at `path`, owned by `owner`,
435/// with Detcore's procfs device, the inode its path names, and every time at
436/// `epoch`, on every backend.
437fn rendered_stat(path: &Path, epoch: Timespec, owner: (u32, u32)) -> DetStat {
438    let permissions = path
439        .to_str()
440        .and_then(FixedProcfsFile::from_path)
441        .map_or(0o444, FixedProcfsFile::permissions);
442    DetStat {
443        // What a procfs file's `statx` reports.
444        mask: StatxMask::STATX_BASIC_STATS | StatxMask::STATX_MNT_ID,
445        mode: libc::S_IFREG | permissions,
446        uid: owner.0,
447        gid: owner.1,
448        attributes: StatxAttributes::empty(),
449        // What `vfs_getattr_nosec` and `vfs_statx` report as supported.
450        attributes_mask: StatxAttributes::STATX_ATTR_AUTOMOUNT
451            | StatxAttributes::STATX_ATTR_MOUNT_ROOT
452            | StatxAttributes::STATX_ATTR_DAX,
453        dev: procfs_inode::VIRTUAL_PROC_DEVICE,
454        rdev: 0,
455        inode: procfs_inode::rendered_inode(path),
456        nlink: 1,
457        size: 0,
458        blksize: procfs_inode::PROC_BLOCK_SIZE,
459        blocks: 0,
460        atime: epoch,
461        btime: epoch,
462        ctime: epoch,
463        mtime: epoch,
464    }
465}

495/// Linux's answer to opening an existing rendered procfs file (owned by
496/// global root, writable by its owner at most) with `flags`, from a process
497/// that is not global root, before it opens anything: `build_open_flags`, then the lookup's `O_EXCL` and type
498/// checks, then `may_open` and `do_dentry_open`. On success, the flags the
499/// placeholder is reopened with.
500fn rendered_open_flags(flags: OFlag) -> Result<OFlag, Errno> {
501    if let Some(errno) = open_flags_refused(flags) {
502        return Err(errno);
503    }
504    // `O_PATH` keeps only its own flags (`O_TMPFILE` includes `O_DIRECTORY`).
505    if flags.contains(OFlag::O_PATH) {
506        if flags.contains(OFlag::O_DIRECTORY) {
507            return Err(Errno::ENOTDIR);
508        }
509        return Ok(OFlag::O_PATH | (flags & OFlag::O_CLOEXEC));
510    }
511    // A temporary file is made in a directory, which a file is not.
512    if flags.contains(OFlag::O_TMPFILE) {
513        return Err(Errno::ENOTDIR);
514    }
515    if flags.contains(OFlag::O_CREAT | OFlag::O_EXCL) {
516        return Err(Errno::EEXIST);
517    }
518    if flags.contains(OFlag::O_DIRECTORY) {
519        return Err(Errno::ENOTDIR);
520    }
521    // Only the owner bits allow writing; `O_TRUNC` asks for write access.
522    if flags & OFlag::O_ACCMODE != OFlag::O_RDONLY || flags.contains(OFlag::O_TRUNC) {
523        return Err(Errno::EACCES);
524    }
525    // `O_NOATIME` needs the file's owner.
526    if flags.contains(OFlag::O_NOATIME) {
527        return Err(Errno::EPERM);
528    }
529    // Procfs sysctl files cannot do direct I/O.
530    if flags.contains(OFlag::O_DIRECT) {
531        return Err(Errno::EINVAL);
532    }
533    let kept = OFlag::O_APPEND
534        | OFlag::O_ASYNC
535        | OFlag::O_CLOEXEC
536        | OFlag::O_DSYNC
537        | OFlag::O_LARGEFILE
538        | OFlag::O_NONBLOCK
539        | OFlag::O_SYNC;
540    Ok(OFlag::O_RDONLY | (flags & kept))
541}

543/// What `build_open_flags` refuses with EINVAL before any path is looked up,
544/// once `build_open_how` has kept only `O_PATH`'s own flags: a malformed
545/// `__O_TMPFILE` (it needs `O_DIRECTORY` and no `O_CREAT`), a temporary file
546/// without write access, and `O_CREAT` with `O_DIRECTORY`.
547pub(super) fn open_flags_refused(flags: OFlag) -> Option<Errno> {
548    if flags.contains(OFlag::O_PATH) {
549        return None;
550    }
551    let tmpfile = OFlag::O_TMPFILE & !OFlag::O_DIRECTORY;
552    if flags.intersects(tmpfile)
553        && (flags & (OFlag::O_TMPFILE | OFlag::O_CREAT) != OFlag::O_TMPFILE
554            || flags & OFlag::O_ACCMODE == OFlag::O_RDONLY)
555    {
556        return Some(Errno::EINVAL);
557    }
558    flags
559        .contains(OFlag::O_CREAT | OFlag::O_DIRECTORY)
560        .then_some(Errno::EINVAL)
561}

563/// Linux's answer to mapping a procfs sysctl file, which has no `mmap`
564/// method, in `do_mmap`'s order. `noexec` says whether procfs is mounted
565/// `noexec`.
566fn rendered_mmap_answer(
567    addr: usize,
568    len: usize,
569    offset: libc::off_t,
570    prot: ProtFlags,
571    flags: MapFlags,
572    noexec: bool,
573) -> Result<i64, Errno> {
574    // `ksys_mmap_pgoff` refuses MAP_HUGETLB for a file that is not on
575    // hugetlbfs before `do_mmap` looks at anything.
576    if flags.contains(MapFlags::MAP_HUGETLB) || len == 0 || offset & 4095 != 0 {
577        return Err(Errno::EINVAL);
578    }
579    // A length that cannot fit the address space fails before the file is
580    // looked at (`PAGE_ALIGN` overflowing, then `__get_unmapped_area`).
581    // The 47-bit user address space of x86_64 without 5-level paging, which
582    // Hermit assumes: on a host with LA57, Linux would take a higher fixed
583    // address and answer ENODEV where this answers ENOMEM.
584    const TASK_SIZE: usize = 0x7fff_ffff_f000;
585    let Some(len) = len.checked_add(4095).map(|length| length & !4095) else {
586        return Err(Errno::ENOMEM);
587    };
588    if len > TASK_SIZE {
589        return Err(Errno::ENOMEM);
590    }
591    // A fixed address is taken as given; then it must leave room for the
592    // mapping below TASK_SIZE, and be page aligned (measured: EINVAL before
593    // ENODEV).
594    if flags.intersects(MapFlags::MAP_FIXED | MapFlags::MAP_FIXED_NOREPLACE) {
595        if addr > TASK_SIZE - len {
596            return Err(Errno::ENOMEM);
597        }
598        if addr & 4095 != 0 {
599            return Err(Errno::EINVAL);
600        }
601        // `security_mmap_addr`: below `vm.mmap_min_addr` (the machine's, which
602        // does not depend on the configuration) needs CAP_SYS_RAWIO, which a
603        // guest lacks.
604        let mmap_min_addr = crate::machine::MachineDescription::default()
605            .limits
606            .mmap_min_addr;
607        if (addr as u64) < mmap_min_addr {
608            return Err(Errno::EPERM);
609        }
610    }
611    // `file_mmap_ok`: the mapping must end within the largest offset a
612    // regular file can have (`MAX_LFS_FILESIZE`), the offset taken unsigned.
613    const MAX_LFS_FILESIZE: u64 = i64::MAX as u64;
614    if offset as u64 / 4096 > (MAX_LFS_FILESIZE - len as u64) >> 12 {
615        return Err(Errno::EOVERFLOW);
616    }
617    let map_type = flags.bits() & libc::MAP_TYPE;
618    if map_type != libc::MAP_SHARED
619        && map_type != libc::MAP_PRIVATE
620        && map_type != libc::MAP_SHARED_VALIDATE
621    {
622        return Err(Errno::EINVAL);
623    }
624    // MAP_SHARED_VALIDATE refuses flags the file does not support, as
625    // MAP_SYNC on procfs.
626    if map_type == libc::MAP_SHARED_VALIDATE && flags.contains(MapFlags::MAP_SYNC) {
627        return Err(Errno::EOPNOTSUPP);
628    }
629    if map_type != libc::MAP_PRIVATE && prot.contains(ProtFlags::PROT_WRITE) {
630        return Err(Errno::EACCES);
631    }
632    if noexec && prot.contains(ProtFlags::PROT_EXEC) {
633        return Err(Errno::EPERM);
634    }
635    Err(Errno::ENODEV)
636}

638/// Linux's answer to setting the extended attribute `name` of a procfs file,
639/// which has no attribute handlers, to `value` (`size` bytes, `None` if the
640/// guest's buffer cannot be read), or to removing it (`size` is `None`), in
641/// the kernel's order: the name, the size, the value, then the namespace.
642/// Measured against /proc/sys/kernel/random/uuid as a user its user namespace
643/// does not own.
644fn xattr_change_answer(
645    name: &[u8],
646    size: Option<usize>,
647    value: Option<&[u8]>,
648) -> Result<i64, Errno> {
649    if name.is_empty() || name.len() > 255 {
650        return Err(Errno::ERANGE);
651    }
652    if size.is_some_and(|size| size > XATTR_SIZE_MAX) {
653        return Err(Errno::E2BIG);
654    }
655    if size.is_some_and(|size| size > 0) && value.is_none() {
656        return Err(Errno::EFAULT);
657    }
658    let setting = size.is_some();
659    if name.starts_with(b"trusted.") {
660        Err(Errno::EPERM)
661    } else if name == b"system.posix_acl_access" || name == b"system.posix_acl_default" {
662        // A list is parsed first; procfs then has no way to keep it.
663        if setting && posix_acl_parser_refuses(value.unwrap_or_default()) {
664            Err(Errno::EINVAL)
665        } else {
666            Err(Errno::EOPNOTSUPP)
667        }
668    } else if name == b"security.capability" && setting {
669        // The capability module parses the value first.
670        Err(Errno::EINVAL)
671    } else if name.starts_with(b"security.") {
672        // Changing a security attribute needs CAP_SYS_ADMIN.
673        Err(Errno::EPERM)
674    } else if name.starts_with(b"system.") {
675        Err(Errno::EOPNOTSUPP)
676    } else {
677        // `user.` and unknown namespaces need write permission first.
678        Err(Errno::EACCES)
679    }
680}

682/// Whether the kernel's parser (`posix_acl_from_xattr`) refuses `value` as a
683/// POSIX access control list: a value too short for its header, a size that
684/// is not a whole number of entries, an unknown tag, or a named user or group
685/// the caller's user namespace does not map. A bad version, an empty value or
686/// an empty list are not refused here; procfs then has no way to keep any
687/// list (EOPNOTSUPP). The checks of a list's order, permissions and counts
688/// (`posix_acl_valid`) come after that, so a procfs file never reaches them.
689/// Measured on Linux 7.1, as a user and as root in a user namespace.
690fn posix_acl_parser_refuses(value: &[u8]) -> bool {
691    if value.is_empty() {
692        return false;
693    }
694    let Some((header, entries)) = value.split_first_chunk::<4>() else {
695        return true;
696    };
697    if u32::from_le_bytes(*header) != 2 {
698        return false;
699    }
700    if entries.len() % 8 != 0 {
701        return true;
702    }
703    entries.as_chunks::<8>().0.iter().any(|entry| {
704        let tag = u16::from_le_bytes([entry[0], entry[1]]);
705        let id = u32::from_le_bytes([entry[4], entry[5], entry[6], entry[7]]);
706        match tag {
707            // ACL_USER_OBJ, ACL_GROUP_OBJ, ACL_MASK, ACL_OTHER.
708            0x01 | 0x04 | 0x10 | 0x20 => false,
709            0x02 => !GuestProcfs::maps_id(false, id),
710            0x08 => !GuestProcfs::maps_id(true, id),
711            _ => true,
712        }
713    })
714}

718/// without the global capability.
719fn xattr_get_answer(name: &[u8]) -> Result<i64, Errno> {

721        Err(Errno::ERANGE)
722    } else if name.starts_with(b"trusted.") {
723        Err(Errno::ENODATA)
724    } else {

726    }

737/// Write `link`, truncated to `buffer_len`, as `readlink` writes a link.
738fn write_link<M: MemoryAccess>(
739    memory: &mut M,
740    link: &Path,
741    buffer: Option<AddrMut<'_, libc::c_char>>,
742    buffer_len: usize,
743) -> Result<i64, Error> {

749}

759    /// Whether Detcore renders procfs files for this guest thread now.
760    pub(super) fn renders_procfs<G: Guest<Self>>(&self, guest: &G) -> bool {
761        !self.cfg.recordreplay_modes

764                .changed_root

768    /// The procfs file Detcore renders that `path`, absolute in the guest's
769    /// view, names: directly, or as this process's `/proc/*/fd/N` or
770    /// `/dev/fd/N` link to a descriptor Detcore rendered. A spelling that
771    /// asks for a directory fails as Linux fails it.
772    pub(super) async fn rendered_procfs_at<G: Guest<Self>>(
773        &self,
774        guest: &mut G,
775        path: &Path,
776    ) -> Result<Option<RenderedTarget>, Error> {
777        // A `..` component is the kernel's to resolve: it walks through the
778        // components before it, which need not exist or be directories, while
779        // Detcore's names drop them. (A valid `..` spelling of a rendered file
780        // so reaches the kernel's own: a known limitation, see
781        // <https://github.com/rrnewton/hermit/issues/4263>.)
782        if !self.renders_procfs(guest)
783            || path
784                .components()
785                .any(|component| component == std::path::Component::ParentDir)
786        {
787            return Ok(None);
788        }
789        if let Some(procfs) = ProcfsFile::from_path(path) {
790            if procfs.rendered_path().is_none() {
791                return Ok(None);
792            }
793            if names_a_directory(path) {
794                return Err(Errno::ENOTDIR.into());
795            }
796            return Ok(Some(RenderedTarget {
797                procfs,
798                via_link: false,
799            }));
800        }
801        // procfs names descriptors without leading zeros, so `fd/03` does not
802        // exist even where descriptor 3 is a placeholder. Detcore answers so
803        // on every backend (KVM's own procfs resolves the spelling).
804        if let Some((subject, fd)) = proc_fd_target_with_leading_zeros(path) {
805            let own = match subject {
806                Some(subject) => guest.inject(syscalls::Getpid::new()).await? == i64::from(subject),
807                None => true,
808            };
809            let rendered = guest
810                .thread_state()
811                .with_detfd(fd, |detfd| detfd.procfs_rendered_path().is_some())
812                .unwrap_or(false);
813            if own && rendered {
814                return Err(Errno::ENOENT.into());
815            }
816            return Ok(None);
817        }
818        let Some((subject, fd)) = proc_fd_target(path) else {
819            return Ok(None);
820        };
821        if let Some(subject) = subject
822            && guest.inject(syscalls::Getpid::new()).await? != i64::from(subject)
823        {
824            return Ok(None);
825        }
826        let target = guest
827            .thread_state()
828            .with_detfd(fd, |detfd| detfd.procfs_rendered_path())
829            .ok()
830            .flatten()
831            .and_then(ProcfsFile::from_path)
832            .map(|procfs| RenderedTarget {
833                procfs,
834                via_link: true,
835            });
836        // The link reaches a file, which a trailing slash or `/.` refuses.
837        if target.is_some() && names_a_directory(path) {
838            return Err(Errno::ENOTDIR.into());
839        }
840        Ok(target)
841    }

843    /// The rendered procfs file that descriptor `fd`, which the kernel just
844    /// opened, is a placeholder for: one the guest reached through a link
845    /// Detcore does not resolve itself, such as another process's
846    /// `/proc/<pid>/fd/N` or `/proc/*/task/*/fd/N`. A placeholder this tool
847    /// process made is known by its kernel identity. One it did not (another
848    /// process's, under a backend whose tool runs in each guest process) is
849    /// known by its memfd's name and seals, which a placeholder carries where
850    /// the backend supports seals.
851    pub(super) async fn rendered_procfs_behind<G: Guest<Self>>(
852        &self,
853        guest: &mut G,
854        fd: RawFd,
855    ) -> Result<Option<&'static Path>, Error> {
856        if !self.renders_procfs(guest) {
857            return Ok(None);
858        }
859        if let Ok(stat) = self.inject_fstat(guest, fd).await
860            && let Some(path) = placeholder_identity(stat.st_dev, stat.st_ino)
861        {
862            return Ok(Some(path));
863        }
864        // Where the tool runs in one process for every guest, it made every
865        // placeholder, and its identity is the whole answer. Elsewhere a name
866        // and seals a guest could imitate decide: a known limitation, see
867        // <https://github.com/rrnewton/hermit/issues/4263>.
868        if !self.cfg.backend.tool_shares_guest_descriptor_table {
869            return Ok(None);
870        }
871        let named = self.placeholder_named(guest, fd).await?;
872        if named.is_none() {
873            return Ok(None);
874        }
875        let seals = guest
876            .inject(
877                syscalls::Fcntl::new()
878                    .with_fd(fd)
879                    .with_cmd(syscalls::FcntlCmd::F_GET_SEALS),
880            )
881            .await;
882        // `MFD_NOEXEC_SEAL` adds `F_SEAL_EXEC` to the seals Detcore sets. An
883        // `O_PATH` descriptor cannot show its seals (EBADF); its name decides.
884        let wanted = i64::from(PLACEHOLDER_SEALS);
885        Ok(named.filter(|_| match seals {
886            Ok(seals) => seals & wanted == wanted,
887            Err(errno) => errno == Errno::EBADF,
888        }))
889    }

891    /// The rendered procfs file that descriptor `fd`'s memfd is named for,
892    /// read from its link.
893    async fn placeholder_named<G: Guest<Self>>(
894        &self,
895        guest: &mut G,
896        fd: RawFd,
897    ) -> Result<Option<&'static Path>, Error> {
898        const TARGET_CAPACITY: usize = crate::procfs::PLACEHOLDER_LINK_CAPACITY;
899        let mut stack = guest.stack().await;
900        let link = stack.push(c_string::<32>(format!("/proc/self/fd/{fd}").as_bytes()));
901        let target = stack.reserve::<[u8; TARGET_CAPACITY]>();
902        let stack_guard = stack.commit()?;
903        let length = guest
904            .inject(
905                syscalls::Readlinkat::new()
906                    .with_dirfd(libc::AT_FDCWD)
907                    .with_path(PathPtr::from_ptr(link.as_raw() as *const libc::c_char))
908                    .with_buf(Some(target.cast::<libc::c_char>()))
909                    .with_buf_len(TARGET_CAPACITY),
910            )
911            .await;
912        let path = match length {
913            Ok(length) => {
914                let mut bytes = vec![0; (length as usize).min(TARGET_CAPACITY)];
915                guest.memory().read_exact(target.cast::<u8>(), &mut bytes)?;
916                placeholder_link_path(&bytes)
917            }
918            Err(_) => None,
919        };
920        drop(stack_guard);
921        Ok(path)
922    }

924    /// The rendered file whose placeholder the guest's path at `path` (guest
925    /// address, relative to `dirfd`) reaches through a spelling Detcore does
926    /// not resolve itself, such as a symbolic link to a descriptor link or
927    /// `/dev/stdin`: by the kernel identity of what it reaches. Only while
928    /// this process holds a rendered descriptor, so a process that holds none
929    /// pays nothing.
930    pub(super) async fn placeholder_reached<G: Guest<Self>>(
931        &self,
932        guest: &mut G,
933        dirfd: RawFd,
934        path: Option<PathPtr<'_>>,
935        follow: bool,
936    ) -> Option<&'static Path> {
937        let pointer = path?;
938        // No placeholder known to this tool process, which serves every
939        // guest process: nothing can reach one. (Checked first: it costs
940        // nothing, while the next looks at every descriptor.) Where the tool
941        // runs in each guest process, another process's may.
942        let in_guest = self.cfg.backend.tool_shares_guest_descriptor_table;
943        if (!placeholders_exist() && !in_guest) || !self.renders_procfs(guest) {
944            return None;
945        }
946        // Another process's descriptor link (or `/dev/std*`) may reach a
947        // placeholder this process does not hold. Where the tool runs in each
948        // guest process, any spelling may reach one this image inherited and
949        // has not yet looked at (descriptors are discovered when used).
950        if !in_guest && !guest.thread_state().holds_a_rendered_procfs_file() {
951            let text: std::path::PathBuf = pointer.read(&guest.memory()).ok()?;
952            let names_a_link = super::names_descriptor_link(&text)
953                || text.starts_with("/dev/std")
954                || text.starts_with("/dev/fd/");
955            if !names_a_link {
956                return None;
957            }
958        }
959        let path = pointer;
960        let flags = if follow {
961            AtFlags::empty()
962        } else {
963            AtFlags::AT_SYMLINK_NOFOLLOW
964        };
965        let stat = Self::inject_statat(guest, dirfd, path, [Some(path), None], flags)
966            .await
967            .ok()?;
968        let target = StatTarget::Path {
969            dirfd,
970            path: Some(path),
971            follow,
972        };
973        self.recognize_placeholder(guest, &DetStat::from(stat), target)
974            .await
975    }

977    /// The rendered file whose placeholder the kernel object with metadata
978    /// `stat`, reached through `target`, is. By its kernel identity, where this
979    /// tool process made or learned the placeholder. Where the tool runs in
980    /// each guest process, a placeholder another process made, or an earlier
981    /// image of this one (inherited across `fork` or `exec`), is learned here:
982    /// an empty regular file with no link, as a placeholder is, is checked by
983    /// its memfd's name and seals, and then known by its identity. A path is
984    /// opened `O_PATH` for that, which no file notices; an `O_PATH`
985    /// descriptor shows no seals, so for a path, or an `O_PATH` descriptor,
986    /// the name alone decides. (A guest could imitate the name and seals: a known
987    /// limitation, see <https://github.com/rrnewton/hermit/issues/4263>.)
988    pub(super) async fn recognize_placeholder<G: Guest<Self>>(
989        &self,
990        guest: &mut G,
991        stat: &DetStat,
992        target: StatTarget<'_>,
993    ) -> Option<&'static Path> {
994        if let Some(path) = placeholder_identity(stat.dev, stat.inode) {
995            return Some(path);
996        }
997        let has = |field| stat.mask.contains(field);
998        let may_be_placeholder = (!has(StatxMask::STATX_TYPE)
999            || stat.mode & libc::S_IFMT == libc::S_IFREG)
1000            && (!has(StatxMask::STATX_NLINK) || stat.nlink == 0)
1001            && (!has(StatxMask::STATX_SIZE) || stat.size == 0);
1002        if !self.cfg.backend.tool_shares_guest_descriptor_table
1003            || self.cfg.recordreplay_modes
1004            || !may_be_placeholder
1005        {
1006            return None;
1007        }
1008        let (fd, opened) = match target {
1009            StatTarget::Descriptor(fd) => (fd, false),
1010            StatTarget::Path {
1011                dirfd,
1012                path: Some(path),
1013                follow,
1014            } => {
1015                let flags = OFlag::O_PATH
1016                    | OFlag::O_CLOEXEC
1017                    | if follow {
1018                        OFlag::empty()
1019                    } else {
1020                        OFlag::O_NOFOLLOW
1021                    };
1022                let fd = guest
1023                    .inject(
1024                        syscalls::Openat::new()
1025                            .with_dirfd(dirfd)
1026                            .with_path(Some(path))
1027                            .with_flags(flags),
1028                    )
1029                    .await
1030                    .ok()?;
1031                (fd as RawFd, true)
1032            }
1033            _ => return None,
1034        };
1035        let path = self.rendered_procfs_behind(guest, fd).await.ok().flatten();
1036        if opened {
1037            let _ = guest.inject(syscalls::Close::new().with_fd(fd)).await;
1038        }
1039        if let Some(path) = path {
1040            register_placeholder(stat.dev, stat.inode, path);
1041        }
1042        path
1043    }

1046    /// did not record it as one (an alias the kernel opened while metadata is
1047    /// not virtualized). Where one tool process serves every guest, only while
1048    /// this process holds a rendered descriptor; where the tool runs in each
1049    /// guest process, for any descriptor (an `fstat` per call, only without
1050    /// virtualized metadata).

1064            .await
1065    }

1067    /// Open the rendered procfs file `target` names with the guest's `flags`.
1068    pub(super) async fn open_rendered_procfs<G: Guest<Self>>(
1069        &self,
1070        guest: &mut G,
1071        flags: OFlag,
1072        target: RenderedTarget,
1073    ) -> Result<i64, Error> {
1074        // A descriptor link is a symbolic link: `O_NOFOLLOW` refuses it, once
1075        // the flags themselves passed. (With `O_PATH`, the caller let the
1076        // kernel open the link itself.)
1077        if let Some(errno) = open_flags_refused(flags) {
1078            return Err(errno.into());
1079        }
1080        // `O_CREAT|O_EXCL` follows no final link: the link exists (EEXIST).
1081        if target.via_link
1082            && flags.contains(OFlag::O_CREAT | OFlag::O_EXCL)
1083            && !flags.contains(OFlag::O_PATH)
1084        {
1085            return Err(Errno::EEXIST.into());
1086        }
1087        if target.via_link && flags.contains(OFlag::O_NOFOLLOW) {
1088            return Err(Errno::ELOOP.into());
1089        }
1090        let reopen = rendered_open_flags(flags)?;
1091        GuestProcfs::remember();
1092        // Read before the placeholder exists, so that a failure leaves the
1093        // guest nothing open.
1094        let owner = GuestProcfs::sysctl_owner()?;
1095        let path = target
1096            .procfs
1097            .rendered_path()
1098            .expect("an opened rendered procfs file has a path");
1099        // The snapshot this open takes, which its placeholder's name carries,
1100        // so that every backend has it after `exec`; a fixed file's is
1101        // rendered again there.
1102        let procfs = target.procfs.into_rendered();
1103        let snapshot = self.render_snapshot(guest, &procfs);
1104        let carried: &[u8] = if procfs.snapshot_is_fixed() {
1105            &[]
1106        } else {
1107            &snapshot
1108        };
1109        let fd = self
1110            .inject_placeholder(guest, path, carried, reopen)
1111            .await?;
1112        // The status flags Linux keeps, which include `O_NOFOLLOW`: the
1113        // placeholder cannot be reopened with it through its link.
1114        guest.thread_state().add_fd(
1115            fd,
1116            reopen | (flags & OFlag::O_NOFOLLOW),
1117            FdType::Regular,
1118            Some(rendered_stat(path, epoch_time(guest.config().epoch), owner)),
1119        )?;
1120        let procfs = procfs.with_rendered_snapshot(snapshot);

1156    /// Inject a placeholder for `path` open with `flags`, at the lowest free
1157    /// descriptor number, as `open` would place the file.
1158    async fn inject_placeholder<G: Guest<Self>>(
1159        &self,
1160        guest: &mut G,
1161        path: &'static Path,
1162        snapshot: &[u8],
1163        flags: OFlag,
1164    ) -> Result<RawFd, Error> {
1165        let name = placeholder_name(path, snapshot);
1166        let mut stack = guest.stack().await;
1167        let name = stack.push(c_string::<PLACEHOLDER_NAME_CAPACITY>(name.as_bytes()));
1168        let link = stack.reserve::<[u8; 32]>();
1169        let stack_guard = stack.commit()?;
1170        let name = name.as_raw();
1171        let create = move |flags| {
1172            syscalls::MemfdCreate::new()
1173                .with_name(PathPtr::from_ptr(name as *const libc::c_char))
1174                .with_flags(flags)
1175        };
1176        // Not executable, which a hardened `vm.memfd_noexec` requires and
1177        // kernels before 6.3 do not know; sealed, so that a spelling Detcore
1178        // does not recognise cannot change the empty file.
1179        let sealable = libc::MFD_CLOEXEC | libc::MFD_ALLOW_SEALING;
1180        let memfd = match guest.inject(create(sealable | libc::MFD_NOEXEC_SEAL)).await {
1181            Err(Errno::EINVAL) => guest.inject(create(sealable)).await?,
1182            created => created?,
1183        } as RawFd;
1184        // A backend without seals (KVM refuses `F_ADD_SEALS`) leaves the
1185        // placeholder unsealed; Detcore still answers every call it knows.
1186        let _ = guest
1187            .inject(
1188                syscalls::Fcntl::new()
1189                    .with_fd(memfd)
1190                    .with_cmd(syscalls::FcntlCmd::F_ADD_SEALS(PLACEHOLDER_SEALS)),
1191            )
1192            .await;
1193        guest.memory().write_exact(
1194            link.cast::<u8>(),
1195            &c_string::<32>(format!("/proc/self/fd/{memfd}").as_bytes()),
1196        )?;
1197        // Reopening through the descriptor link gives a description with the
1198        // guest's flags, which then takes the memfd's number.
1199        let reopened = guest
1200            .inject(
1201                syscalls::Openat::new()
1202                    .with_dirfd(libc::AT_FDCWD)
1203                    .with_path(PathPtr::from_ptr(link.as_raw() as *const libc::c_char))
1204                    .with_flags(flags | OFlag::O_CLOEXEC),
1205            )
1206            .await;
1207        let moved = match reopened {
1208            Ok(reopened) => {
1209                let moved = guest
1210                    .inject(
1211                        syscalls::Dup3::new()
1212                            .with_oldfd(reopened as i32)
1213                            .with_newfd(memfd)
1214                            .with_flags(flags & OFlag::O_CLOEXEC),
1215                    )
1216                    .await;
1217                let _ = guest
1218                    .inject(syscalls::Close::new().with_fd(reopened as i32))
1219                    .await;
1220                moved
1221            }
1222            Err(errno) => Err(errno),
1223        };
1224        drop(stack_guard);
1225        // The guest gets the errno that kept the file from opening, as an open
1226        // that runs out of descriptors or memory reports it; nothing is left
1227        // open.
1228        if let Err(errno) = moved {
1229            let _ = guest.inject(syscalls::Close::new().with_fd(memfd)).await;
1230            return Err(errno.into());
1231        }
1232        let identity = match self.inject_fstat(guest, memfd).await {
1233            Ok(stat) => stat,
1234            Err(errno) => {
1235                let _ = guest.inject(syscalls::Close::new().with_fd(memfd)).await;
1236                return Err(errno.into());
1237            }
1238        };
1239        register_placeholder(identity.st_dev, identity.st_ino, path);
1240        Ok(memfd)
1241    }

1264    /// from its kernel position, before a read or seek uses it. Processes
1265    /// that share the open file across `fork` share only that position, which
1266    /// Detcore keeps at each one's cursor after every read and seek
1267    /// (`sync_rendered_procfs_position`). Where one tool process serves every
1268    /// guest, the cursor itself is shared.
1269    pub(super) async fn adopt_rendered_procfs_position<G: Guest<Self>>(
1270        &self,
1271        guest: &mut G,
1272        fd: RawFd,
1273    ) -> Result<(), Error> {
1274        if !self.cfg.backend.tool_shares_guest_descriptor_table {
1275            return Ok(());
1276        }
1277        let kept = guest
1278            .thread_state()
1279            .with_detfd(fd, |detfd| detfd.procfs_keeps_kernel_position())
1280            .unwrap_or(false);
1281        if !kept {
1282            return Ok(());
1283        }

1290                    .with_whence(syscalls::Whence::SEEK_CUR),
1291            )
1292            .await
1293        {
1294            guest
1295                .thread_state()
1296                .with_detfd(fd, |detfd| detfd.set_procfs_offset(offset as usize))?;
1297        }
1298        Ok(())
1299    }

1301    /// Keep a placeholder's kernel position at Detcore's cursor, which a
1302    /// backend that rediscovers descriptors after `exec`, or another process
1303    /// sharing the open file, reads back (`adopt_rendered_procfs_position`).
1304    pub(super) async fn sync_rendered_procfs_position<G: Guest<Self>>(
1305        &self,
1306        guest: &mut G,
1307        fd: RawFd,
1308    ) -> Result<(), Error> {
1309        let (offset, placeholder) = guest.thread_state().with_detfd(fd, |detfd| {
1310            let offset = detfd
1311                .procfs_keeps_kernel_position()
1312                .then(|| detfd.procfs_position().map(|(offset, _)| offset))
1313                .flatten();
1314            (offset, detfd.procfs_rendered_path().is_some())
1315        })?;
1316        if let Some(offset) = offset {
1317            let moved = guest
1318                .inject(
1319                    syscalls::Lseek::new()
1320                        .with_fd(fd)
1321                        .with_offset(offset as libc::off_t)
1322                        .with_whence(syscalls::Whence::SEEK_SET),
1323                )
1324                .await;
1325            // A placeholder takes any position. The kernel's own file may
1326            // refuse one past its contents, which differ from Detcore's;
1327            // Detcore's cursor still answers the guest.
1328            if placeholder {
1329                moved?;
1330            }
1331        }
1332        Ok(())
1333    }

1335    /// The metadata of the rendered procfs file `target` names, if it names
1336    /// one.
1337    pub(super) async fn rendered_procfs_stat<G: Guest<Self>>(
1338        &self,
1339        guest: &mut G,
1340        target: StatTarget<'_>,
1341    ) -> Result<Option<DetStat>, Error> {
1342        let path = match target {
1343            StatTarget::Descriptor(fd) => {
1344                match guest
1345                    .thread_state()
1346                    .with_detfd(fd, |detfd| detfd.procfs_rendered_path())
1347                    .ok()
1348                    .flatten()
1349                {
1350                    Some(path) => Some(path),
1351                    // With metadata virtualized, `determinize_stat` already
1352                    // knows a placeholder by its identity.
1353                    None if !guest.config().virtualize_metadata => {
1354                        self.placeholder_descriptor(guest, fd).await
1355                    }
1356                    None => None,
1357                }
1358            }
1359            StatTarget::Path {
1360                dirfd,
1361                path: Some(pointer),
1362                follow,
1363            } => {
1364                let path = pointer.read(&guest.memory())?;
1365                // Without `AT_EMPTY_PATH` (which names a descriptor), an
1366                // empty path names nothing: the kernel's ENOENT.
1367                if path.as_os_str().is_empty() {
1368                    return Ok(None);
1369                }

1372                    // A link's own metadata is not its target's: `lstat` of

1376                    None if !guest.config().virtualize_metadata => {
1377                        self.placeholder_reached(guest, dirfd, Some(pointer), follow)
1378                            .await
1379                    }
1380                    None => None,
1381                }
1382            }
1383            StatTarget::Path { path: None, .. } | StatTarget::Cwd | StatTarget::Overwritten => None,
1384        };
1385        let Some(path) = path else {
1386            return Ok(None);
1387        };
1388        Ok(Some(rendered_stat(
1389            path,
1390            epoch_time(guest.config().epoch),
1391            GuestProcfs::sysctl_owner()?,
1392        )))

1432    /// bytes, if it names a placeholder: another process's link to a
1433    /// rendered file names the file's path, as Linux's does. (A buffer too
1434    /// small for the placeholder's link keeps the kernel's truncated bytes.)
1435    pub(crate) async fn rewrite_placeholder_link<G: Guest<Self>>(
1436        &self,
1437        guest: &mut G,
1438        link: (RawFd, Option<PathPtr<'_>>),
1439        buffer: Option<AddrMut<'_, libc::c_char>>,
1440        buffer_len: usize,
1441        result: i64,
1442        saved: Option<&[u8]>,
1443    ) -> Result<i64, Error> {
1444        let (Some(address), Ok(length)) = (buffer, usize::try_from(result)) else {
1445            return Ok(result);
1446        };

1451            || (!placeholders_exist() && !self.cfg.backend.tool_shares_guest_descriptor_table)

1454        }
1455        let mut bytes = vec![0; length.min(buffer_len)];

1465                    written as usize,

1471            _ => Ok(result),

1516    /// Answer `call` if it must not reach the placeholder behind a rendered
1517    /// procfs descriptor: what a procfs sysctl file refuses, in the order
1518    /// Linux checks. The placeholder's read-only description already makes
1519    /// the kernel refuse writes and size changes, and an `O_PATH` one every
1520    /// call but `fstat`, which Detcore answers.
1521    pub(crate) fn answer_rendered_procfs_operation<G: Guest<Self>>(
1522        &self,
1523        guest: &mut G,
1524        call: &Syscall,
1525    ) -> Option<Result<i64, Error>> {
1526        let (sysno, args) = call.into_parts();
1527        let fd = match sysno {
1528            Sysno::mmap => args.arg4 as i32,
1529            Sysno::fsync
1530            | Sysno::fdatasync
1531            | Sysno::fchmod
1532            | Sysno::fgetxattr
1533            | Sysno::flistxattr
1534            | Sysno::fsetxattr
1535            | Sysno::fremovexattr
1536            | Sysno::ioctl
1537            | Sysno::fcntl => args.arg0 as i32,
1538            _ => return None,
1539        };
1540        let status_flags = guest
1541            .thread_state()
1542            .with_detfd(fd, |detfd| {
1543                detfd.procfs_rendered_path().map(|_| detfd.status_flags())
1544            })
1545            .ok()
1546            .flatten()?;
1547        if status_flags & libc::O_PATH != 0 {
1548            return None;
1549        }
1550        let answer = match call {
1551            Syscall::Mmap(c) => {
1552                if c.flags().contains(MapFlags::MAP_ANONYMOUS) {
1553                    return None;
1554                }
1555                let noexec = match GuestProcfs::statfs() {
1556                    Ok((_, flags)) => flags & libc::ST_NOEXEC != 0,
1557                    Err(errno) => return Some(Err(errno.into())),
1558                };
1559                rendered_mmap_answer(
1560                    c.addr().map_or(0, |addr| addr.as_raw()),
1561                    c.len(),
1562                    c.offset(),
1563                    c.prot(),
1564                    c.flags(),
1565                    noexec,
1566                )
1567            }
1568            Syscall::Fsync(_) | Syscall::Fdatasync(_) => Err(Errno::EINVAL),
1569            Syscall::Fchmod(_) => Err(Errno::EPERM),
1570            Syscall::Fgetxattr(c) => match c.name().map(|name| name.read(&guest.memory())) {
1571                Some(Ok(name)) => xattr_get_answer(name.as_bytes()),
1572                Some(Err(errno)) => Err(errno),
1573                None => Err(Errno::EFAULT),
1574            },
1575            Syscall::Flistxattr(_) => Ok(0),
1576            Syscall::Fsetxattr(c)
1577                if c.flags() & !(libc::XATTR_CREATE | libc::XATTR_REPLACE) != 0 =>
1578            {
1579                Err(Errno::EINVAL)
1580            }
1581            Syscall::Fsetxattr(c) => match c.name().map(|name| name.read(&guest.memory())) {
1582                Some(Ok(name)) => xattr_change_answer(
1583                    name.as_bytes(),
1584                    Some(c.size()),
1585                    read_value(&guest.memory(), args.arg2 as u64, c.size()).as_deref(),
1586                ),
1587                Some(Err(errno)) => Err(errno),
1588                None => Err(Errno::EFAULT),
1589            },
1590            Syscall::Fremovexattr(c) => match c.name().map(|name| name.read(&guest.memory())) {
1591                Some(Ok(name)) => xattr_change_answer(name.as_bytes(), None, None),
1592                Some(Err(errno)) => Err(errno),
1593                None => Err(Errno::EFAULT),
1594            },
1595            Syscall::Ioctl(_) => match args.arg1 as u32 as usize {
1596                // The generic file ioctls Linux answers before the file's own:
1597                // FIBMAP and FIFREEZE/FITHAW need capabilities, FS_IOC_FIEMAP
1598                // a file system that maps extents, and FICLONE/FICLONERANGE a
1599                // source on the same file system.
1600                1 | 0xc004_5877 | 0xc004_5878 => Err(Errno::EPERM),
1601                0xc020_660b => Err(Errno::EOPNOTSUPP),
1602                0x4004_9409 => Self::clone_ioctl_answer(guest, args.arg2 as i32),
1603                0x4020_940d => {
1604                    // The whole `struct file_clone_range` is read first.
1605                    match reverie::syscalls::Addr::<[u64; 4]>::from_raw(args.arg2)
1606                        .map(|address| guest.memory().read_value(address))
1607                    {
1608                        Some(Ok(range)) => Self::clone_ioctl_answer(guest, range[0] as i32),
1609                        _ => Err(Errno::EFAULT),
1610                    }
1611                }
1612                // FIONCLEX, FIOCLEX, FIOASYNC and FIONBIO, which the kernel
1613                // answers alike for both files.
1614                0x5450..=0x5452 | 0x5421 => return None,
1615                // FIONREAD: the file's size, 0, less the position, as an int.
1616                // Where the tool runs in each guest process, the kernel answers
1617                // from the empty placeholder's position, which Detcore keeps
1618                // current and which processes sharing the open file share.
1619                // Elsewhere Detcore answers from its own cursor, shared there
1620                // too (KVM's executor has no answer for a memfd).
1621                0x541b if self.cfg.backend.tool_shares_guest_descriptor_table => return None,
1622                0x541b => match AddrMut::<libc::c_int>::from_raw(args.arg2) {
1623                    Some(address) => {
1624                        let position = guest
1625                            .thread_state()
1626                            .with_detfd(args.arg0 as RawFd, |detfd| {
1627                                detfd.procfs_position().map_or(0, |(offset, _)| offset)
1628                            })
1629                            .unwrap_or(0);
1630                        let unread = 0_i64.wrapping_sub(position as i64) as libc::c_int;
1631                        guest.memory().write_value(address, &unread).map(|()| 0)
1632                    }
1633                    None => Err(Errno::EFAULT),
1634                },
1635                // FIOQSIZE: the bytes the file uses, none for procfs.
1636                0x5460 => match AddrMut::<libc::loff_t>::from_raw(args.arg2) {
1637                    Some(address) => guest.memory().write_value(address, &0).map(|()| 0),
1638                    None => Err(Errno::EFAULT),
1639                },
1640                // FIGETBSZ: procfs's block size.
1641                2 => match AddrMut::<libc::c_int>::from_raw(args.arg2) {
1642                    Some(address) => guest
1643                        .memory()
1644                        .write_value(address, &(procfs_inode::PROC_BLOCK_SIZE as libc::c_int))
1645                        .map(|()| 0),
1646                    None => Err(Errno::EFAULT),
1647                },
1648                _ => Err(Errno::ENOTTY),
1649            },
1650            Syscall::Fcntl(_) => match args.arg1 as libc::c_int {
1651                // Adding seals needs a writable descriptor, before Linux
1652                // looks for seals the file cannot have.
1653                libc::F_ADD_SEALS => Err(Errno::EPERM),
1654                libc::F_GET_SEALS => Err(Errno::EINVAL),
1655                // A lease needs the file's owner, so the file never has one.
1656                libc::F_SETLEASE => Err(Errno::EACCES),
1657                // F_SET_RW_HINT needs the file's owner.
1658                1036 => Err(Errno::EPERM),
1659                libc::F_GETLEASE => Ok(i64::from(libc::F_UNLCK)),
1660                // `O_NOATIME` needs the file's owner, and procfs sysctl
1661                // files cannot do direct I/O.
1662                libc::F_SETFL if args.arg2 as libc::c_int & libc::O_NOATIME != 0 => {
1663                    Err(Errno::EPERM)
1664                }
1665                libc::F_SETFL if args.arg2 as libc::c_int & libc::O_DIRECT != 0 => {
1666                    Err(Errno::EINVAL)
1667                }
1668                _ => return None,
1669            },
1670            _ => return None,
1671        };
1672        Some(answer.map_err(Into::into))
1673    }
1674}

1676/// What a path call asks of the file it names.
1677#[derive(Clone, Copy, Debug, PartialEq, Eq)]
1678enum PathRequest {
1679    /// `access`: `mode`, with `faccessat2`'s `flags`.
1680    Access {
1681        mode: u64,
1682        flags: u64,
1683    },
1684    /// `getxattr`, whose attribute name the answer reads.
1685    GetXattr,
1686    ListXattr,
1687    /// `setxattr` with the value's size, the value at guest address `value`,
1688    /// and its flags; or `removexattr`.
1689    ChangeXattr {
1690        size: Option<usize>,
1691        value: u64,
1692        flags: u64,
1693    },
1694    Chmod {
1695        flags: u64,
1696    },
1697    Truncate {
1698        length: i64,
1699    },
1700    /// Setting times: `None` for the current time, as when `times` is NULL.
1701    Times {
1702        explicit: Option<bool>,
1703    },
1704    Chdir,
1705    /// Arguments Linux refuses before it looks at the file.
1706    Invalid,
1707    /// Arguments Linux cannot read.
1708    Fault,
1709}
1710
1711/// Linux's answer to `request` for a rendered procfs file (0444, or 0644 for
1712/// a fixed tunable: only the owner may write either) owned by a user the
1713/// guest's user namespace does not map, given the attribute `name` when it
1714/// changes one.
1715fn path_request_answer(
1716    request: PathRequest,
1717    name: Option<&[u8]>,
1718    value: Option<&[u8]>,
1719) -> Result<i64, Errno> {
1720    match request {
1721        PathRequest::Access { mode, flags } => {
1722            if access_arguments_invalid(mode, flags) {
1723                Err(Errno::EINVAL)
1724            } else if mode & (libc::W_OK | libc::X_OK) as u64 != 0 {
1725                Err(Errno::EACCES)
1726            } else {
1727                Ok(0)
1728            }
1729        }
1730        PathRequest::GetXattr => xattr_get_answer(name.ok_or(Errno::EFAULT)?),
1731        PathRequest::ListXattr => Ok(0),
1732        PathRequest::ChangeXattr { size, flags, .. } => {
1733            if flags & !(libc::XATTR_CREATE | libc::XATTR_REPLACE) as u64 != 0 {
1734                return Err(Errno::EINVAL);
1735            }
1736            xattr_change_answer(name.ok_or(Errno::EFAULT)?, size, value)
1737        }
1738        PathRequest::Chmod { flags } => {
1739            if flags & !(libc::AT_SYMLINK_NOFOLLOW | libc::AT_EMPTY_PATH) as u64 != 0 {
1740                Err(Errno::EINVAL)
1741            } else {
1742                Err(Errno::EPERM)
1743            }
1744        }
1745        PathRequest::Truncate { length } if length < 0 => Err(Errno::EINVAL),
1746        PathRequest::Truncate { .. } => Err(Errno::EACCES),
1747        // Leaving both times alone changes nothing; the current time needs
1748        // write permission, and any other time the owner.
1749        PathRequest::Times { explicit: None } => Err(Errno::EACCES),
1750        PathRequest::Times {
1751            explicit: Some(false),
1752        } => Ok(0),
1753        PathRequest::Times {
1754            explicit: Some(true),
1755        } => Err(Errno::EPERM),
1756        PathRequest::Chdir => Err(Errno::ENOTDIR),
1757        PathRequest::Invalid => Err(Errno::EINVAL),
1758        PathRequest::Fault => Err(Errno::EFAULT),
1759    }
1760}

1762/// How `utimensat`'s `times` asks to change a file's times: `None` for the
1763/// current time (NULL, or both `UTIME_NOW`), `Some(false)` to change nothing
1764/// (both `UTIME_OMIT`), `Some(true)` for any other time, including one
1765/// `UTIME_NOW` with one `UTIME_OMIT`, which needs the owner.
1766fn utimens_request(times: Option<[libc::timespec; 2]>) -> Option<bool> {
1767    let times = times?;
1768    if times.iter().all(|time| time.tv_nsec == libc::UTIME_OMIT) {
1769        Some(false)
1770    } else if times.iter().all(|time| time.tv_nsec == libc::UTIME_NOW) {
1771        None
1772    } else {
1773        Some(true)
1774    }
1775}

1777/// Whether Linux refuses an access probe's mode or flags. It checks them
1778/// before it looks up the path (`do_faccessat`), with both already narrowed
1779/// to `int`.
1780fn access_arguments_invalid(mode: u64, flags: u64) -> bool {
1781    flags & !(libc::AT_EACCESS | libc::AT_SYMLINK_NOFOLLOW | libc::AT_EMPTY_PATH) as u64 != 0
1782        || mode & !7 != 0
1783}

1785impl<T: RecordOrReplay> Detcore<T> {
1786    /// Answer a path call (or an `AT_EMPTY_PATH`/NULL-path call on a
1787    /// descriptor) if it names a procfs file Detcore renders, as Linux
1788    /// answers for the procfs file: the backend has no such file, and the
1789    /// placeholder must not be changed or shown. A spelling Detcore does not
1790    /// resolve lexically that may be a descriptor link is resolved by the
1791    /// object it reaches.
1792    pub(crate) async fn answer_rendered_procfs_path_call<G: Guest<Self>>(
1793        &self,
1794        guest: &mut G,
1795        call: &Syscall,
1796    ) -> Result<Option<i64>, Error> {
1797        // A rendered descriptor stays the file after a chroot, so only the
1798        // path spellings below depend on the root.
1799        if self.cfg.recordreplay_modes {
1800            return Ok(None);
1801        }
1802        let (sysno, args) = call.into_parts();
1803        let [a0, a1, a2, a3, a4] =
1804            [args.arg0, args.arg1, args.arg2, args.arg3, args.arg4].map(|arg| arg as u64);
1805        let at = libc::AT_FDCWD as u64;
1806        // (dirfd, path, flags, request, attribute name pointer)
1807        // For `futimens` on an `O_PATH` descriptor: whether the call is done
1808        // before the descriptor is looked up (both times omitted), or refused
1809        // first (unknown flags).
1810        let mut decided_before_descriptor = false;
1811        let (dirfd, path, flags, request, name) = match sysno {
1812            // Linux declares mode and flags `int`: it ignores the upper bits
1813            // of the raw registers.
1814            Sysno::access => (
1815                at,
1816                a0,
1817                0,
1818                PathRequest::Access {
1819                    mode: u64::from(a1 as u32),
1820                    flags: 0,
1821                },
1822                0,
1823            ),
1824            Sysno::faccessat => (
1825                a0,
1826                a1,
1827                0,
1828                PathRequest::Access {
1829                    mode: u64::from(a2 as u32),
1830                    flags: 0,
1831                },
1832                0,
1833            ),
1834            Sysno::faccessat2 => (
1835                a0,
1836                a1,
1837                u64::from(a3 as u32),
1838                PathRequest::Access {
1839                    mode: u64::from(a2 as u32),
1840                    flags: u64::from(a3 as u32),
1841                },
1842                0,
1843            ),
1844            Sysno::getxattr => (at, a0, 0, PathRequest::GetXattr, a1),
1845            Sysno::lgetxattr => (at, a0, nofollow(), PathRequest::GetXattr, a1),
1846            Sysno::listxattr => (at, a0, 0, PathRequest::ListXattr, 0),
1847            Sysno::llistxattr => (at, a0, nofollow(), PathRequest::ListXattr, 0),
1848            Sysno::setxattr | Sysno::lsetxattr => (
1849                at,
1850                a0,
1851                if sysno == Sysno::lsetxattr {
1852                    nofollow()
1853                } else {
1854                    0
1855                },
1856                PathRequest::ChangeXattr {
1857                    size: Some(a3 as usize),
1858                    value: a2,
1859                    // Linux declares the flags `int`.
1860                    flags: u64::from(a4 as u32),
1861                },
1862                a1,
1863            ),
1864            Sysno::removexattr | Sysno::lremovexattr => (
1865                at,
1866                a0,
1867                if sysno == Sysno::lremovexattr {
1868                    nofollow()
1869                } else {
1870                    0
1871                },
1872                PathRequest::ChangeXattr {
1873                    size: None,
1874                    value: 0,
1875                    flags: 0,
1876                },
1877                a1,
1878            ),
1879            Sysno::chmod => (at, a0, 0, PathRequest::Chmod { flags: 0 }, 0),
1880            Sysno::fchmodat => (a0, a1, 0, PathRequest::Chmod { flags: 0 }, 0),
1881            Sysno::fchmodat2 => (a0, a1, a3, PathRequest::Chmod { flags: a3 }, 0),
1882            // Ownership changes keep Detcore's policy for every file
1883            // (`handle_ownership_change_noop`).
1884            Sysno::truncate => (at, a0, 0, PathRequest::Truncate { length: a1 as i64 }, 0),
1885            Sysno::utimensat => {
1886                // Unreadable times are the kernel's EFAULT to report.
1887                let Ok(times) = read_pair::<libc::timespec, _>(&guest.memory(), a2) else {
1888                    return Ok(None);
1889                };
1890                // Both times omitted is success before anything is looked at;
1891                // unknown flags, or any flag with a NULL path, are refused
1892                // before the file is looked up; out-of-range times after it
1893                // (measured on Linux 7.1). Detcore answers the first two for
1894                // every file where it renders procfs files: Linux's answer
1895                // does not depend on the file.
1896                let omitted = times
1897                    .is_some_and(|times| times.iter().all(|time| time.tv_nsec == libc::UTIME_OMIT));
1898                // A NULL path with flags is refused only for a descriptor:
1899                // with AT_FDCWD, Linux looks the NULL path up (EFAULT).
1900                let flags_valid = a3 & !((libc::AT_SYMLINK_NOFOLLOW | libc::AT_EMPTY_PATH) as u64)
1901                    == 0
1902                    && (a1 != 0 || a3 == 0 || a0 as i32 == libc::AT_FDCWD);
1903                if self.renders_procfs(guest) {
1904                    if omitted {
1905                        return Ok(Some(0));
1906                    }
1907                    if !flags_valid {
1908                        return Err(Errno::EINVAL.into());
1909                    }
1910                    // A NULL path with AT_FDCWD is looked up, and faults (KVM's
1911                    // executor answers otherwise).
1912                    if a1 == 0 && a0 as i32 == libc::AT_FDCWD {
1913                        return Err(Errno::EFAULT.into());
1914                    }
1915                }
1916                decided_before_descriptor = omitted || !flags_valid;
1917                let valid = omitted
1918                    || (flags_valid
1919                        && times.is_none_or(|times| times.iter().all(valid_utime_nsec)));
1920                (
1921                    a0,
1922                    a1,
1923                    a3,
1924                    if valid {
1925                        PathRequest::Times {
1926                            explicit: utimens_request(times),
1927                        }
1928                    } else {
1929                        PathRequest::Invalid
1930                    },
1931                    0,
1932                )
1933            }
1934            Sysno::utime => (
1935                at,
1936                a0,
1937                0,
1938                legacy_times_request(&guest.memory(), a1, false),
1939                0,
1940            ),
1941            Sysno::utimes => (
1942                at,
1943                a0,
1944                0,
1945                legacy_times_request(&guest.memory(), a1, true),
1946                0,
1947            ),
1948            // `futimesat` is ENOSYS under Detcore for every file.
1949            Sysno::chdir => (at, a0, 0, PathRequest::Chdir, 0),
1950            _ => return Ok(None),
1951        };
1952        // Linux refuses invalid access arguments before it looks up the path
1953        // or the descriptor, for every file; leave that order to the backend.
1954        if let PathRequest::Access { mode, flags } = request
1955            && access_arguments_invalid(mode, flags)
1956        {
1957            return Ok(None);
1958        }
1959        // Likewise invalid setxattr flags, before the name, the value or the
1960        // path, for every file (`setxattr_copy`). KVM's executor reads the
1961        // name first, so answer here rather than leave it to the backend.
1962        if let PathRequest::ChangeXattr { flags, .. } = request
1963            && flags & !(libc::XATTR_CREATE | libc::XATTR_REPLACE) as u64 != 0
1964            && self.renders_procfs(guest)
1965        {
1966            return Err(Errno::EINVAL.into());
1967        }
1968        let dirfd = dirfd as i32;
1969        let empty_path_allowed = flags & libc::AT_EMPTY_PATH as u64 != 0;
1970        // An unreadable path is the kernel's EFAULT to report.
1971        let path_text = match PathPtr::from_ptr(path as *const libc::c_char) {
1972            Some(path) => match path.read(&guest.memory()) {
1973                Ok(path) => Some(path),
1974                Err(_) => return Ok(None),
1975            },
1976            None => None,
1977        };
1978        // A descriptor named by an empty path (`AT_EMPTY_PATH`), or by none
1979        // (`futimens`).
1980        let by_descriptor = match &path_text {
1981            None => sysno == Sysno::utimensat,
1982            Some(path) => path.as_os_str().is_empty() && empty_path_allowed,
1983        };
1984        // Linux then copies and checks the attribute name, the size and the
1985        // value, still before it looks up the path (`setxattr_copy`,
1986        // `import_xattr_name`), so a path error cannot hide them. Detcore
1987        // checks them first where it may answer: a spelling that could name
1988        // a rendered file. Any other path goes to the kernel untouched.
1989        let xattr_arguments = if self.renders_procfs(guest)
1990            && (by_descriptor || path_text.as_deref().is_some_and(could_name_rendered))
1991        {
1992            checked_xattr_arguments(&guest.memory(), request, name)?
1993        } else {
1994            None
1995        };
1996        let rendered = if by_descriptor {
1997            let rendered = guest
1998                .thread_state()
1999                .with_detfd(dirfd, |detfd| {
2000                    detfd
2001                        .procfs_rendered_path()
2002                        .map(|path| (path, detfd.status_flags() & libc::O_PATH != 0))
2003                })
2004                .ok()
2005                .flatten();
2006            // `futimens` needs a descriptor opened for I/O: an `O_PATH` one
2007            // is EBADF, after the flags are checked and before the times are
2008            // (measured on Linux 7.1).
2009            if let Some((_, true)) = rendered
2010                && path_text.is_none()
2011                && !decided_before_descriptor
2012            {
2013                return Err(Errno::EBADF.into());
2014            }
2015            rendered.map(|(path, _)| path)
2016        } else {
2017            let Some(text) = path_text else {
2018                return Ok(None);
2019            };
2020            if text.as_os_str().is_empty() || !self.renders_procfs(guest) {
2021                return Ok(None);
2022            }
2023            // The spellings Detcore resolves itself, which end in a rendered
2024            // file's name or a descriptor number.
2025            let lexical = if could_name_rendered(&text) {
2026                let observed = Self::observed_path(guest, dirfd, &text)?;
2027                match self.rendered_procfs_at(guest, &observed).await? {
2028                    // A call that does not follow a descriptor link acts on
2029                    // the link itself, which the kernel has.
2030                    Some(target) if target.via_link && flags & nofollow() != 0 => return Ok(None),
2031                    Some(target) => target.procfs.rendered_path(),
2032                    None => None,
2033                }
2034            } else {
2035                None
2036            };
2037            match lexical {
2038                Some(rendered) => Some(rendered),
2039                // Any other spelling (a symbolic link to a descriptor link,
2040                // `/dev/stdin`): what the kernel reaches, by its identity.
2041                // Only while this process holds a rendered descriptor, so a
2042                // process that holds none pays nothing.
2043                None => {
2044                    self.placeholder_reached(
2045                        guest,
2046                        dirfd,
2047                        PathPtr::from_ptr(path as *const libc::c_char),
2048                        flags & nofollow() == 0,
2049                    )
2050                    .await
2051                }
2052            }
2053        };
2054        if rendered.is_none() {
2055            return Ok(None);
2056        }
2057        // A spelling that reached a rendered file another way (a link) has
2058        // its attribute arguments checked here, after the path.
2059        let xattr_arguments = match xattr_arguments {
2060            Some(arguments) => Some(arguments),
2061            None => checked_xattr_arguments(&guest.memory(), request, name)?,
2062        };
2063        let (name, value) = match xattr_arguments {
2064            Some((name, value)) => (Some(name), value),
2065            None => (None, None),
2066        };
2067        path_request_answer(request, name.as_deref(), value.as_deref())
2068            .map(Some)
2069            .map_err(Into::into)
2070    }
2071}

2073/// An extended attribute's name, and the value a change sets.
2074type XattrArguments = (Vec<u8>, Option<Vec<u8>>);

2076/// An extended-attribute call's name and value, checked as Linux checks them
2077/// before the path (`setxattr_copy`, `import_xattr_name`): the name (EFAULT,
2078/// or ERANGE when empty or longer than `XATTR_NAME_MAX`), then the size
2079/// (E2BIG), then the value (EFAULT). None for any other request.
2080fn checked_xattr_arguments<M: MemoryAccess>(
2081    memory: &M,
2082    request: PathRequest,
2083    name: u64,
2084) -> Result<Option<XattrArguments>, Errno> {
2085    if !matches!(
2086        request,
2087        PathRequest::GetXattr | PathRequest::ChangeXattr { .. }
2088    ) {
2089        return Ok(None);
2090    }
2091    let name = read_xattr_name(memory, name)?;
2092    if name.is_empty() || name.len() > XATTR_NAME_MAX {
2093        return Err(Errno::ERANGE);
2094    }
2095    let value = match request {
2096        PathRequest::ChangeXattr {
2097            size: Some(size),
2098            value,
2099            ..
2100        } => {
2101            if size > XATTR_SIZE_MAX {
2102                return Err(Errno::E2BIG);
2103            }
2104            let read = read_value(memory, value, size);
2105            if size > 0 && read.is_none() {
2106                return Err(Errno::EFAULT);
2107            }
2108            read
2109        }
2110        _ => None,
2111    };
2112    Ok(Some((name, value)))
2113}

2115/// An extended attribute's name at guest address `address`, imported as
2116/// Linux does (`import_xattr_name`): at most `XATTR_NAME_MAX + 1` bytes are
2117/// read, so a name with no NUL within them is too long (ERANGE) even when
2118/// the bytes after them cannot be read, and only a fault within them is
2119/// EFAULT. The answer checks the length.
2120///
2121/// Each read stays within one page and uses the guest's own access rights
2122/// (`read_exact_with_user_access`): an exact read that faults only when a
2123/// needed byte does, as the kernel's copy faults.
2124fn read_xattr_name<M: MemoryAccess>(memory: &M, address: u64) -> Result<Vec<u8>, Errno> {
2125    const PAGE: u64 = 4096;
2126    let mut name = Vec::new();
2127    let limit = XATTR_NAME_MAX + 1;
2128    while name.len() < limit {
2129        let start = address
2130            .checked_add(name.len() as u64)
2131            .ok_or(Errno::EFAULT)?;
2132        let page_left = (PAGE - start % PAGE) as usize;
2133        let mut chunk = vec![0u8; page_left.min(limit - name.len())];
2134        let chunk_address =
2135            reverie::syscalls::Addr::<u8>::from_raw(start as usize).ok_or(Errno::EFAULT)?;
2136        memory
2137            .read_exact_with_user_access(chunk_address, &mut chunk)
2138            .map_err(|_| Errno::EFAULT)?;
2139        match chunk.iter().position(|byte| *byte == 0) {
2140            Some(end) => {
2141                name.extend_from_slice(&chunk[..end]);
2142                return Ok(name);
2143            }
2144            None => name.extend_from_slice(&chunk),
2145        }
2146    }
2147    Ok(name)
2148}

2155/// Whether `path` could be a spelling Detcore resolves to a rendered file

2160        return false;

2164        || (!last.is_empty() && last.iter().all(u8::is_ascii_digit))

2167/// An attribute value of `size` bytes at guest address `address` (at most
2168/// `XATTR_SIZE_MAX`, which the answer checks first), if the guest can read it.
2169fn read_value<M: MemoryAccess>(memory: &M, address: u64, size: usize) -> Option<Vec<u8>> {
2170    let size = size.min(XATTR_SIZE_MAX);
2171    if size == 0 {
2172        return Some(Vec::new());
2173    }
2174    let address = reverie::syscalls::Addr::<u8>::from_raw(address as usize)?;
2175    let mut value = vec![0; size];
2176    // Exact, with the guest's own access rights, as the kernel copies it.
2177    memory
2178        .read_exact_with_user_access(address, &mut value)

2183/// What `utime` (a `struct utimbuf`) or `utimes`/`futimesat` (two `struct
2184/// timeval`, when `timevals`) asks of a file's times at guest address
2185/// `times`: NULL is the current time; the times are read, and for `timeval`s
2186/// their microseconds checked, before the file is.
2187fn legacy_times_request<M: MemoryAccess>(memory: &M, times: u64, timevals: bool) -> PathRequest {
2188    if times == 0 {
2189        return PathRequest::Times { explicit: None };
2190    }
2191    if !timevals {
2192        // A `struct utimbuf`: two `time_t`, with nothing to check.
2193        return match read_pair::<libc::time_t, M>(memory, times) {
2194            Err(_) => PathRequest::Fault,
2195            Ok(_) => PathRequest::Times {
2196                explicit: Some(true),
2197            },
2198        };
2199    }
2200    match read_pair::<libc::timeval, M>(memory, times) {
2201        Err(_) => PathRequest::Fault,
2202        Ok(Some(values))
2203            if values
2204                .iter()
2205                .any(|value| !(0..1_000_000).contains(&value.tv_usec)) =>
2206        {
2207            PathRequest::Invalid
2208        }
2209        Ok(_) => PathRequest::Times {
2210            explicit: Some(true),
2211        },
2212    }
2213}
