62const SECCOMP_GET_ACTION_AVAIL: u32 = 2;

66fn seccomp_result(op: u32, flags: u32, has_args: bool) -> Result<i64, Errno> {
67    if op > SECCOMP_GET_NOTIF_SIZES {
68        return Err(Errno::EINVAL);
69    }
70    if op == SECCOMP_SET_MODE_STRICT && (flags != 0 || has_args) {
71        return Err(Errno::EINVAL);
72    }
73    if op == SECCOMP_SET_MODE_FILTER && flags & !SECCOMP_FILTER_FLAG_TSYNC != 0 {
74        return Err(Errno::EINVAL);
75    }
76    if matches!(
77        op,
78        SECCOMP_SET_MODE_FILTER | SECCOMP_GET_ACTION_AVAIL | SECCOMP_GET_NOTIF_SIZES
79    ) && !has_args
80    {
81        return Err(Errno::EFAULT);
82    }
83
84    // Hermit cannot enforce a guest-installed BPF policy in every backend.
85    // Report the operation as unsupported instead of claiming a filter exists.
86    Err(Errno::EOPNOTSUPP)
87}

89fn is_supported_prctl_option(option: libc::c_int) -> bool {
90    matches!(
91        option,
92        libc::PR_SET_NAME
93            | libc::PR_GET_NAME
94            | libc::PR_SET_THP_DISABLE
95            | libc::PR_GET_THP_DISABLE
96            // AUTONOMOUS-BOT-IMPLEMENTED
97            // TODO-HUMAN-REVIEW(#919)
98            //
99            // Dumpability is per-process state initialized deterministically by
100            // ordinary exec and subsequently controlled only by the guest. Keep
101            // Linux's 0/1 validation and state transitions rather than refusing
102            // applications that explicitly disable or restore core-dump access.
103            | libc::PR_SET_DUMPABLE
104            | libc::PR_GET_DUMPABLE
105            // PR_SET_TIMERSLACK and PR_GET_TIMERSLACK are deliberately absent:
106            // `handle_prctl` emulates them against per-thread Detcore state.
107            // AUTONOMOUS-BOT-IMPLEMENTED
108            // TODO-HUMAN-REVIEW(#802)
109            //
110            // PR_{SET,GET}_KEEPCAPS only read/toggle the calling thread's
111            // "keep capabilities across a UID change" flag. The result is a pure
112            // function of the guest's own prior prctl calls (0/1), never host
113            // state, so passthrough is deterministic and bitwise-identical across
114            // runs. Supporting it lets `setpriv` (and the `date`/privilege-drop
115            // wrappers that call it) run under --strict instead of aborting with
116            // "keep process capabilities failed: Function not implemented".
117            | libc::PR_SET_KEEPCAPS
118            | libc::PR_GET_KEEPCAPS
119            // AUTONOMOUS-BOT-IMPLEMENTED
120            // TODO-HUMAN-REVIEW(#824)
121            //
122            // PR_{SET,GET}_PDEATHSIG only read/toggle the calling thread's
123            // parent-death-signal attribute. PR_SET_PDEATHSIG validates its
124            // signal argument (a valid signal or 0 succeeds, anything else
125            // faults EINVAL) and PR_GET_PDEATHSIG reports the value the guest
126            // previously set. The result is a pure function of the guest's own
127            // prior prctl calls and its argument, never host state, so
128            // passthrough is deterministic and bitwise-identical across runs.
129            // The registered signal fires when the parent thread exits, and the
130            // kernel posts it at a moment set by host timing, so `handle_prctl`
131            // first records a nonzero PR_SET_PDEATHSIG signal as host-timed for
132            // the caller's process: no gated wait of that process ends for it
133            // (https://github.com/rrnewton/hermit/issues/3146). Supporting it lets
134            // `setpriv --pdeathsig` run under --strict instead of aborting with
135            // "set parent death signal failed: Function not implemented".
136            | libc::PR_SET_PDEATHSIG
137            | libc::PR_GET_PDEATHSIG
138    )
139}

147/// Whether Detcore can admit `PR_SET_CHILD_SUBREAPER`: threads are
148/// sequentialized, so every exit reaches the scheduler through
149/// `deregister_thread` and re-parents there (`Scheduler::reparent_orphans`);
150/// without that, Detcore's record never learns the orphan moved while it still
151/// decides ECHILD. And on `backend`, its own record of children decides waits
152/// (`tracks_process_children`), the kernel rather than the backend performs
153/// them (`!emulates_child_waits`), and a process's exit reaches Detcore only
154/// after the kernel's `exit_notify` has re-parented its children (neither
155/// asynchronous nor separately reported physical exits), so the record agrees
156/// with the kernel. Today that is ptrace, and e9patch preprocessing run under
157/// ptrace, in a sequentialized run.
158fn models_child_subreapers(sequentialize_threads: bool, backend: &BackendCapabilities) -> bool {
159    sequentialize_threads
160        && backend.tracks_process_children
161        && !backend.emulates_child_waits
162        && !backend.process_exits_complete_asynchronously
163        && !backend.reports_physical_process_exits
164}

166/// Is `which` one of the Linux `PRIO_*` target selectors for get/setpriority?
167fn is_valid_prio_which(which: i32) -> bool {
168    which == libc::PRIO_PROCESS as i32
169        || which == libc::PRIO_PGRP as i32
170        || which == libc::PRIO_USER as i32
171}

175/// Deterministic raw `getpriority(2)` result under Hermit's inert-nice model.
176///
177/// Returns `20 - nice` (nice 0 -> 20) for any valid target regardless of `who`,
178/// because nice is inert under the virtualized scheduler and `getpriority` never

182        Ok(20)
183    } else {
184        Err(Errno::EINVAL)

212/// Write get_mempolicy's output as Linux's `put_user`/`copy_to_user` do:
213/// through the guest's page protections, so a buffer that cannot take all the
214/// bytes is `EFAULT`. Any other failure is the backend's, not a guest fault,
215/// and is reported as a tool error rather than handed to the guest as an
216/// errno get_mempolicy never returns (Claude re-check of
217/// https://github.com/rrnewton/hermit/pull/4249).
218fn write_get_mempolicy_output<M: MemoryAccess>(
219    memory: &mut M,
220    address: AddrMut<'_, u8>,
221    bytes: &[u8],
222) -> Result<(), Error> {
223    match memory.write_with_user_access(address, bytes) {
224        Ok(written) if written == bytes.len() => Ok(()),
225        Ok(_) | Err(Errno::EFAULT) => Err(Errno::EFAULT.into()),
226        Err(errno) => Err(Error::Tool(anyhow::anyhow!(
227            "get_mempolicy output write failed: {errno}"
228        ))),
229    }
230}

232fn from_str(s: &str) -> [i8; 65] {
233    let mut ret: [i8; 65] = [0; 65];
234    for (i, ch) in s.bytes().take(64).enumerate() {
235        ret[i] = ch as i8;
236    }
237    ret
238}

240const RANDOM_DEVICE_BYTE_STRIDE: u8 = 73;
241const RANDOM_DEVICE_FIRST_BYTE: u8 = 41;

243// AUTONOMOUS-BOT-IMPLEMENTED
244// TODO-HUMAN-REVIEW(PR-1096): Review the backend-independent random-device stream.
245fn canonical_random_device_byte(seed: u64, index: u64) -> u8 {
246    let seed_byte = seed.rotate_right(((index % 8) * 8) as u32) as u8;
247    (index as u8)
248        .wrapping_mul(RANDOM_DEVICE_BYTE_STRIDE)
249        .wrapping_add(RANDOM_DEVICE_FIRST_BYTE)
250        ^ seed_byte
251}

253/// Scatter one stream through an already-imported array. Scratch space is
254/// independent of request size; only EFAULT becomes a successful copied prefix.
255fn fill_canonical_random_iovecs(
256    memory: &mut impl MemoryAccess,
257    iovecs: &[crate::iovecs::ImportedIovec],
258    seed: u64,
259    stream_offset: u64,
260    hasher: &mut DefaultHasher,
261) -> Result<usize, Error> {
262    let mut local_words = [0_u64; RANDOM_FILL_CHUNK_BYTES / std::mem::size_of::<u64>()];
263    let mut written = 0_usize;
264    for iov in iovecs {
265        let mut segment_written = 0;
266        while segment_written < iov.len {
267            let remote_chunk = match iov
268                .base
269                .checked_add(segment_written)
270                .and_then(AddrMut::<u8>::from_raw)
271            {
272                Some(address) => address,
273                None if written == 0 => return Err(Errno::EFAULT.into()),
274                None => return Ok(written),
275            };
276            let chunk_len = (iov.len - segment_written).min(RANDOM_FILL_CHUNK_BYTES);
277            let local_buf = unsafe {
278                std::slice::from_raw_parts_mut(local_words.as_mut_ptr().cast::<u8>(), chunk_len)
279            };
280            for (index, byte) in local_buf.iter_mut().enumerate() {
281                *byte = canonical_random_device_byte(
282                    seed,
283                    stream_offset
284                        .saturating_add(written as u64)
285                        .saturating_add(index as u64),
286                );
287            }
288            let n = match write_random_chunk(memory, remote_chunk, local_buf) {
289                Ok(n) => n,
290                Err(Errno::EFAULT) if written > 0 => return Ok(written),
291                Err(error) => return Err(crate::random::copy_error(error)),
292            };
293            if n == 0 && written == 0 {
294                return Err(Errno::EFAULT.into());
295            }
296            if cfg!(debug_assertions) {
297                Hash::hash_slice(&local_buf[..n], hasher);
298            }
299            written += n;
300            segment_written += n;
301            if n < chunk_len {
302                return Ok(written);
303            }
304        }
305    }
306    Ok(written)
307}

331    // AUTONOMOUS-BOT-IMPLEMENTED
332    // TODO-HUMAN-REVIEW(#539): Confirm the virtual arch_prctl control policy.
333    /// Preserve thread-local bases while hiding host CPU feature controls.
334    pub async fn handle_arch_prctl<G: Guest<Self>>(
335        &self,
336        guest: &mut G,
337        call: syscalls::ArchPrctl,
338    ) -> Result<i64, Error> {
339        let cpuid_uses_backend_policy =
340            self.cfg.virtualize_cpuid && self.cfg.backend.virtualizes_cpuid;
341        let cpuid_uses_faulting = self.cfg.virtualize_cpuid && guest.has_cpuid_interception();
342        match call.cmd() {
343            ArchPrctlCmd::ARCH_SET_FS(_)
344            | ArchPrctlCmd::ARCH_SET_GS(_)
345            | ArchPrctlCmd::ARCH_GET_FS(_)
346            | ArchPrctlCmd::ARCH_GET_GS(_) => Ok(guest.inject(call).await?),
347
348            // KVM installs a deterministic CPUID table while leaving the instruction enabled.
349            ArchPrctlCmd::ARCH_GET_CPUID(_) if cpuid_uses_backend_policy => Ok(1),
350            ArchPrctlCmd::ARCH_SET_CPUID(value) if cpuid_uses_backend_policy => {
351                if value == 0 {
352                    Err(Errno::EPERM.into())
353                } else {
354                    Ok(0)
355                }
356            }
357
358            // When Reverie successfully disables native CPUID, Detcore answers its fault from a
359            // fixed table. Preserve that backend state and reject attempts to re-enable CPUID.
360            ArchPrctlCmd::ARCH_GET_CPUID(_) if cpuid_uses_faulting => Ok(0),
361            ArchPrctlCmd::ARCH_SET_CPUID(value) if cpuid_uses_faulting => {
362                if value == 0 {
363                    Ok(0)
364                } else {
365                    Err(Errno::EPERM.into())
366                }
367            }
368            // Reverie cannot faithfully deliver a CPUID fault requested by the tracee. In
369            // explicit host-CPUID mode, expose a fixed enabled control state and reject disable.
370            ArchPrctlCmd::ARCH_GET_CPUID(_) if !self.cfg.virtualize_cpuid => Ok(1),
371            ArchPrctlCmd::ARCH_SET_CPUID(value) if !self.cfg.virtualize_cpuid => {
372                if value == 0 {
373                    Err(Errno::EPERM.into())
374                } else {
375                    Ok(0)
376                }
377            }
378
379            // Ptrace hosts without CPUID-faulting support retain the kernel's honest state.
380            ArchPrctlCmd::ARCH_GET_CPUID(_) | ArchPrctlCmd::ARCH_SET_CPUID(_) => {
381                Ok(guest.inject(call).await?)
382            }
383
384            // Expose a conservative virtual CPU with no optional extended-state permissions.
385            ArchPrctlCmd::Other(
386                ARCH_GET_XCOMP_SUPP | ARCH_GET_XCOMP_PERM | ARCH_GET_XCOMP_GUEST_PERM,
387                addr,
388            ) => self.write_arch_prctl_u64(guest, addr, 0),
389            ArchPrctlCmd::Other(ARCH_REQ_XCOMP_PERM | ARCH_REQ_XCOMP_GUEST_PERM, _) => {
390                Err(Errno::EINVAL.into())
391            }
392
393            // Keep shadow stacks disabled in the virtual policy. Disabling an already-disabled
394            // feature is idempotent; enable/lock/unlock requests cannot be honored.
395            ArchPrctlCmd::Other(ARCH_SHSTK_STATUS, addr) => {
396                self.write_arch_prctl_u64(guest, addr, 0)
397            }
398            ArchPrctlCmd::Other(ARCH_SHSTK_DISABLE, features)
399                if features != 0 && features & !ARCH_SHSTK_VALID_MASK == 0 =>
400            {
401                Ok(0)
402            }
403            ArchPrctlCmd::Other(ARCH_SHSTK_DISABLE, _)
404            | ArchPrctlCmd::Other(ARCH_SHSTK_ENABLE | ARCH_SHSTK_LOCK | ARCH_SHSTK_UNLOCK, _) => {
405                Err(Errno::EINVAL.into())
406            }
407
408            ArchPrctlCmd::Other(_, _) => Err(Errno::EINVAL.into()),
409        }
410    }

412    // AUTONOMOUS-BOT-IMPLEMENTED
413    // TODO-HUMAN-REVIEW(#663)
414    /// Preserve deterministic Ruby thread controls, report the container's fixed
415    /// capability bounding set, and reject options that expose unmodeled process
416    /// or host state.
417    pub async fn handle_prctl<G: Guest<Self>>(
418        &self,
419        guest: &mut G,
420        call: syscalls::Prctl,
421    ) -> Result<i64, Error> {
422        match call.option() {
423            // The capability bounding set is fixed by the container launch policy.
424            libc::PR_CAPBSET_READ => Ok(self.record_or_replay(guest, call).await?),
425            // AUTONOMOUS-BOT-IMPLEMENTED
426            // TODO-HUMAN-REVIEW(PR-2150): Timer slack is shared with the
427            // virtual `/proc/<tid>/timerslack_ns` channel.  Never pass either
428            // setter through to the physical tracee: a large physical slack
429            // changes wake timing on Detcore's remaining host-timed waits.
430            libc::PR_SET_TIMERSLACK => {
431                let requested = call.arg2();
432                let state = guest.thread_state_mut();
433                // Linux treats zero as reset-to-default. Hermit virtualizes the
434                // Linux scheduling policy as SCHED_OTHER, so the kernel's
435                // RT/DL no-op branch is unreachable in the guest model.
436                state.timer_slack_ns = if requested == 0 {
437                    state.default_timer_slack_ns
438                } else {
439                    requested
440                };
441                Ok(0)
442            }
443            libc::PR_GET_TIMERSLACK => Ok(guest.thread_state().timer_slack_ns as i64),
444            // The kernel sends a parent-death signal process-directed
445            // (`group_send_sig_info` with `PIDTYPE_TGID`) when the parent thread
446            // exits, at a moment set by host timing. Record it in this turn,
447            // before the call reaches the kernel, for every thread of the
448            // caller's process, and never forget it: a later disarm or `exec`
449            // does not remove it (`Scheduler::record_host_timed_signals`). Linux
450            // accepts 0 (disarm) and 1 to 64; anything else fails with EINVAL in
451            // the kernel and arms nothing.
452            libc::PR_SET_PDEATHSIG => {
453                let signal = call.arg2();
454                if signal == libc::SIGALRM as libc::c_ulong {
455                    refuse_sigalrm(guest, SigalrmControl::ArmProducer).await?;
456                }
457                if (1..=64).contains(&signal)
458                    && guest
459                        .config()
460                        .backend_supports_blocked_wait_signal_interruption
461                {
462                    record_host_timed_signals(
463                        guest,
464                        HostTimedSignalScope::Caller,
465                        1_u64 << (signal - 1),
466                    )
467                    .await;
468                }
469                self.passthrough(guest, call.into()).await
470            }
471            // AUTONOMOUS-BOT-IMPLEMENTED
472            // TODO-HUMAN-REVIEW(PR-4000)
473            //
474            // The subreaper flag is per-process state only the guest sets, so
475            // the kernel call is deterministic. It is injected in replay too,
476            // like setpgid, rather than replayed from the log: the kernel must
477            // hold the flag for orphans to become this process's children, so
478            // that the waits Detcore injects for them succeed. On success
479            // Detcore mirrors it in its thread tree, in this turn, so that an
480            // orphan re-parented to this process becomes its child for
481            // Detcore's waits too (`Scheduler::reparent_orphans`). The process
482            // is the one Detcore registered the calling thread in, resolved
483            // before the call: a raw clone child with a non-SIGCHLD exit
484            // signal is its own process there, although the backend reports
485            // its creator's pid. A thread with no registered process is
486            // refused before any effect. Linux treats any nonzero argument as
487            // "on". PR_GET_CHILD_SUBREAPER reads the caller's own flag back
488            // from the kernel.
489            libc::PR_SET_CHILD_SUBREAPER
490                if models_child_subreapers(
491                    guest.config().sequentialize_threads,
492                    &guest.config().backend,
493                ) =>
494            {
495                let dettid = guest.thread_state().dettid;
496                let Some(process) = registered_process_of(guest, dettid).await else {
497                    return Err(Errno::ENOSYS.into());
498                };
499                let result = guest.inject(call).await?;
500                set_child_subreaper(guest, process, call.arg2() != 0).await;
501                Ok(result)
502            }
503            libc::PR_GET_CHILD_SUBREAPER
504                if models_child_subreapers(
505                    guest.config().sequentialize_threads,
506                    &guest.config().backend,
507                ) =>
508            {
509                Ok(guest.inject(call).await?)
510            }
511            option
512                if guest.config().backend.virtualizes_capability_prctls
513                    && is_backend_virtualized_capability_prctl(option) =>
514            {
515                self.passthrough(guest, call.into()).await
516            }
517            option if is_supported_prctl_option(option) => {
518                self.passthrough(guest, call.into()).await
519            }
520            _ => Err(Errno::ENOSYS.into()),
521        }
522    }

698    /// setpgid system call. The kernel remains authoritative for validation;
699    /// after success Detcore mirrors the guest-visible process-group change so
700    /// group-selecting waits do not consult host `/proc` state.
701    pub async fn handle_setpgid<G: Guest<Self>>(
702        &self,
703        guest: &mut G,
704        call: syscalls::Setpgid,
705    ) -> Result<i64, Error> {
706        let res = guest.inject(call).await?;
707        let caller = guest.thread_state().detpid.expect("detpid unset");
708        // A nested caller's ids are its own PID namespace's; the kernel took
709        // them so, and the model records the tasks they name
710        // (https://github.com/rrnewton/hermit/issues/4155).
711        let process = if call.pid() == 0 {
712            caller
713        } else {
714            self.task_named_after_success(guest, Sysno::setpgid, call.pid())
715                .await?
716        };
717        let group = if call.pgid() == 0 {
718            process
719        } else {
720            self.task_named_after_success(guest, Sysno::setpgid, call.pgid())
721                .await?
722        };
723        let _ = set_process_group(guest, process, group).await;
724        Ok(res)
725    }

727    /// The task a positive id the calling thread passed names, after the
728    /// kernel accepted it: the id itself at depth 0, translated from the
729    /// caller's PID namespace below that. An id the record cannot name is
730    /// refused by name; the kernel found it, so the record is incomplete.
731    async fn task_named_after_success<G: Guest<Self>>(
732        &self,
733        guest: &mut G,
734        sysno: Sysno,
735        id: libc::pid_t,
736    ) -> Result<DetPid, Error> {
737        if !guest.thread_state().nested_pid_namespace {
738            return Ok(DetPid::from_raw(id));
739        }
740        let reason = match crate::tool_global::guest_pid(guest, id).await {
741            crate::scheduler::pid_namespaces::GuestPidAnswer::Task { task, .. } => return Ok(task),
742            crate::scheduler::pid_namespaces::GuestPidAnswer::NoSuchPid => {
743                format!(
744                    "the kernel accepted id {id}, which no recorded task of the caller's namespace has"
745                )
746            }
747            crate::scheduler::pid_namespaces::GuestPidAnswer::Untranslatable(reason) => reason,
748        };
749        tracing::error!(
750            "[detcore, dtid {}] {} from a nested PID namespace is not supported: {} (https://github.com/rrnewton/hermit/issues/4155)",
751            guest.thread_state().dettid,
752            sysno,
753            reason,
754        );
755        match self
756            .refuse_unserviceable_operation(guest, sysno, Errno::ENOSYS)
757            .await
758        {
759            Err(error) => Err(error),
760            Ok(_) => Err(Errno::ENOSYS.into()),
761        }
762    }

764    /// membarrier (system call).
765    ///
766    /// `membarrier(2)` issues process-wide memory barriers so that userspace can
767    /// use asymmetric fences (e.g. CPython's QSBR, RCU-style reclamation).
768    /// Detcore serializes all guest threads onto a single logical CPU with a
769    /// total memory order, so any requested barrier is *already* satisfied and
770    /// every command is a deterministic no-op. For `MEMBARRIER_CMD_QUERY` we
771    /// report the set of commands we emulate so the guest stays on this
772    /// controlled path instead of a host-dependent fallback; every other command
773    /// returns success without doing anything.
774    pub async fn handle_membarrier<G: Guest<Self>>(
775        &self,
776        guest: &mut G,
777        call: syscalls::Membarrier,
778    ) -> Result<i64, Error> {
779        // Values from <linux/membarrier.h>.
780        const MEMBARRIER_CMD_QUERY: i32 = 0;
781        const MEMBARRIER_CMD_GLOBAL: i32 = 1 << 0;
782        const MEMBARRIER_CMD_GLOBAL_EXPEDITED: i32 = 1 << 1;
783        const MEMBARRIER_CMD_REGISTER_GLOBAL_EXPEDITED: i32 = 1 << 2;
784        const MEMBARRIER_CMD_PRIVATE_EXPEDITED: i32 = 1 << 3;
785        const MEMBARRIER_CMD_REGISTER_PRIVATE_EXPEDITED: i32 = 1 << 4;
786        const SUPPORTED: i32 = MEMBARRIER_CMD_GLOBAL
787            | MEMBARRIER_CMD_GLOBAL_EXPEDITED
788            | MEMBARRIER_CMD_REGISTER_GLOBAL_EXPEDITED
789            | MEMBARRIER_CMD_PRIVATE_EXPEDITED
790            | MEMBARRIER_CMD_REGISTER_PRIVATE_EXPEDITED;
791
792        let cmd = call.cmd();
793        if cmd == MEMBARRIER_CMD_QUERY {
794            detlog!(
795                "[dtid {}] membarrier(QUERY) => reporting emulated commands {:#x}",
796                guest.thread_state().dettid,
797                SUPPORTED,
798            );
799            Ok(SUPPORTED as i64)
800        } else {
801            detlog!(
802                "[dtid {}] membarrier(cmd={}) no-op (threads are serialized on one CPU)",
803                guest.thread_state().dettid,
804                cmd,
805            );
806            Ok(0)
807        }
808    }

878    /// get_mempolicy under Hermit. The container exposes a single virtual NUMA
879    /// node, so the effective policy is always the default and every address
880    /// resolves to node 0. The result is fully emulated (never injected), so it
881    /// is bitwise-identical across the two --verify runs and under record/replay,
882    /// removing the host-NUMA-topology dependence a passthrough would introduce.
883    ///
884    /// It follows Linux's `kernel_get_mempolicy` and `do_get_mempolicy`
885    /// (mm/mempolicy.c) for a kernel with one node (`nr_node_ids` 1):
886    /// - a node mask with `maxnode` below 1 is EINVAL;
887    /// - an unknown flag (any bit beyond `MPOL_F_NODE`, `MPOL_F_ADDR` and
888    ///   `MPOL_F_MEMS_ALLOWED`, including those above bit 31 of the raw
889    ///   argument, which Reverie's typed call drops), `MPOL_F_MEMS_ALLOWED`
890    ///   with `MPOL_F_NODE` or `MPOL_F_ADDR`, and an address without
891    ///   `MPOL_F_ADDR` are EINVAL;
892    /// - `*policy` is 0: `MPOL_DEFAULT`, or node 0 for `MPOL_F_NODE`;
893    /// - the node mask is written as `copy_nodes_to_user` writes it: node 0
894    ///   alone for `MPOL_F_MEMS_ALLOWED`, as `/proc/<pid>/status` reports
895    ///   `Mems_allowed_list` (<https://github.com/rrnewton/hermit/issues/4214>),
896    ///   and no node for the default policy, in `ALIGN(maxnode - 1, 64) / 8`
897    ///   bytes (none for a `maxnode` of 1), EINVAL beyond a page;
898    /// - both are written with the guest's own access rights, so a read-only or
899    ///   unmapped buffer is EFAULT, as from `put_user` and `copy_to_user`.
900    ///
901    /// Limits: `set_mempolicy` and `mbind` succeed without recording a policy,
902    /// so this always reports the default policy, where Linux reports, say,
903    /// `MPOL_INTERLEAVE` after `set_mempolicy(MPOL_INTERLEAVE)`. For the same
904    /// reason `MPOL_F_NODE` without `MPOL_F_ADDR` answers node 0, as Linux does
905    /// for an interleave policy (libnuma's `numa_get_interleave_node`), where
906    /// it refuses it under the default policy. `MPOL_F_ADDR` does not check
907    /// that the address is mapped, where Linux answers EFAULT for an unmapped
908    /// one.
909    // AUTONOMOUS-BOT-IMPLEMENTED
910    // TODO-HUMAN-REVIEW(#720)
911    pub async fn handle_get_mempolicy<G: Guest<Self>>(
912        &self,
913        guest: &mut G,
914        call: syscalls::GetMempolicy,
915    ) -> Result<i64, Error> {
916        const MPOL_F_NODE: u64 = 1;
917        const MPOL_F_ADDR: u64 = 2;
918        const MPOL_F_MEMS_ALLOWED: u64 = 4;
919        const PAGE: u64 = 4096;
920        let mask = call.nodemask();
921        if mask.is_some() && call.maxnode() < 1 {
922            return Err(Errno::EINVAL.into());
923        }
924        // The flags argument is an unsigned long; Reverie decodes it as u32.
925        let flags = guest.regs().await.r8;
926        if flags & !(MPOL_F_NODE | MPOL_F_ADDR | MPOL_F_MEMS_ALLOWED) != 0 {
927            return Err(Errno::EINVAL.into());
928        }
929        let mems_allowed = flags & MPOL_F_MEMS_ALLOWED != 0;
930        if mems_allowed {
931            if flags & (MPOL_F_NODE | MPOL_F_ADDR) != 0 {
932                return Err(Errno::EINVAL.into());
933            }
934        } else if flags & MPOL_F_ADDR == 0 && call.addr().is_some() {
935            return Err(Errno::EINVAL.into());
936        }
937        if let Some(policy) = call.policy() {
938            let policy = AddrMut::<u8>::from_raw(policy.as_raw()).ok_or(Errno::EFAULT)?;
939            write_get_mempolicy_output(&mut guest.memory(), policy, &0i32.to_ne_bytes())?;
940        }
941        if let Some(mask) = mask {
942            let bytes = (call.maxnode() - 1).div_ceil(64) * 8;
943            if bytes > PAGE.max(8) {
944                return Err(Errno::EINVAL.into());
945            }
946            let mut words = vec![0u8; bytes as usize];
947            if mems_allowed && !words.is_empty() {
948                words[0] = 1;
949            }
950            if !words.is_empty() {
951                let mask = AddrMut::<u8>::from_raw(mask.as_raw()).ok_or(Errno::EFAULT)?;
952                write_get_mempolicy_output(&mut guest.memory(), mask, &words)?;
953            }
954        }
955        Ok(0)
956    }
