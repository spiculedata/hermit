74/// Signals hermit takes from the guest's namespace, and what a guest loses.
75///
76/// ⚠️ ENUMERATED BY MEASUREMENT, NOT BY READING (2026-08-25). Every signal 1..64
77/// was run under two delivery paths -- self-directed `raise` and a sibling
78/// thread's `pthread_kill` -- with a native run as the control for each. Exactly
79/// TWO differ from native, and they fail in different ways at different points:
80///
81///   SIGSTKFLT (16)  `rt_sigaction` is NO-OPED below, so the handler is never
82///                   installed and the default disposition terminates the guest:
83///                   observed exit 144 (= 128 + 16).
84///   SIGTRAP    (5)  `rt_sigaction` passes through and the handler IS installed,
85///                   but ptrace consumes every SIGTRAP (syscall stops, seccomp
86///                   stops, breakpoints), so the handler never runs. ⚠️ THE GUEST
87///                   THEN EXITS 0 WITH NO DIAGNOSTIC AT ALL -- a clean pass that
88///                   behaved differently from native, which no cell can catch.
89///
90/// Everything else in 1..31 matches native exactly; 9/19 and 32/33 refuse
91/// `sigaction` natively too and are not hermit's. Realtime 34..64 are
92/// delivered and handled like any other signal under ptrace
93/// (<https://github.com/rrnewton/reverie/issues/207>); they are not
94/// appropriation.
95const APPROPRIATED_SIGNALS: [(i32, &str); 2] = [
96    (
97        libc::SIGTRAP,
98        "ptrace consumes every SIGTRAP (syscall/seccomp stops, breakpoints)",
99    ),
100    (
101        libc::SIGSTKFLT,
102        "reverie uses it as PERF_EVENT_SIGNAL, the PMU preemption timer",
103    ),
104];

106/// Say, once per installation, that a guest handler will never run.
107///
108/// ⚠️ WHY A DIAGNOSTIC AND NOT A REFUSAL. Returning `EINVAL` from `sigaction`
109/// was considered and deliberately rejected for SIGSTKFLT -- see the comment at
110/// the no-op below: the Go runtime registers that handler, and refusing would
111/// break every Go guest at startup. That reasoning generalises: installing a
112/// handler defensively is common, actually raising these signals is rare, so
113/// refusal breaks MORE programs than the current behaviour. The contract
114/// question of what a guest is owed here is genuinely open.
115///
116/// What is NOT open is that hermit currently says NOTHING. This line commits to
117/// no policy, breaks no conforming program, and turns a silent wrong answer into
118/// a visible one -- the same reasoning as the `HERMIT_INTERNAL_FAILURE` marker.
119/// The decision, separated from the reporting so a test can exercise THIS and
120/// not a copy of it. A unit test that re-implements a predicate keeps passing
121/// when the real one is gutted -- measured in this project's own pipe work,
122/// where deleting the production wiring left every unit test green.
123fn appropriated_reason(signum: i32, handler: u64) -> Option<&'static str> {
124    // SIG_DFL (0) and SIG_IGN (1) lose nothing: the guest is not asking to be
125    // called back, so there is no expectation to disappoint.
126    if handler <= 1 {
127        return None;
128    }
129    APPROPRIATED_SIGNALS
130        .iter()
131        .find(|(s, _)| *s == signum)
132        .map(|(_, why)| *why)
133}
134
135fn warn_appropriated_signal(signum: i32, handler: u64) {
136    if let Some(why) = appropriated_reason(signum, handler) {
137        tracing::warn!(
138            "HERMIT_APPROPRIATED_SIGNAL signum={signum} effect=handler-installed-but-never-invoked reason={why}"
139        );
140    }
141}

151        Err(Errno::EINVAL)

155/// The mask Linux installs when a call names `mask` as the signal mask to sleep
156/// under (`rt_sigsuspend`, `ppoll`, `pselect6`): `set_current_blocked` drops
157/// `SIGKILL` and `SIGSTOP`, which can never be blocked.
158pub(crate) fn kernel_installed_signal_mask(mask: KernelSigset) -> KernelSigset {
159    let unblockable = (1_u64 << (libc::SIGKILL - 1)) | (1_u64 << (libc::SIGSTOP - 1));
160    mask & !unblockable
161}

163fn without_perf_event_signal(mask: KernelSigset) -> KernelSigset {
164    let bit = (reverie::PERF_EVENT_SIGNAL as u32) - 1;
165    mask & !(1_u64 << bit)
166}

168/// Read one raw kernel signal mask while preserving the kernel's user-access check.
169///
170/// `safeptrace::Stopped::read` deliberately uses `PTRACE_PEEKDATA` for reads of
171/// eight bytes or less. That operation can read a `PROT_NONE` page, unlike the
172/// kernel's `copy_from_user`, so using `MemoryAccess::read_value` alone would
173/// turn an `EFAULT` from a raw signal syscall into success. An invalid `how`
174/// value makes `rt_sigprocmask` copy exactly the kernel-sized input and then
175/// return `EINVAL` without changing the mask. Use that as a permission probe,
176/// then read the already-validated word while the guest is stopped.
177/// The x86-64 System V red zone below the stack pointer, which a leaf function
178/// may use and an injected call therefore must not.
179pub(super) const STACK_RED_ZONE: usize = 128;
180
181/// The two eight-byte scratch cells `handle_rt_sigsuspend` places below the
182/// red zone: the rt_sigpending result and the copy of the call's mask.
183const SCRATCH_CELLS_SIZE: usize = 2 * std::mem::size_of::<u64>();

185pub(super) async fn read_kernel_sigset<G, T>(
186    guest: &mut G,
187    address: Addr<'_, libc::sigset_t>,
188) -> Result<KernelSigset, Error>
189where
190    G: Guest<Detcore<T>>,
191    T: RecordOrReplay,
192{
193    let validation = syscalls::RtSigprocmask::new()
194        .with_how(-1)
195        .with_set(Some(address))
196        .with_oldset(None)
197        .with_sigsetsize(KERNEL_SIGSET_SIZE);
198    match guest.inject(validation).await {
199        Err(Errno::EINVAL) => {}
200        Err(errno) => return Err(errno.into()),
201        Ok(_) => {
202            // Both Linux and the KVM syscall implementation reject an unknown
203            // operation. Success means the backend did not validate the probe.
204            return Err(Errno::EIO.into());
205        }
206    }
207    Ok(guest.memory().read_value(address.cast())?)
208}

210/// Signal phase 1: the guest's SIGALRM disposition (a guest handler or not)
211/// and SIGALRM blocked bit, as its runtime keeps them virtual, read back with
212/// a query `rt_sigaction` and `rt_sigprocmask` that the runtime answers from
213/// its virtual state.
214async fn virtual_sigalrm_state<G, T>(guest: &mut G) -> Result<(bool, bool), Error>
215where
216    G: Guest<Detcore<T>>,
217    T: RecordOrReplay,
218{
219    let state = read_sigalrm_sleep_state(guest).await?;
220    Ok((
221        state.disposition == SigalrmSleepDisposition::Handled,
222        state.blocked,
223    ))
224}

226/// Query through the runtime, which reports the guest's action and mask even
227/// when it owns a physical SIGALRM handler. These queries run in an owned turn;
228/// a scheduler-side read of the callback's physical mask is not a substitute.
229async fn read_sigalrm_sleep_state<G, T>(guest: &mut G) -> Result<InGuestSleepSigalrm, Error>
230where
231    G: Guest<Detcore<T>>,
232    T: RecordOrReplay,
233{
234    let mut stack = guest.stack().await;
235    let action = stack.reserve::<KernelSigaction>();
236    let mask = stack.reserve::<KernelSigset>();
237    let _stack_guard = stack.commit()?;
238    guest
239        .inject(
240            syscalls::RtSigaction::new()
241                .with_signum(libc::SIGALRM)
242                .with_action(None)
243                .with_old_action(Some(action.cast()))
244                .with_sigsetsize(KERNEL_SIGSET_SIZE),
245        )
246        .await?;
247    guest
248        .inject(
249            syscalls::RtSigprocmask::new()
250                .with_how(libc::SIG_BLOCK)
251                .with_set(None)
252                .with_oldset(Some(mask.cast()))
253                .with_sigsetsize(KERNEL_SIGSET_SIZE),
254        )
255        .await?;
256    let action: KernelSigaction = guest.memory().read_value(action)?;
257    let mask: KernelSigset = guest.memory().read_value(mask)?;
258    let disposition = if action.handler == libc::SIG_DFL as u64 {
259        SigalrmSleepDisposition::Default
260    } else if action.handler == libc::SIG_IGN as u64 {
261        SigalrmSleepDisposition::Ignored
262    } else {
263        SigalrmSleepDisposition::Handled
264    };
265    Ok(InGuestSleepSigalrm {
266        disposition,
267        blocked: mask & kernel_sigset_bit(libc::SIGALRM) != 0,
268    })
269}

271/// Take one pending copy of `signo` from the calling thread's queues, its own
272/// first and then its process's shared one, as Linux dequeues it, with a
273/// zero-timeout `rt_sigtimedwait` injected while the thread holds its turn.
274/// Returns the signal taken, or the call's error (EAGAIN: none was pending).
275pub(crate) async fn take_pending_signal<G, T>(guest: &mut G, signo: i32) -> Result<i64, Error>
276where
277    G: Guest<Detcore<T>>,
278    T: RecordOrReplay,
279{
280    let mut stack = guest.stack().await;
281    let set = stack.push(kernel_sigset_bit(signo));
282    let timeout = stack.push(Timespec {
283        tv_sec: 0,
284        tv_nsec: 0,
285    });
286    let stack_guard = stack.commit()?;
287    // SAFETY: the kernel only reads the set; Reverie types the argument as
288    // mutable.
289    let set = unsafe { set.cast::<libc::sigset_t>().into_mut() };
290    let taken = guest
291        .inject(
292            syscalls::RtSigtimedwait::new()
293                .with_set(Some(set))
294                .with_info(None)
295                .with_timeout(Some(timeout))
296                .with_sigsetsize(KERNEL_SIGSET_SIZE),
297        )
298        .await;
299    drop(stack_guard);
300    Ok(taken?)
301}

353fn timeval_to_logical_time(value: libc::timeval) -> Result<LogicalTime, Errno> {
354    let seconds = u64::try_from(value.tv_sec).map_err(|_| Errno::EINVAL)?;
355    let micros = u64::try_from(value.tv_usec).map_err(|_| Errno::EINVAL)?;
356    if micros >= 1_000_000 {
357        return Err(Errno::EINVAL);
358    }
359    let nanos = seconds
360        .checked_mul(1_000_000_000)
361        .and_then(|nanos| nanos.checked_add(micros * 1_000))
362        .ok_or(Errno::EINVAL)?;
363    Ok(LogicalTime::from_nanos(nanos))
364}

389/// The 128 siginfo bytes at `info`. A page the kernel may write a siginfo to
390/// can lack read permission (`PROT_WRITE` alone), which `process_vm_readv`
391/// refuses; the tracer then reads it through `/proc/<tid>/mem`, as ptrace
392/// access does, which needs no read permission on the mapping.
393fn read_siginfo<T, G: Guest<Detcore<T>>>(
394    guest: &mut G,
395    info: AddrMut<libc::siginfo_t>,
396) -> Option<[u8; 128]>
397where
398    T: RecordOrReplay,
399{
400    if let Ok(bytes) = guest.memory().read_value(info.cast::<[u8; 128]>()) {
401        return Some(bytes);
402    }
403    use std::os::unix::fs::FileExt;
404    let mem = std::fs::File::open(format!("/proc/{}/mem", guest.tid().as_raw())).ok()?;
405    let mut bytes = [0u8; 128];
406    mem.read_exact_at(&mut bytes, info.as_raw() as u64).ok()?;
407    Some(bytes)
408}

410/// The thread a process-directed signal goes to. `zombie` is the process's
411/// exit state (`Scheduler::exited_process_is_zombie`). `Ok(None)` is a zombie:
412/// no live threads, but its parent can still wait for it, so Linux still finds
413/// the process and the send succeeds (https://github.com/rrnewton/hermit/issues/4091).
414fn deterministic_kill_target(
415    targets: &[DetTid],
416    zombie: Option<bool>,
417    sig: libc::c_int,
418) -> Result<Option<DetTid>, Errno> {
419    match targets {
420        [] if zombie == Some(true) => Ok(None),
421        [] => Err(Errno::ESRCH),
422        [target] => Ok(Some(*target)),
423        [target, ..] if sig == 0 => Ok(Some(*target)),
424        _ => Err(Errno::ENOSYS),
425    }
426}

438/// Whether one of Linux's three ordinary signal syscalls names the calling
439/// task exactly and asks for the one signal that cannot return successfully.
440///
441/// Keep process-group and broadcast spellings out of this predicate. Even when
442/// the caller belongs to the named group, KVM can refuse a group containing a
443/// second process; reserving an exit before that refusal would strand the
444/// scheduler's terminal barrier.
445fn self_sigkill_targets_current_task(
446    signal: libc::c_int,
447    target_process: Option<DetPid>,
448    target_thread: Option<DetTid>,
449    current_process: DetPid,
450    current_thread: DetTid,
451) -> bool {
452    signal == libc::SIGKILL
453        && (target_process.is_some() || target_thread.is_some())
454        && target_process.is_none_or(|target| target == current_process)
455        && target_thread.is_none_or(|target| target == current_thread)
456}

459    /// Bind a supported in-guest emulated sleep to its guest disposition/mask.
460    /// Ordinary admitted creation does not share signal dispositions between
461    /// processes, and each supported in-guest process has one guest thread.
462    /// The snapshot therefore stays current until this request resumes.
463    pub(crate) async fn in_guest_sleep_sigalrm_snapshot<G: Guest<Self>>(
464        &self,
465        guest: &mut G,
466    ) -> Option<InGuestSleepSigalrm> {
467        if !self.cfg.backend.virtualizes_guest_sigalrm {
468            return None;
469        }
470        match read_sigalrm_sleep_state(guest).await {
471            Ok(state) => Some(state),
472            Err(error) => {
473                self.stop_after_in_guest_sleep_signal_loss(
474                    guest,
475                    format!("could not query its SIGALRM action/mask before sleeping: {error}"),
476                )
477                .await
478            }
479        }
480    }

482    /// A typed wake means the scheduler generated a one-shot SIGALRM for this
483    /// exact sleep without sending it physically. Stage it through the frozen
484    /// runtime self-signal path, which runs signal and exit callbacks before
485    /// Linux kills the process. The caller must immediately return its handler:
486    /// it must neither resume guest code nor copy nanosleep's remainder out.
487    pub(crate) async fn stage_in_guest_fatal_alarm<G: Guest<Self>>(&self, guest: &mut G) {
488        if !self.cfg.backend.virtualizes_guest_sigalrm {
489            self.stop_after_in_guest_sleep_signal_loss(
490                guest,
491                "received an in-guest fatal alarm on another backend".into(),
492            )
493            .await;
494        }
495        match read_sigalrm_sleep_state(guest).await {
496            Ok(InGuestSleepSigalrm {
497                disposition: SigalrmSleepDisposition::Default,
498                blocked: false,
499            }) => {}
500            other => {
501                self.stop_after_in_guest_sleep_signal_loss(
502                    guest,
503                    format!("its default/unblocked SIGALRM sleep snapshot changed: {other:?}"),
504                )
505                .await;
506            }
507        }
508        let call = syscalls::Tgkill::new()
509            .with_tgid(guest.pid().as_raw())
510            .with_tid(guest.tid().as_raw())
511            .with_sig(libc::SIGALRM);
512        if guest.inject(call).await != Ok(0) {
513            self.stop_after_in_guest_sleep_signal_loss(
514                guest,
515                "could not stage its default-fatal SIGALRM after a sleep wake".into(),
516            )
517            .await;
518        }
519    }

525    ) -> ! {

534            )

536    }

572    ) -> Result<i64, Error> {
573        if !guest.config().sequentialize_threads {
574            info!(
575                "[dtid {}] Running without scheduler, so letting setitimer call through...",
576                guest.thread_state().dettid
577            );
578            return Ok(guest.inject(call).await?);
579        }
580        if call.which() != libc::ITIMER_REAL {
581            return Err(Error::Errno(Errno::ENOSYS));
582        }
583
584        let value = call.value().ok_or(Errno::EFAULT)?;
585        let timer: libc::itimerval = guest.memory().read_value(value)?;
586        let interval = timeval_to_logical_time(timer.it_interval)?;
587        let duration = timeval_to_logical_time(timer.it_value)?;
588        // A zero value disarms the timer, whatever its interval.
589        if duration.as_nanos() != 0 && interval.as_nanos() != 0 {
590            refuse_sigalrm(guest, SigalrmControl::ArmRecurringTimer).await?;
591        }
592        let (remaining, old_interval) =
593            register_alarm(guest, duration, interval, Signal::SIGALRM).await;
594        if let Some(old_value) = call.ovalue() {
595            let old_timer = libc::itimerval {
596                it_interval: logical_time_to_timeval(old_interval),
597                it_value: logical_time_to_timeval(remaining),
598            };
599            guest.memory().write_value(old_value, &old_timer)?;
600        }
601        Ok(0)
602    }

636    /// A pause is really just an unbounded sleep.
637    pub async fn handle_pause<G: Guest<Self>>(
638        &self,
639        guest: &mut G,
640        call: syscalls::Pause,
641    ) -> Result<i64, Error> {
642        if guest.config().sequentialize_threads {
643            // `pause` has no deadline: it returns only when a signal is delivered.
644            // `LogicalTime::INDEFINITE` records that, and the scheduler refuses to
645            // fast-forward virtual time onto it (see `step2d_handle_empty_queue`),
646            // so the `Normal` arm below stays unreachable.
647            loop {
648                // A signal that already ended the call before it parked: one sent
649                // while this thread was stopped short of filing the request, as
650                // when its timeslice ended in the syscall's prehook.
651                if sleep_signal(guest, SleepCheck::BeforePark, true).await? == SleepSignal::Pending
652                {
653                    return Err(Errno::ERESTARTNOHAND.into());
654                }
655                let mut req = Self::sleep_request_abs(guest, LogicalTime::INDEFINITE).await;
656                req.fyi(crate::resources::PAUSE_FYI);
657                req.in_guest_sleep_sigalrm = self.in_guest_sleep_sigalrm_snapshot(guest).await;
658                match crate::tool_global::parked_wait_request(
659                    guest,
660                    req,
661                    crate::scheduler::parked::ParkedWaitPolicy::PauseNoHandlerRestart,
662                )
663                .await
664                {
665                    ResumeStatus::InGuestFatalAlarm => {
666                        self.stage_in_guest_fatal_alarm(guest).await;
667                        return Err(Errno::EINTR.into());
668                    }
669                    ResumeStatus::Normal => {
670                        panic!(
671                            "Internal violation: pause should never return from the scheduler except by interruption!"
672                        )
673                    }
674                    ResumeStatus::Signaled(signals) => {
675                        match sleep_signal(guest, SleepCheck::AfterWake(signals), true).await? {
676                            SleepSignal::Pending => return Err(Errno::ERESTARTNOHAND.into()),
677                            SleepSignal::Held => return Err(Errno::EINTR.into()),
678                            SleepSignal::None => {
679                                info!(
680                                    "[dtid {}] pause woken for a signal no longer pending for it; waiting again",
681                                    guest.thread_state().dettid
682                                );
683                            }
684                        }
685                    }
686                }
687            }
688        } else {
689            info!(
690                "[dtid {}] Running without scheduler, so letting pause call through...",
691                guest.thread_state().dettid
692            );
693            Ok(guest.inject(call).await?)
694        }
695    }

697    /// Run rt_sigsuspend without holding the deterministic scheduler turn.
698    ///
699    /// The kernel must perform the temporary mask swap atomically and restore the
700    /// original mask after signal delivery, so execute the real blocking syscall
701    /// while marking this thread as blocked outside the runnable set.
702    pub async fn handle_rt_sigsuspend<G: Guest<Self>>(
703        &self,
704        guest: &mut G,
705        call: syscalls::RtSigsuspend,
706    ) -> Result<i64, Error> {
707        // Invalid arguments return immediately from the kernel and therefore are
708        // not signal-only waits.
709        validate_kernel_sigset_size(call.sigsetsize())?;
710        let Some(mask_addr) = call.mask() else {
711            return Err(Errno::EFAULT.into());
712        };
713
714        let temporary_mask = read_kernel_sigset(guest, mask_addr).await?;
715        // The temporary mask lets the kernel take a pending real-time timer
716        // expiration that every thread blocked when it expired
717        // (https://github.com/rrnewton/hermit/issues/4089).
718        observe_under_mask(guest, temporary_mask).await?;
719        // The real call sleeps under a private copy of the mask read here, not
720        // under the guest's buffer. The scheduler decides from this copy which
721        // signals can end the call (`Resources::blocked_signal_mask`), and
722        // another process sharing the buffer (MAP_SHARED) may rewrite it before
723        // the call runs; Linux copies the mask when the call starts, and from
724        // the guest's view the call has already started.
725        //
726        // The copy and the rt_sigpending result live in two cells just below
727        // the stack's red zone, the scratch area an injected call may use, but
728        // never on the caller's buffer: a guest may keep its mask there while
729        // it blocks every signal, and a copy placed on that very buffer would
730        // be the original again, while the pending cell would overwrite the
731        // guest's mask. If the buffer overlaps the two cells, they move to just
732        // below it. They are written directly rather than through a scratch
733        // stack, whose commit writes one contiguous region down from the red
734        // zone and so would cover a buffer inside it.
735        let caller_mask = mask_addr.as_raw()..mask_addr.as_raw() + KERNEL_SIGSET_SIZE;
736        let below_red_zone = (guest.regs().await.rsp as usize).wrapping_sub(STACK_RED_ZONE);
737        let cells = (below_red_zone - SCRATCH_CELLS_SIZE)..below_red_zone;
738        let cells_start = if cells.start < caller_mask.end && caller_mask.start < cells.end {
739            (caller_mask.start - SCRATCH_CELLS_SIZE) & !(std::mem::align_of::<u64>() - 1)
740        } else {
741            cells.start
742        };
743        let pending_addr = Addr::<u64>::from_raw(cells_start).ok_or(Errno::EFAULT)?;
744        let mask_copy = AddrMut::<u64>::from_raw(cells_start + std::mem::size_of::<u64>())
745            .ok_or(Errno::EFAULT)?;
746        guest.memory().write_value(mask_copy, &temporary_mask)?;
747        let pending_out = AddrMut::<libc::sigset_t>::from_raw(pending_addr.as_raw())
748            .expect("stack address must be non-null");
749        let pending_call = syscalls::RtSigpending::new()
750            .with_set(Some(pending_out))
751            .with_sigsetsize(KERNEL_SIGSET_SIZE);
752        guest.inject_with_retry(pending_call).await?;
753        let pending: u64 = guest.memory().read_value(pending_addr)?;
754        let call = call.with_mask(Some(Addr::from_raw(mask_copy.as_raw()).expect("non-null")));
755
756        // The scheduler records the mask the call sleeps under, read here in
757        // this thread's turn, to choose a SIGCHLD target while the thread is
758        // outside the runnable set (`Resources::blocked_signal_mask`).
759        let installed_mask = kernel_installed_signal_mask(temporary_mask);
760        if pending & !temporary_mask != 0 {
761            // The kernel will consume an already-pending signal as soon as it
762            // atomically installs the temporary mask. Keep this immediate case
763            // out of the terminal-wait classification; the real syscall still
764            // performs delivery and restores the old mask.
765            self.record_or_replay_blocking_with_mask(guest, call.into(), Some(installed_mask))
766                .await
767        } else {
768            self.record_or_replay_rt_sigsuspend(guest, call, installed_mask)
769                .await
770        }
771    }

773    /// rt_sigaction
774    pub async fn handle_rt_sigaction<G: Guest<Self>>(
775        &self,
776        guest: &mut G,
777        call: syscalls::RtSigaction,
778    ) -> Result<i64, Error> {
779        // Linux rejects an invalid size before inspecting either user pointer.
780        // Preserve that EINVAL-before-EFAULT ordering even for the signal that
781        // Hermit reserves for deterministic preemption.
782        validate_kernel_sigset_size(call.sigsetsize())?;
783
784        // Copy the complete kernel object before inspecting the signal number or
785        // writing old_action. This preserves Linux's action-before-old_action
786        // pointer ordering, including when both arguments alias.
787        let kernel_action = match call.action() {
788            Some(action) => Some(read_kernel_sigaction(guest, action).await?),
789            None => None,
790        };
791
792        // Both appropriated signals are reported here, at the one point where the
793        // guest states its expectation. SIGTRAP falls through to the ordinary
794        // path below (its handler really is installed; ptrace just eats the
795        // signal), so this must run before the SIGSTKFLT early return.
796        if let Some(action) = kernel_action {
797            warn_appropriated_signal(call.signum(), action.handler);
798        }
799
800        // PERF_EVENT_SIGNAL is reserved.
801        if call.signum() == reverie::PERF_EVENT_SIGNAL as i32 {
802            // The go runtime attempts to register this (unused) signal handler.  We will never
803            // deliver signals of this kind to the guest, so we just turn this action into a noop
804            // rather than returning `Err(Errno::EINVAL.into())`. Preserve that established
805            // policy while still honoring the raw syscall's pointer accesses: the virtual old
806            // disposition is the default action, never Reverie's private preemption handler.
807            if call.old_action().is_some() {
808                // SIGKILL's disposition cannot be changed by userspace, so its
809                // query copies the same four zero words as our virtual default.
810                // Let the kernel copy them: a short write_value can finish with
811                // PTRACE_POKEDATA and overwrite a protected final word. This also
812                // preserves the kernel's partial-copy effects on EFAULT without
813                // exposing Reverie's private preemption action.
814                return Ok(guest
815                    .inject(call.with_signum(libc::SIGKILL).with_action(None))
816                    .await?);
817            }
818            return Ok(0);
819        }
820        // Signal phase 1 (step I3): on a backend that keeps a handled SIGALRM
821        // virtual, the scheduler's ledger must agree with the runtime.
822        let phase1 = self.cfg.backend.virtualizes_guest_sigalrm
823            && call.signum() == libc::SIGALRM
824            && kernel_action.is_some();
825        let was_handled = guest.thread_state().sigalrm_handled;
826        let to_handler = kernel_action.is_some_and(|action| {
827            action.handler != libc::SIG_DFL as u64 && action.handler != libc::SIG_IGN as u64
828        });
829        if phase1 {
830            if to_handler {
831                self.refuse_sigalrm_handler(guest).await?;
832            } else if was_handled
833                && kernel_action.is_some_and(|action| action.handler == libc::SIG_DFL as u64)
834            {
835                refuse_sigalrm(guest, SigalrmControl::HandlerToDefault).await?;
836            }
837        }
838        let result = if let Some(kernel_action) = kernel_action {
839            // The kernel treats `action` as input. Sanitize a private copy instead
840            // of writing through the guest pointer: the input may be read-only,
841            // and changing it would make the syscall wrapper observable.
842            let mut kernel_action = kernel_action;
843            kernel_action.mask = without_perf_event_signal(kernel_action.mask);
844            let mut stack = guest.stack().await;
845            let sanitized_action = stack.push(kernel_action);
846            let _stack_guard = stack.commit()?;
847            guest
848                .inject(call.with_action(Some(sanitized_action.cast())))
849                .await
850        } else {
851            guest.inject(call).await
852        };
853        // Publish the state the runtime now holds whenever the disposition may
854        // have changed: from a handler, or to one the runtime did not refuse.
855        // Any result counts, since an EFAULT copying the old action out comes
856        // after the change. A default or ignored action set while none is
857        // handled changes nothing the scheduler records, so it costs nothing.
858        if phase1 && (was_handled || (to_handler && result != Err(Errno::EPERM))) {
859            self.publish_virtual_sigalrm(guest).await?;
860        }
861        // Linux reaps a process's children automatically at exit when its
862        // SIGCHLD action is SIG_IGN or carries SA_NOCLDWAIT, and a signal to
863        // such a child then finds no zombie. Mirror the new action for the
864        // scheduler's exit model, with the same success rule as above
865        // (https://github.com/rrnewton/hermit/issues/4091).
866        if call.signum() == libc::SIGCHLD
867            && guest.config().sequentialize_threads
868            && let Some(action) = kernel_action
869            && matches!(result, Ok(_) | Err(Errno::EFAULT))
870        {
871            let dettid = guest.thread_state().dettid;
872            if let Some(process) = registered_process_of(guest, dettid).await {
873                set_sigchld_disposition(
874                    guest,
875                    process,
876                    action.handler == libc::SIG_IGN as u64,
877                    action.flags & libc::SA_NOCLDWAIT as u64 != 0,
878                )
879                .await;
880            }
881        }
882        // Linux flushes a pending SIGCHLD when its action becomes SIG_IGN or
883        // SIG_DFL, which ignores it (`do_sigaction`, `sig_handler_ignored`): the
884        // child-exit ledger's send window goes with it, in this committed turn
885        // (https://github.com/rrnewton/hermit/issues/4126).
886        if call.signum() == libc::SIGCHLD
887            && guest.config().sequentialize_threads
888            && matches!(result, Ok(_) | Err(Errno::EFAULT))
889            && kernel_action.is_some_and(|action| {
890                action.handler == libc::SIG_IGN as u64 || action.handler == libc::SIG_DFL as u64
891            })
892        {
893            crate::tool_global::child_exit_sigchld(
894                guest,
895                crate::scheduler::child_exit_sigchld::ChildExitSigchldControl::Flushed,
896            )
897            .await;
898        }
899        // Linux flushes the pending instances of a signal whose disposition
900        // becomes SIG_IGN (`do_sigaction`), among them a pending real-time timer
901        // expiration (https://github.com/rrnewton/hermit/issues/4089).
902        if result.is_ok()
903            && kernel_action.is_some_and(|action| action.handler == libc::SIG_IGN as u64)
904            && (1..=64).contains(&call.signum())
905        {
906            let bit = kernel_sigset_bit(call.signum());
907            let pending = pending_real_time_among(guest, bit).await;
908            act_on_pending_real_time(guest, pending, 0, bit).await;
909        }
910        // The same flush takes a `SIGEV_THREAD_ID` timer's pending copy of a
911        // standard signal, for SIG_IGN or for SIG_DFL on a signal whose
912        // default is to ignore it (`sig_handler_ignored`), so it no longer
913        // counts as the timer's own (https://github.com/rrnewton/hermit/issues/4164).
914        let ignored = |handler: u64| {
915            handler == libc::SIG_IGN as u64
916                || (handler == libc::SIG_DFL as u64
917                    && [libc::SIGCONT, libc::SIGCHLD, libc::SIGWINCH, libc::SIGURG]
918                        .contains(&call.signum()))
919        };
920        // The kernel reads the new action from Detcore's sanitized copy, so an
921        // EFAULT can only come from writing `old_action`, after the new action
922        // is installed.
923        if (result.is_ok() || matches!(result, Err(Errno::EFAULT)))
924            && kernel_action.is_some_and(|action| ignored(action.handler))
925            && (1..32).contains(&call.signum())
926            && crate::scheduler::timer_copies_recorded()
927        {
928            crate::tool_global::forget_thread_timer_copies(guest, call.signum()).await;
929        }
930        Ok(result?)
931    }

934    /// each EPERM before anything changes (design section 6 and closure 5):
935    /// threads not sequentialized (`alarm` would arm a kernel timer), a
936    /// signalfd in the process's descriptor table, and a kernel producer of

1020    async fn rt_sigprocmask_inner<G: Guest<Self>>(
1021        &self,
1022        guest: &mut G,
1023        call: syscalls::RtSigprocmask,
1024    ) -> Result<i64, Error> {
1025        if call.how() != libc::SIG_BLOCK && call.how() != libc::SIG_SETMASK {
1026            Ok(guest.inject_with_retry(call).await?)
1027        } else if let Some(set) = call.set() {
1028            let set_mask = read_kernel_sigset(guest, set).await?;
1029            let mut stack = guest.stack().await;
1030            let new_set = stack.push(without_perf_event_signal(set_mask));
1031            let _stack_guard = stack.commit()?;
1032            let modified_call = syscalls::RtSigprocmask::new()
1033                .with_how(call.how())
1034                .with_set(Some(new_set.cast()))
1035                .with_oldset(call.oldset())
1036                .with_sigsetsize(call.sigsetsize());
1037            // Keep returning to the handler so post_handler_hook can run, but
1038            // do not expose a tracer preemption as ERESTARTSYS to the guest.
1039            Ok(guest.inject_with_retry(modified_call).await?)
1040        } else {
1041            Ok(guest.inject_with_retry(call).await?)
1042        }
1043    }

1045    /// rt_sigtimedwait system call
1046    ///
1047    /// This is handled by the scheduler and not passed to the record/replay layer,
1048    /// because currently signals are not recorded.
1049    pub async fn handle_rt_sigtimedwait<G: Guest<Self>>(
1050        &self,
1051        guest: &mut G,
1052        call: syscalls::RtSigtimedwait,
1053    ) -> Result<i64, Error> {
1054        // Linux rejects an invalid mask width before inspecting its pointers.
1055        validate_kernel_sigset_size(call.sigsetsize())?;
1056        // Linux keeps this entry snapshot for the whole wait. The retry helper
1057        // currently injects from the caller's pointer again, so another thread
1058        // can change the set between retries. Fixing that requires sharing one
1059        // scratch-stack guard with the helper's zero-timeout object; do not hide
1060        // that remaining difference by treating this validation read as a snapshot.
1061
1062        let dettid = guest.thread_state().dettid;
1063
1064        let maybe_timeout = if let Some(timeout) = call.timeout() {
1065            let ts: Timespec = guest.memory().read_value(timeout)?;
1066            let ns_delta =
1067                Duration::from_secs(ts.tv_sec as u64) + Duration::from_nanos(ts.tv_nsec as u64);
1068            let base_time = thread_observe_time(guest).await;
1069            let target_time = base_time + ns_delta;
1070            Some(target_time)
1071        } else {
1072            None
1073        };
1074        let mut rsrc = Resources::new(dettid);
1075        rsrc.insert(ResourceID::InternalIOPolling, Permission::W);
1076        rsrc.fyi("rt_sigtimedwait");
1077        // While it polls, a signal in the set is taken at the next poll

1132    /// The `signalfd_siginfo` records a successful signalfd read returned:
1133    /// the first `read` bytes of the guest buffers `extents` (address,
1134    /// length), in order, as `read`, `readv`, `preadv` and `preadv2` scatter
1135    /// them. They are reconciled with the child-exit ledger as a SIGCHLD
1136    /// delivered to a handler (`Detcore::child_exit_sigchld_delivers`) or
1137    /// consumed by `rt_sigtimedwait` (`synchronous_sigchld_stands`) is: the
1138    /// scheduler's copy gets the siginfo Linux gives the parent, the kernel's
1139    /// coalesced copy for a SIGKILL cohort names the cohort's lowest pid, and
1140    /// a copy for an exit a SIGCHLD already stood for is dropped. Other
1141    /// records are kept as read. The kept records are scattered back over the
1142    /// same extents, in order, with zeros over the bytes of dropped ones.
1143    /// Returns the bytes kept, a whole number of records.
1144    pub(crate) async fn reconcile_signalfd_extents<G: Guest<Self>>(
1145        &self,
1146        guest: &mut G,
1147        extents: &[(usize, usize)],
1148        read: usize,
1149    ) -> usize {
1150        let len = read - read % SIGNALFD_RECORD;
1151        let pieces = signalfd_extent_pieces(extents, len);
1152        let mut bytes = Vec::with_capacity(len);
1153        for (address, length) in &pieces {
1154            let mut piece = vec![0u8; *length];
1155            let Some(address_mut) = AddrMut::<u8>::from_raw(*address) else {
1156                return len;
1157            };
1158            if guest.memory().read_exact(address_mut, &mut piece).is_err() {
1159                use std::os::unix::fs::FileExt;
1160                let read = std::fs::File::open(format!("/proc/{}/mem", guest.tid().as_raw()))
1161                    .and_then(|mem| mem.read_exact_at(&mut piece, *address as u64));
1162                if read.is_err() {
1163                    // The kernel has just written these bytes, so the buffer
1164                    // is mapped and writable; `/proc/<tid>/mem` reads any
1165                    // such page.
1166                    return len;
1167                }
1168            }
1169            bytes.extend_from_slice(&piece);
1170        }
1171        let controls = signalfd_record_controls(&bytes, std::process::id());
1172        let mut answers = Vec::with_capacity(controls.len());
1173        for control in controls {
1174            answers.push(match control {
1175                Some(control) => Some(crate::tool_global::child_exit_sigchld(guest, control).await),
1176                None => None,
1177            });
1178        }
1179        let Some(rewritten) = apply_signalfd_answers(&bytes, &answers) else {
1180            return len;
1181        };
1182        let mut offset = 0;
1183        for (address, length) in &pieces {
1184            let written = AddrMut::<u8>::from_raw(*address).map(|address| {
1185                guest
1186                    .memory()
1187                    .write_exact(address, &rewritten.bytes[offset..offset + length])
1188            });
1189            if !matches!(written, Some(Ok(()))) {
1190                tracing::warn!(
1191                    "[dtid {}] the signalfd records at {:#x} keep the kernel's siginfo",
1192                    guest.thread_state().dettid,
1193                    address,
1194                );
1195                return len;
1196            }
1197            offset += length;
1198        }
1199        rewritten.kept
1200    }

1205    /// kernel and the scheduler send it (`scheduler::child_exit_sigchld`).
1206    /// The scheduler's copy is returned with the siginfo Linux gives the
1207    /// parent, and a copy for an exit that a SIGCHLD already stood for is
1208    /// dropped. Returns whether the consumed signal stands.
1209    async fn synchronous_sigchld_stands<G: Guest<Self>>(
1210        &self,
1211        guest: &mut G,
1212        call: syscalls::RtSigtimedwait,
1213        waited: Option<u64>,
1214        result: &Result<i64, Error>,
1215    ) -> bool {
1216        if !self.cfg.backend.reports_child_exit_publication {
1217            return true;
1218        }
1219        // Linux copies the set in before it dequeues, and the siginfo out
1220        // after, so an EFAULT with a readable set (`waited`) took a signal it
1221        // never reports; an unreadable set dequeued nothing. Which copy it
1222        // took: a recorded private one if SIGCHLD is gone from the thread's
1223        // private queue, which only committed guest sends fill; otherwise,
1224        // with SIGCHLD the only signal in the set, the shared queue's. A wider
1225        // set falls back to the shared queue's pending bit (the Claude and
1226        // Codex re-checks of https://github.com/rrnewton/hermit/pull/4131).

1228            && guest.config().sequentialize_threads
1229            && let Some(set) = waited
1230        {

1232            if set & sigchld != 0
1233                && let Ok(state) = crate::syscalls::threads::read_thread_wait_signal_state(
1234                    guest.pid(),
1235                    guest.tid(),
1236                )
1237            {
1238                let shared_pending = if set & !sigchld == 0 {
1239                    false
1240                } else {
1241                    state.shared_pending & sigchld != 0
1242                };
1243                crate::tool_global::child_exit_sigchld(
1244                    guest,
1245                    crate::scheduler::child_exit_sigchld::ChildExitSigchldControl::WaitFaulted {
1246                        private_pending: state.thread_pending & sigchld != 0,
1247                        shared_pending,
1248                    },
1249                )
1250                .await;
1251            }
1252            return true;
1253        }
1254        if !matches!(result, Ok(signal) if *signal == libc::SIGCHLD as i64) {
1255            return true;
1256        }
1257        // Without a siginfo buffer the guest sees only the signal number,
1258        // which every copy shares; the ledger still learns that the wait took
1259        // one (`DequeuedCopy::Unknown`), which may be the shared queue's.
1260        let Some(info) = call.info() else {
1261            crate::tool_global::child_exit_sigchld(
1262                guest,
1263                crate::scheduler::child_exit_sigchld::ChildExitSigchldControl::Dequeue {
1264                    copy: crate::scheduler::child_exit_sigchld::DequeuedCopy::Unknown,
1265                    admits: false,
1266                },
1267            )
1268            .await;
1269            return true;
1270        };
1271        let Some(bytes) = read_siginfo(guest, info) else {
1272            return true;
1273        };
1274        Self::synchronous_sigchld_copy_stands(guest, info, &bytes).await
1275    }

1734    /// What an `rt_sigqueueinfo` whose `siginfo_t` Hermit may not be able to
1735    /// read comes to, decided before admission (`do_rt_sigqueueinfo` in
1736    /// kernel/signal.c). No call that Linux could deliver reaches the kernel
1737    /// without admission (Codex re-check 2 of
1738    /// https://github.com/rrnewton/hermit/pull/4116).
1739    ///
1740    /// The kernel always copies the first 48 bytes (`kernel_siginfo_t`) and
1741    /// the other 80 only for a `si_code` whose layout it does not know
1742    /// (`copy_siginfo_from_user`). Its `copy_from_user` reads what user mode
1743    /// can: on x86 a page with any of read, write or execute permission; it
1744    /// faults on a `PROT_NONE` page; and a fault in a hole just below a
1745    /// grow-down mapping can grow that mapping and succeed. The tracer's own
1746    /// reads say nothing about that (`process_vm_readv` refuses a write-only
1747    /// page, and `/proc/<tid>/mem` force-reads a `PROT_NONE` one), so the
1748    /// mappings of `/proc/<tid>/smaps` decide first:
1749    /// - the first byte of the 48 that user mode cannot read lies in a
1750    ///   `PROT_NONE` mapping, or in a hole whose next mapping above is not
1751    ///   grow-down (`VmFlags` `gd`; only that mapping can grow into it):
1752    ///   EFAULT is certain, and Hermit answers it without the kernel;
1753    /// - any other byte of the 128 that user mode cannot read (a hole below a
1754    ///   grow-down mapping, a tail running into a hole or a `PROT_NONE` page),
1755    ///   or a listing that cannot be read: `Unreadable`, refused by name;
1756    /// - all 128 user-readable: the tracer reads them (`process_vm_readv`, then
1757    ///   `/proc/<tid>/mem` for a write-only page), and a page neither reaches
1758    ///   (a `VM_IO`/`VM_PFNMAP` page such as the vDSO's `[vvar]`) is
1759    ///   `Unreadable`. A `si_code` only the target thread itself may use
1760    ///   (`si_code >= 0` or `SI_TKILL`, from a caller that is not `pid`) is
1761    ///   then `KernelFails`: after a complete copy Linux fails it with EPERM,
1762    ///   or E2BIG first, and sends nothing, so it goes to the kernel unchanged.
1763    fn rt_sigqueueinfo_precheck<G: Guest<Self>>(
1764        &self,
1765        guest: &mut G,
1766        call: &syscalls::RtSigqueueinfo,
1767    ) -> RtSigqueueinfoPrecheck {
1768        const SIZE: usize = std::mem::size_of::<libc::siginfo_t>();
1769        // `sizeof(kernel_siginfo_t)` on x86_64: the part every copy reads.
1770        const KERNEL_SIGINFO: usize = 48;
1771        let address = call.siginfo().map_or(0, |address| address.as_raw());
1772        let Ok(smaps) = std::fs::read_to_string(format!("/proc/{}/smaps", guest.tid().as_raw()))
1773        else {
1774            return RtSigqueueinfoPrecheck::Unreadable;
1775        };
1776        let mappings = parse_smaps(&smaps);
1777        match first_user_unreadable(&mappings, address, KERNEL_SIGINFO) {
1778            None => {}
1779            Some(Unreadable::NoPermission)
1780            | Some(Unreadable::Hole {
1781                below_growsdown: false,
1782            }) => {
1783                return RtSigqueueinfoPrecheck::Efault;
1784            }
1785            Some(Unreadable::Hole {
1786                below_growsdown: true,
1787            }) => {
1788                return RtSigqueueinfoPrecheck::Unreadable;
1789            }
1790        }
1791        if first_user_unreadable(&mappings, address, SIZE).is_some() {
1792            return RtSigqueueinfoPrecheck::Unreadable;
1793        }
1794        let read = call.siginfo().and_then(|info| {
1795            guest.memory().read_value(info).ok().or_else(|| {
1796                use std::os::unix::fs::FileExt;
1797                let mut bytes = [0u8; SIZE];
1798                std::fs::File::open(format!("/proc/{}/mem", guest.tid().as_raw()))
1799                    .and_then(|mem| mem.read_exact_at(&mut bytes, address as u64))
1800                    .ok()?;
1801                // SAFETY: `siginfo_t` is plain data, and `bytes` holds exactly
1802                // one, read from the guest.
1803                Some(unsafe { std::ptr::read_unaligned(bytes.as_ptr().cast::<libc::siginfo_t>()) })
1804            })
1805        });
1806        let Some(info) = read else {
1807            return RtSigqueueinfoPrecheck::Unreadable;
1808        };
1809        let code = info.si_code;
1810        if (code >= 0 || code == libc::SI_TKILL)
1811            && guest.thread_state().dettid.as_raw() != call.tgid()
1812        {
1813            RtSigqueueinfoPrecheck::KernelFails
1814        } else {
1815            RtSigqueueinfoPrecheck::Proceeds
1816        }
1817    }

1867    /// forwarded to the kernel keeps the guest's number, which the kernel reads

2004    /// Send a thread-directed signal through the kernel. Guest PID/TID values are
2005    /// stable in the fresh PID namespace and delivery is scheduler-serialized.
2006    pub async fn handle_tgkill<G: Guest<Self>>(
2007        &self,
2008        guest: &mut G,
2009        call: syscalls::Tgkill,
2010    ) -> Result<i64, Error> {
2011        self.record_orphaned_group_signals(guest, call.sig()).await;
2012        // A nested caller names its own namespace's ids; Detcore's model uses
2013        // the tasks they name, and the forwarded call keeps them
2014        // (https://github.com/rrnewton/hermit/issues/4155). Non-positive ids
2015        // are EINVAL in Linux before any lookup.
2016        let nested = guest.thread_state().nested_pid_namespace;
2017        let (tgid, tid) = if nested && call.tgid() > 0 && call.tid() > 0 {
2018            let (tid, own_init_caught) =
2019                match self.name_guest_pid(guest, Sysno::tgkill, call.tid()).await {
2020                    NamedGuestPid::Task {
2021                        task,
2022                        own_init_caught,
2023                    } => (task, own_init_caught),
2024                    NamedGuestPid::Answer(answer) => return answer,
2025                };
2026            let tgid = match self.name_guest_pid(guest, Sysno::tgkill, call.tgid()).await {
2027                NamedGuestPid::Task { task, .. } => task,
2028                NamedGuestPid::Answer(answer) => return answer,
2029            };
2030            if Self::own_namespace_init_drops(own_init_caught, call.sig()) {
2031                return Ok(self.record_or_replay(guest, call).await?);
2032            }
2033            (tgid.as_raw(), tid.as_raw())
2034        } else {
2035            (call.tgid(), call.tid())
2036        };
2037        if call.sig() == libc::SIGALRM {
2038            refuse_sigalrm(guest, SigalrmControl::SendTo(DetTid::from_raw(tid))).await?;
2039        }
2040        if !nested
2041            && self
2042                .drop_signal_to_hermit(guest, Sysno::tgkill, Some(tgid), tid, call.sig())
2043                .await
2044        {
2045            return Ok(0);
2046        }
2047        let _reserved = self
2048            .reserve_controlled_self_sigkill_exit(
2049                guest,
2050                call.sig(),
2051                Some(DetPid::from_raw(tgid)),
2052                Some(DetTid::from_raw(tid)),
2053            )
2054            .await;
2055        // A SIGKILL to a thread of another process ends that whole process,
2056        // and is retired at this turn, like `kill`'s. Only a valid pair is
2057        // reserved: Linux rejects a thread outside the named group (ESRCH) or a
2058        // non-positive id (EINVAL) before anything happens, so such a send
2059        // takes the ordinary path and gets the kernel's errno.
2060        if call.sig() == libc::SIGKILL
2061            && guest.config().sequentialize_threads
2062            && tgid > 0
2063            && tid > 0
2064            && registered_process_of(guest, DetTid::from_raw(tid)).await
2065                == Some(DetPid::from_raw(tgid))
2066            && let Some(reservation) = reserve_sigkill(guest, DetTid::from_raw(tid)).await
2067        {
2068            return self.send_reserved_sigkill(guest, reservation, call).await;
2069        }
2070        let value = self
2071            .send_guest_signal(
2072                guest,
2073                call,
2074                DetTid::from_raw(call.tid()),
2075                call.sig(),
2076                None,
2077                true,
2078            )
2079            .await?;

2081        // guest THREAD signals a sibling. Like `kill`, a successful cross-task
2082        // send must tell the scheduler, or a target parked in a child wait is
2083        // never woken and the wait hangs.
2084        self.notify_cross_task_signal(guest, DetTid::from_raw(tid), call.sig(), None)
2085            .await;
2086        Ok(value)
2087    }

2677            && self.cfg.backend_supports_blocked_wait_signal_interruption)
2678            || guest.thread_state().handler_restore_rt_reported == REAL_TIME_SIGNALS
2679        {
2680            return;
2681        }
2682        // Only a caught signal runs a handler that returns. The dispositions
2683        // are the process's, so they say that whatever mask the delivery
2684        // interrupted; a signal acted on by default (the teardown's own
2685        // termination signals among them) records nothing.
2686        if let Ok(state) = read_thread_wait_signal_state(guest.pid(), guest.tid())
2687            && state.caught & kernel_sigset_bit(signal.raw()) == 0
2688        {
2689            return;
2690        }
2691        let Some(detpid) = guest.thread_state().detpid else {
2692            return;
2693        };
2694        crate::tool_global::record_handler_restore_mask(guest, detpid, REAL_TIME_SIGNALS).await;
2695        guest.thread_state_mut().handler_restore_rt_reported = REAL_TIME_SIGNALS;
2696    }
2697
2698    /// rt_sigreturn: the mask it restores, from the frame's `uc_sigmask`, may
2699    /// unblock a pending real-time timer expiration that every thread blocked
2700    /// when it expired, which the kernel then delivers as the call returns
2701    /// (https://github.com/rrnewton/hermit/issues/4089). The call itself
2702    /// passes through.
2703    pub(crate) async fn observe_sigreturn_mask<G: Guest<Self>>(&self, guest: &mut G) {
2704        let pending = pending_real_time_among(guest, REAL_TIME_SIGNALS).await;
2705        if pending == 0 {
2706            return;
2707        }
2708        // x86-64 `rt_sigreturn` reads `frame = sp - sizeof(long)` and restores
2709        // `frame->uc.uc_sigmask`; `uc` starts at `sp`.
2710        let sp = guest.regs().await.rsp as usize;
2711        let restored = Addr::<u64>::from_raw(sp.wrapping_add(UC_SIGMASK_OFFSET))
2712            .and_then(|addr| guest.memory().read_value(addr).ok());
2713        let (observable, discarded) = match (
2714            restored,
2715            read_thread_wait_signal_state(guest.pid(), guest.tid()),
2716        ) {
2717            (Some(mask), Ok(state)) => unblocked_by(&state, mask),
2718            // A frame the kernel cannot read either kills the guest with
2719            // SIGSEGV, which delivers nothing pending.
2720            (None, _) => (0, 0),
2721            // A state that cannot be read could hide any delivery.
2722            (Some(_), Err(_)) => (REAL_TIME_SIGNALS, 0),
2723        };
2724        act_on_pending_real_time(guest, pending, observable, discarded).await;
2725    }
2726}

2747/// What ends an emulated `pause` or `nanosleep`, per [`sleep_signal`].
2748#[derive(Clone, Copy, Debug, Eq, PartialEq)]
2749pub(crate) enum SleepSignal {
2750    /// A signal pending for this thread, unblocked and caught or acted on by
2751    /// default: the kernel delivers it as the call returns. `pause` returns
2752    /// `ERESTARTNOHAND` for it and `nanosleep` `EINTR`.
2753    Pending,
2754    /// The scheduler woke the sleep for a held signal of its own that ends it
2755    /// (`held_wake_ends_sleep`), or this backend does not report the guest's
2756    /// signal state: the call returns `EINTR`, as every wake did before.
2757    Held,
2758    /// Nothing ends the sleep, and it waits (again) with its original deadline.
2759    None,
2760}

2762/// Decides, in this thread's own turn, whether a signal ends an emulated
2763/// `pause` or `nanosleep` (<https://github.com/rrnewton/hermit/issues/3982>).
2764///
2765/// Only on a backend whose kernel reports the guest's signal state; on any
2766/// other, a sleep never checks before it parks, and a wake always ends it
2767/// (`Held`). A signal ends the sleep when it is pending for this thread,
2768/// unblocked, and caught or acted on by default
2769/// (`KernelSignalState::pending_interrupting`). Before the sleep parks, a stop
2770/// signal does so only when `stops_end_it`, which `nanosleep` leaves unset
2771/// because Detcore does not keep its deadline for the kernel's restart after a
2772/// stop, so the stop takes effect when the sleep ends, as before. After a wake,
2773/// which only the scheduler's own sends (a timer, a child-exit `SIGCHLD`) can
2774/// commit for a stop on a `nanosleep`, a pending stop ends it, as every wake
2775/// did before, so that the stop takes effect promptly.
2776///
2777/// `SIGCHLD` and the signals a host-timed source armed by a guest can post
2778/// (`host_timed_signals`) are held: the kernel can post them at a moment set by
2779/// host timing, and `/proc` cannot tell that copy from one a guest sent. They
2780/// never end a sleep before it parks, nor does the scheduler record a guest's
2781/// cross-task send of one for a sleep. A wake the scheduler committed for its
2782/// own child-exit `SIGCHLD` ends the sleep with `Held` only while the copy it
2783/// sent is still pending in the process's shared queue
2784/// (`ChildExitSigchldControl::PlacedPending`, which `/proc` cannot answer),
2785/// `/proc` shows it pending, and it is unblocked and caught or default-fatal
2786/// (`held_wake_ends_sleep`); a host-timed held signal, from the mask and
2787/// dispositions alone. Like the precise futex
2788/// wait's check, this reads the pending set before the sleep parks as well as
2789/// after a wake: a signal sent while the thread was stopped short of filing
2790/// its request, as when its timeslice ended in the syscall's prehook, is
2791/// pending and was never recorded for its sleep. After a wake whose
2792/// non-held signals are no longer pending, a `SIGCONT` having cancelled a stop
2793/// or `sigaction(SIG_IGN)` having flushed it, a Linux sleep would still be
2794/// asleep, so this returns `None`.
2795///
2796/// A state that cannot be read is handled as a wait's first read is
2797/// (`read_wait_signal_state`, here `read_thread_wait_signal_state`, since
2798/// Reverie can report the creator's pid for a process a raw `clone` made): a
2799/// thread that no longer exists ends the call
2800/// with `ERESTARTNOINTR`, which nothing observes, and otherwise the run ends
2801/// with a diagnostic.
2802pub(crate) async fn sleep_signal<G, T>(
2803    guest: &mut G,
2804    check: SleepCheck,
2805    stops_end_it: bool,
2806) -> Result<SleepSignal, Error>
2807where
2808    G: Guest<Detcore<T>>,
2809    T: RecordOrReplay,
2810{
2811    if !guest
2812        .config()
2813        .backend_supports_blocked_wait_signal_interruption
2814    {
2815        return Ok(match check {
2816            SleepCheck::BeforePark => SleepSignal::None,
2817            SleepCheck::AfterWake(_) => SleepSignal::Held,
2818        });
2819    }
2820    let host_timed = host_timed_signals(guest).await;
2821    let held = kernel_sigset_bit(libc::SIGCHLD) | host_timed;
2822    let before_park = check == SleepCheck::BeforePark;
2823    // `held_woken`: the held signals the scheduler woke this sleep for.
2824    // `unknown_held_wake`: a wake that names no signal.
2825    let (deciding, held_woken, unknown_held_wake) = match check {
2826        SleepCheck::BeforePark => (!held, 0, false),
2827        SleepCheck::AfterWake(None) => (0, 0, true),
2828        SleepCheck::AfterWake(Some(woken)) => {
2829            let woken = woken
2830                .iter()
2831                .fold(0, |set, signal| set | kernel_sigset_bit(signal.raw()));
2832            (woken & !held, woken & held, false)
2833        }
2834    };
2835    if deciding != 0 || held_woken != 0 {
2836        let state = read_thread_wait_signal_state(guest.pid(), guest.tid())?;
2837        let mut ending = state.pending_interrupting(state.blocked, false) & deciding;
2838        if !stops_end_it && before_park {
2839            ending &= !(state.default_job_control_stops() | kernel_sigset_bit(libc::SIGSTOP));
2840        }
2841        if ending != 0 {
2842            return Ok(SleepSignal::Pending);
2843        }
2844        // A committed `SIGCHLD` counts only while the schedule says this thread
2845        // will take it (`PlacedPending`): a guest's thread-directed copy in its
2846        // private queue, or the scheduler's or a guest's shared copy still in
2847        // the window. `/proc` cannot tell either from a kernel copy that host
2848        // timing posted, so its bits only prompt the question: a woken sleep
2849        // asks for the `SIGCHLD` it was woken for, and any sleep asks when its
2850        // private queue holds one, which a guest sent before it parked
2851        // (https://github.com/rrnewton/hermit/issues/4132).
2852        let sigchld = kernel_sigset_bit(libc::SIGCHLD) & !host_timed;
2853        let committed = if asks_committed_sigchld(held_woken, &state, sigchld)
2854            && matches!(
2855                crate::tool_global::child_exit_sigchld(
2856                    guest,
2857                    crate::scheduler::child_exit_sigchld::ChildExitSigchldControl::PlacedPending,
2858                )
2859                .await,
2860                crate::scheduler::child_exit_sigchld::ChildExitSigchldAnswer::Placed(true)
2861            ) {
2862            // A committed copy shows in `/proc`: an invariant, not a second
2863            // condition.
2864            if state.pending & sigchld == 0 {
2865                tracing::error!(
2866                    "[dtid {}] a committed SIGCHLD is pending by the schedule but not in /proc",
2867                    guest.thread_state().dettid
2868                );
2869                debug_assert!(false, "a committed SIGCHLD is not pending in /proc");
2870            }
2871            sigchld
2872        } else {
2873            0
2874        };
2875        if let Some(verdict) = committed_sigchld_verdict(&state, held_woken, committed) {
2876            return Ok(verdict);
2877        }
2878        let held_woken = held_woken & !sigchld;
2879        if held_wake_ends_sleep(&state, held_woken) {
2880            return Ok(SleepSignal::Held);
2881        }
2882        return Ok(SleepSignal::None);
2883    }
2884    Ok(if unknown_held_wake {
2885        SleepSignal::Held
2886    } else {
2887        SleepSignal::None
2888    })
2889}

2891/// Whether a sleep's own check (`sleep_signal`) asks the scheduler whether a
2892/// committed `SIGCHLD` is pending for it (`PlacedPending`): it was woken for
2893/// one (`held_woken`), or its private queue holds one. The latter is how a
2894/// guest's thread-directed copy sent before the sleeper filed its request is
2895/// not lost: the sender's turn found no sleep to wake, so only the sleeper's
2896/// check before it parks sees it (https://github.com/rrnewton/hermit/issues/4132).
2897fn asks_committed_sigchld(
2898    held_woken: KernelSigset,
2899    state: &crate::syscalls::threads::KernelSignalState,
2900    sigchld: KernelSigset,
2901) -> bool {
2902    (held_woken | state.thread_pending) & sigchld != 0
2903}

2905/// What a committed `SIGCHLD` (`committed`, the scheduler's answer) makes of
2906/// the sleep: it ends it if unblocked and caught or acted on by default, as
2907/// the held signal the wake named (`Held`) or as one pending for it
2908/// (`Pending`); `None` leaves the decision to the rest of the check.
2909fn committed_sigchld_verdict(
2910    state: &crate::syscalls::threads::KernelSignalState,
2911    held_woken: KernelSigset,
2912    committed: KernelSigset,
2913) -> Option<SleepSignal> {
2914    let verdict = if held_woken & committed != 0 {
2915        SleepSignal::Held
2916    } else {
2917        SleepSignal::Pending
2918    };
2919    (state.interrupting(state.blocked) & committed != 0).then_some(verdict)
2920}

2922/// The size of a `struct signalfd_siginfo`, which has ssi_signo at 0,
2923/// ssi_code at 8, ssi_pid at 12, ssi_uid at 16, ssi_status at 40, ssi_int at
2924/// 44, ssi_utime at 56 and ssi_stime at 64.
2925const SIGNALFD_RECORD: usize = 128;
2926
2927fn signalfd_field(record: &[u8], at: usize) -> u32 {
2928    u32::from_ne_bytes(record[at..at + 4].try_into().expect("four bytes"))
2929}

2931/// For each `signalfd_siginfo` record in `bytes`, the child-exit ledger
2932/// question it raises (`Detcore::reconcile_signalfd_records`): a SIGCHLD with
2933/// a child-exit code is the kernel's copy for that child, a SIGCHLD the
2934/// scheduler sent (`is_own_sigchld_mark`) is its copy, and any other record
2935/// raises none.
2936fn signalfd_record_controls(
2937    bytes: &[u8],
2938    tracer: u32,
2939) -> Vec<Option<crate::scheduler::child_exit_sigchld::ChildExitSigchldControl>> {
2940    use crate::scheduler::child_exit_sigchld::ChildExitSigchldControl;
2941    use crate::scheduler::child_exit_sigchld::DequeuedCopy;
2942    bytes
2943        .chunks(SIGNALFD_RECORD)
2944        .map(|record| {
2945            if signalfd_field(record, 0) != libc::SIGCHLD as u32 {
2946                return None;
2947            }
2948            let code = signalfd_field(record, 8) as i32;
2949            let pid = signalfd_field(record, 12);
2950            let value = signalfd_field(record, 44) as i32;
2951            let copy = if crate::scheduler::child_exit_sigchld::is_child_exit_code(code) {
2952                DequeuedCopy::Kernel {
2953                    child: DetPid::from_raw(pid as i32),
2954                }
2955            } else if crate::scheduler::child_exit_sigchld::is_own_sigchld_mark(
2956                code, pid as i32, value, tracer,
2957            ) {
2958                DequeuedCopy::Scheduler
2959            } else if code == libc::SI_TKILL {
2960                DequeuedCopy::Thread
2961            } else {
2962                DequeuedCopy::Other
2963            };
2964            // Read, a kept record makes no `InboundSignal` request.
2965            Some(ChildExitSigchldControl::Dequeue {
2966                copy,
2967                admits: false,
2968            })
2969        })
2970        .collect()
2971}

2995/// Signalfd records rewritten by the child-exit ledger's answers.
2996#[derive(Debug, PartialEq, Eq)]
2997struct RewrittenSignalfdRecords {
2998    /// The buffer to write back: the kept records in order, then zeros over
2999    /// the bytes of dropped ones, so no dropped record stays readable.
3000    bytes: Vec<u8>,
3001    /// How many bytes the read returns.
3002    kept: usize,
3003}

3005/// Apply the ledger's `answers`, one per record of `bytes`
3006/// (`signalfd_record_controls`): a claimed duplicate or a duplicate scheduler
3007/// copy is dropped, a copy that stands for an owned child is given that
3008/// child's siginfo with zero CPU times, and every other record is kept as read.
3009/// `None` when no record changed.
3010fn apply_signalfd_answers(
3011    bytes: &[u8],
3012    answers: &[Option<crate::scheduler::child_exit_sigchld::ChildExitSigchldAnswer>],
3013) -> Option<RewrittenSignalfdRecords> {
3014    use crate::scheduler::child_exit_sigchld::ChildExitSigchldAnswer;
3015    use crate::scheduler::child_exit_sigchld::OwnCopy;
3016    let mut kept = Vec::with_capacity(bytes.len());
3017    let mut changed = false;
3018    for (record, answer) in bytes.chunks(SIGNALFD_RECORD).zip(answers) {
3019        let mut record = record.to_vec();
3020        match answer {
3021            Some(ChildExitSigchldAnswer::Claimed(true))
3022            | Some(ChildExitSigchldAnswer::OwnCopy(OwnCopy::Duplicate)) => {
3023                changed = true;
3024                continue;
3025            }
3026            Some(ChildExitSigchldAnswer::OwnCopy(OwnCopy::Deliver(own)))
3027            | Some(ChildExitSigchldAnswer::ClaimedFor(own)) => {
3028                // The record Linux writes for a child exit (SIL_CHLD) starts
3029                // zeroed and fills in only these: nothing of the copy it
3030                // replaces is kept, such as the own copy's SI_QUEUE mark in
3031                // ssi_int and ssi_ptr. CPU times are 0 (natively, host time).
3032                let signo = signalfd_field(&record, 0);
3033                record.fill(0);
3034                record[0..4].copy_from_slice(&signo.to_ne_bytes());
3035                record[8..12].copy_from_slice(&own.code.to_ne_bytes());
3036                record[12..16].copy_from_slice(&(own.pid as u32).to_ne_bytes());
3037                record[16..20].copy_from_slice(&own.uid.to_ne_bytes());
3038                record[40..44].copy_from_slice(&own.status.to_ne_bytes());
3039                changed = true;
3040            }
3041            _ => {}
3042        }
3043        kept.extend_from_slice(&record);
3044    }
3045    if !changed {
3046        return None;
3047    }
3048    let kept_len = kept.len();
3049    kept.resize(bytes.len() - bytes.len() % SIGNALFD_RECORD, 0);
3050    Some(RewrittenSignalfdRecords {
3051        bytes: kept,
3052        kept: kept_len,
3053    })
3054}

3416/// The refusal for `send`, or `None` when the kernel's choice of receiver and
3417/// the sender's dequeue are forced: the sender takes the signal on its way back
3418/// to user mode, before any other thread of the process can run
3419/// (https://github.com/rrnewton/hermit/issues/4034).
3420pub(crate) fn own_process_signal_refusal(
3421    send: &OwnProcessSignal,
3422) -> Option<OwnProcessSignalRefusal> {
3423    use OwnProcessSignalRefusal as R;
3424    let bit = kernel_sigset_bit(send.sig);
3425    if !OWN_PROCESS_SIGNALS.contains(&send.sig) {
3426        return Some(R::SignalNotAdmitted);
3427    }
3428    if (send.ignored | send.caught) & bit == 0 {
3429        return Some(R::DefaultDisposition);
3430    }
3431    if send.record_replay {
3432        return Some(R::RecordReplay);
3433    }
3434    if send.schedule_replay {
3435        return Some(R::ScheduleReplay);
3436    }
3437    if send.stacktrace_signal {
3438        return Some(R::StacktraceSignal);
3439    }
3440    if send.blocked & bit != 0 {
3441        return Some(R::SenderBlocks);
3442    }
3443    for (tid, state) in &send.inputs.siblings {
3444        if *state == SiblingSignalState::OutOfSchedule {
3445            return Some(R::OutOfSchedule(*tid));
3446        }
3447    }
3448    for (tid, state) in &send.inputs.siblings {
3449        if *state == SiblingSignalState::InSchedule(None) {
3450            return Some(R::UnknownMask(*tid));
3451        }
3452    }
3453    if !send.names_sender {
3454        for (tid, state) in &send.inputs.siblings {
3455            if matches!(state, SiblingSignalState::InSchedule(Some(mask)) if mask & bit == 0) {
3456                return Some(R::NotSoleReceiver(*tid));
3457            }
3458        }
3459    }
3460    let unblocked = !send.blocked;
3461    let private = without_perf_event_signal(send.thread_pending) & unblocked;
3462    if private != 0 {
3463        return Some(R::PrivatePending(private));
3464    }
3465    let host_timed = send.inputs.host_timed & unblocked;
3466    if host_timed != 0 {
3467        return Some(R::HostTimedSource(host_timed));
3468    }
3469    let foreign = send.inputs.foreign_exit_signals & unblocked;
3470    if foreign != 0 {
3471        return Some(R::ForeignExitSignal(foreign));
3472    }
3473    // A signal numbered at or below `sig` is dequeued first or coalesces.
3474    let at_or_below = bit | (bit - 1);
3475    let undelivered = send.inputs.undelivered_scheduler_signals & unblocked & at_or_below;
3476    if undelivered != 0 {
3477        return Some(R::UndeliveredSchedulerSignal(undelivered));
3478    }
3479    if send.posthook_anchor {
3480        return Some(R::PosthookAnchor);
3481    }
3482    if send.signalfd {
3483        return Some(R::Signalfd);
3484    }
3485    if send.inputs.exec_in_flight {
3486        return Some(R::ExecInFlight);
3487    }
3488    if send.inputs.guest_marker_signal {
3489        return Some(R::GuestMarkerSignal);
3490    }
3491    None
3492}

3537/// One mapping of a `/proc/<pid>/smaps` listing.
3538#[derive(Clone, Copy, Debug, PartialEq, Eq)]
3539struct Mapping {
3540    start: usize,
3541    end: usize,
3542    /// Whether user mode can read it: on x86 any of read, write or execute
3543    /// permission allows a read; none (`---`) faults.
3544    user_readable: bool,
3545    /// `VmFlags` `gd`: a fault just below it can grow it downwards.
3546    growsdown: bool,
3547}

3549/// The mappings of a `/proc/<pid>/smaps` listing, in address order.
3550fn parse_smaps(smaps: &str) -> Vec<Mapping> {
3551    let mut mappings: Vec<Mapping> = Vec::new();
3552    for line in smaps.lines() {
3553        if let Some(flags) = line.strip_prefix("VmFlags:") {
3554            if let Some(last) = mappings.last_mut() {
3555                last.growsdown = flags.split_whitespace().any(|flag| flag == "gd");
3556            }
3557            continue;
3558        }
3559        let mut fields = line.split_whitespace();
3560        let Some((start, end)) = fields.next().and_then(|range| range.split_once('-')) else {
3561            continue;
3562        };
3563        let (Ok(start), Ok(end), Some(perms)) = (
3564            usize::from_str_radix(start, 16),
3565            usize::from_str_radix(end, 16),
3566            fields.next(),
3567        ) else {
3568            continue;
3569        };
3570        if perms.len() < 4 {
3571            continue;
3572        }
3573        mappings.push(Mapping {
3574            start,
3575            end,
3576            user_readable: perms.chars().take(3).any(|perm| perm != '-'),
3577            growsdown: false,
3578        });
3579    }
3580    mappings.sort_by_key(|mapping| mapping.start);
3581    mappings
3582}

3584/// Why a byte is not readable by user mode (`first_user_unreadable`).
3585#[derive(Clone, Copy, Debug, PartialEq, Eq)]
3586enum Unreadable {
3587    /// In a mapping without read, write or execute permission.
3588    NoPermission,
3589    /// In no mapping; `below_growsdown` when the next mapping above it can
3590    /// grow down into it.
3591    Hole { below_growsdown: bool },
3592}
3593
3594/// Where the first byte of `[address, address + len)` that user mode cannot
3595/// read lies, or `None` when user mode can read them all. A range that
3596/// overflows the address space counts as a hole.
3597fn first_user_unreadable(mappings: &[Mapping], address: usize, len: usize) -> Option<Unreadable> {
3598    let Some(end) = address.checked_add(len) else {
3599        return Some(Unreadable::Hole {
3600            below_growsdown: false,
3601        });
3602    };
3603    let mut next = address;
3604    while next < end {
3605        match mappings.iter().find(|mapping| mapping.end > next) {
3606            Some(mapping) if mapping.start <= next => {
3607                if !mapping.user_readable {
3608                    return Some(Unreadable::NoPermission);
3609                }
3610                next = mapping.end;
3611            }
3612            above => {
3613                return Some(Unreadable::Hole {
3614                    below_growsdown: above.is_some_and(|mapping| mapping.growsdown),
3615                });
3616            }
3617        }
3618    }
3619    None
3620}

3626/// SIGCONT at the exiting process's physical exit (`kill_orphaned_pgrp` in

3634        }

3636    }

3639/// Whether a held signal the scheduler woke a sleep for (`held_woken`, a
3640/// child-exit `SIGCHLD`) ends it: as Linux delivery would, only when this
3641/// thread's mask leaves it unblocked and it is caught or acted on by default
3642/// (`KernelSignalState::interrupting`). A blocked one stays pending, and a
3643/// default-ignored one (`SIGCHLD` without a handler) is discarded, and either
3644/// way the sleep goes on (https://github.com/rrnewton/hermit/issues/4126).
3645///
3646/// The caller has already removed a committed `SIGCHLD` that no longer counts:
3647/// the scheduler sends it to the process, so the kernel gives it to whichever
3648/// thread resumes first without blocking it, and a schedule may run a sibling
3649/// first (`--chaos-target-races`). Taken by that sibling, it is gone from the
3650/// child-exit ledger's send window (`ChildExitSigchldControl::PlacedPending`),
3651/// and the sleep goes on to its original deadline, as a futex wait woken for a
3652/// signal waits again (`futex_wait_held_after_wake`). Answering EINTR for it
3653/// anyway was a result Linux cannot give: the call interrupted, the handler run
3654/// in another thread (the Claude re-check of
3655/// https://github.com/rrnewton/hermit/pull/4131). A held signal a host-timed
3656/// source can also post is decided from the mask and dispositions alone.
3657fn held_wake_ends_sleep(
3658    state: &crate::syscalls::threads::KernelSignalState,
3659    held_woken: KernelSigset,
3660) -> bool {
3661    state.interrupting(state.blocked) & held_woken != 0
3662}

4365    /// The parent's ownership of the forked child: it is killed and reaped
4366    /// when the owner is dropped, so a panic or a cancelled await in the test
4367    /// leaves no process behind. `finish` does the same on the normal path
4368    /// and checks that the child was reaped.
4369    struct ChildOwner {
4370        pid: libc::pid_t,
4371        reaped: bool,
4372    }
4373
4374    impl ChildOwner {
4375        /// Wait for the child to stop. A wait that reports it exited or
4376        /// killed reaped it, so the owner is disarmed before anything can
4377        /// unwind: its pid may already belong to another process.
4378        /// An ECHILD means the child is no longer ours to wait for (it was
4379        /// reaped already), so it disarms the owner too.
4380        fn wait_stopped(&mut self) -> Result<(), Result<(libc::pid_t, libc::c_int), i32>> {
4381            let waited = wait_for(self.pid, libc::WUNTRACED);
4382            match waited {
4383                Ok((pid, status))
4384                    if pid == self.pid
4385                        && (libc::WIFEXITED(status) || libc::WIFSIGNALED(status)) =>
4386                {
4387                    self.reaped = true;
4388                }
4389                Err(libc::ECHILD) => self.reaped = true,
4390                _ => {}
4391            }
4392            match waited {
4393                Ok((pid, status)) if pid == self.pid && libc::WIFSTOPPED(status) => Ok(()),
4394                other => Err(other),
4395            }
4396        }
4397
4398        /// Whether the owner had disarmed itself, read once the child is
4399        /// known to be reaped. The owner is disarmed from here on whatever
4400        /// the answer, so a failed check of it unwinds without signalling a
4401        /// pid that may have been reused.
4402        fn disarmed_after_reap(&mut self) -> bool {
4403            std::mem::replace(&mut self.reaped, true)
4404        }
4405
4406        fn kill_and_reap(&mut self) -> Result<(libc::pid_t, libc::c_int), i32> {
4407            // SAFETY: killing our own child, which is not yet reaped.
4408            unsafe { libc::kill(self.pid, libc::SIGKILL) };
4409            let reaped = wait_for(self.pid, 0);
4410            self.reaped = true;
4411            reaped
4412        }
4413
4414        fn finish(mut self) {
4415            let reaped = self.kill_and_reap();
4416            assert!(
4417                matches!(reaped, Ok((pid, status)) if pid == self.pid
4418                    && libc::WIFSIGNALED(status)
4419                    && libc::WTERMSIG(status) == libc::SIGKILL),
4420                "the child was not reaped: {reaped:?}"
4421            );
4422        }
4423    }

4426        fn drop(&mut self) {
4427            if !self.reaped {
4428                // Unwinding already: the reap's outcome is not checked here.
4429                let _ = self.kill_and_reap();
4430            }
4431        }
