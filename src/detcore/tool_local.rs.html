108/// Deserializes a descriptor table with its aliases restored, so a table
109/// carried across fork (or any other state migration) keeps one open file
110/// description object per `OpenFileId`.
111fn deserialize_interned_file_handles<'de, D>(
112    deserializer: D,
113) -> Result<HashMap<RawFd, DetFd>, D::Error>
114where
115    D: serde::Deserializer<'de>,
116{
117    let mut handles = <HashMap<RawFd, DetFd> as Deserialize>::deserialize(deserializer)?;
118    crate::fd::intern_open_files(&mut handles);
119    Ok(handles)
120}

166/// Whether a pidfd target is proven to use the caller's exact descriptor table.
167///
168/// Linux pidfds name a specific task, and `pidfd_open(2)` accepts a thread-group
169/// leader. Equal TGIDs alone are not enough: a `CLONE_THREAD` child may omit
170/// `CLONE_FILES` and therefore have a different `files_struct`. Restrict the
171/// modeled path to the leader itself, where `target == getpid() == gettid()`
172/// proves that the task named by the pidfd and the caller are identical.
173pub(crate) fn pidfd_getfd_targets_calling_task(
174    target: Option<DetPid>,
175    current_tgid: DetPid,
176    current_tid: DetTid,
177) -> bool {
178    target == Some(current_tgid) && current_tid == current_tgid
179}

219/// A single POSIX per-process interval timer created by `timer_create(2)`.
220///
221/// Detcore records arming against virtual time and schedules supported
222/// `SIGEV_SIGNAL` notifications through the deterministic scheduler.
223// AUTONOMOUS-BOT-IMPLEMENTED
224// TODO-HUMAN-REVIEW(#869)
225#[derive(Debug, Clone, Serialize, Deserialize)]
226struct PosixTimer {
227    /// Reload interval for periodic timers, in nanoseconds (0 => one-shot).
228    interval_ns: u64,
229    /// Absolute virtual-time deadline of the next expiration, or `None` when the
230    /// timer is disarmed (`it_value == 0`).
231    deadline: Option<LogicalTime>,
232    /// Signal number configured by `timer_create`, or `None` for notifications
233    /// that Detcore cannot deliver through its scheduler.
234    signal: Option<i32>,
235    /// The thread a `SIGEV_THREAD_ID` timer notifies, or `None` for a
236    /// process-directed one.
237    #[serde(default)]
238    thread: Option<DetTid>,
239}

241/// The set of POSIX timers owned by a *process*.
242///
243/// Timers are shared among all threads of a process and, per POSIX, are **not**
244/// inherited across `fork(2)`. Detcore therefore shares this table on
245/// `CLONE_THREAD` and starts a fresh, empty table for every new process (see
246/// `init_thread_state`). Successful exec deletes the timers as well.
247#[derive(Debug, Clone, Default, Serialize, Deserialize)]
248pub struct PosixTimers {
249    /// Deterministic id allocator. Kernel `timer_t`s are opaque, so we hand out
250    /// ids as 0, 1, 2, ... in creation order to keep them reproducible.
251    next_id: i32,
252    timers: HashMap<i32, PosixTimer>,
253}
254
255impl PosixTimers {
256    // AUTONOMOUS-BOT-IMPLEMENTED
257    // TODO-HUMAN-REVIEW(#869)
258    /// Allocate a new (disarmed) timer, returning its deterministic id.
259    /// `thread` is the thread a `SIGEV_THREAD_ID` timer notifies.
260    pub(crate) fn create(&mut self, signal: Option<i32>, thread: Option<DetTid>) -> i32 {
261        let id = self.next_id;
262        self.next_id += 1;
263        self.timers.insert(
264            id,
265            PosixTimer {
266                interval_ns: 0,
267                deadline: None,
268                signal,
269                thread,
270            },
271        );
272        id
273    }
274
275    /// Arm or disarm timer `id`. `interval_ns` is the periodic reload and
276    /// `deadline` the absolute virtual-time expiration (the caller derives it
277    /// from the request flags and the current virtual clock; `None` disarms).
278    /// Returns the previous `(remaining_ns, interval_ns)` for `old_value`, or
279    /// `None` if the id is unknown.
280    pub(crate) fn settime(
281        &mut self,
282        id: i32,
283        interval_ns: u64,
284        deadline: Option<LogicalTime>,
285        now: LogicalTime,
286    ) -> Option<(u64, u64)> {
287        let timer = self.timers.get_mut(&id)?;
288        let old = (
289            remaining_ns(timer.deadline, timer.interval_ns, now),
290            timer.interval_ns,
291        );
292        timer.interval_ns = interval_ns;
293        timer.deadline = deadline;
294        Some(old)
295    }
296
297    /// Report the current `(remaining_ns, interval_ns)` for `timer_gettime`, or
298    /// `None` if the id is unknown.
299    pub(crate) fn gettime(&self, id: i32, now: LogicalTime) -> Option<(u64, u64)> {
300        let timer = self.timers.get(&id)?;
301        Some((
302            remaining_ns(timer.deadline, timer.interval_ns, now),
303            timer.interval_ns,
304        ))
305    }
306
307    /// Whether a timer with this id currently exists.
308    pub(crate) fn contains(&self, id: i32) -> bool {
309        self.timers.contains_key(&id)
310    }
311
312    // AUTONOMOUS-BOT-IMPLEMENTED
313    // TODO-HUMAN-REVIEW(#869)
314    pub(crate) fn signal(&self, id: i32) -> Option<Option<i32>> {
315        self.timers.get(&id).map(|timer| timer.signal)
316    }
317
318    /// The thread timer `id` notifies, if it is a `SIGEV_THREAD_ID` timer.
319    pub(crate) fn thread(&self, id: i32) -> Option<DetTid> {
320        self.timers.get(&id).and_then(|timer| timer.thread)
321    }
322
323    /// Remove a timer; returns whether it existed.
324    pub(crate) fn remove(&mut self, id: i32) -> bool {
325        self.timers.remove(&id).is_some()
326    }
327
328    /// Linux deletes all POSIX timers on successful exec. Keep the process's
329    /// ID allocator, but invalidate armed, disarmed and non-notifying timers.
330    pub(crate) fn clear_for_exec(&mut self) {
331        self.timers.clear();
332    }
333}

393/// Nanoseconds remaining until `deadline` relative to `now`, saturating at 0.
394/// A disarmed timer (`None`) or an elapsed one-shot reports 0. Periodic timers
395/// advance arithmetically to their next virtual deadline.
396fn remaining_ns(deadline: Option<LogicalTime>, interval_ns: u64, now: LogicalTime) -> u64 {
397    match deadline {
398        Some(d) if d > now => d.as_nanos() - now.as_nanos(),
399        Some(d) if interval_ns != 0 => {
400            let elapsed = now.as_nanos() - d.as_nanos();
401            interval_ns - (elapsed % interval_ns)
402        }
403        Some(_) => 0,
404        None => 0,
405    }
406}

429impl<T: RecordOrReplay> Detcore<T> {
430    /// Delegate to the record/replay tool without collapsing a tool failure
431    /// into a guest-visible errno. Syscall errors remain [`Error::Errno`], while
432    /// failures of the record/replay machinery retain their non-errno variant.
433    pub(crate) async fn record_or_replay_preserving_tool_errors<G, S>(
434        &self,
435        guest: &mut G,
436        syscall: S,
437    ) -> Result<i64, Error>
438    where
439        G: Guest<Self>,
440        S: Into<Syscall>,
441    {
442        preserve_record_or_replay_error(
443            self.record_or_replay
444                .handle_syscall_event(&mut guest.into_guest(), syscall.into())
445                .await,
446        )?
447        .map_err(Error::from)
448    }
449
450    /// Helper function for delegating the injection of a syscall to the
451    /// record_or_replay tool.
452    ///
453    /// It is important to classify the cases where we need to call `inject`. We
454    /// have three main choices to make when handling a syscall:
455    ///  1. Fully determinize the syscall. In this case, it doesn't need to call
456    ///     `inject` at all.
457    ///  2. Partially determinize the syscall. In this case, it can't fully
458    ///     determinize a syscall but only part of it. For example, a `stat` syscall
459    ///     is ultimately non-deterministic because the file may not always exist.
460    ///     However, the mtime or inode numbers can be made deterministic.
461    ///  3. The syscall cannot be determinized at all. For example, a call to
462    ///     `recvfrom` cannot be made deterministic.
463    ///
464    /// Thus, this is called whenever `inject` would be called for non-bookkeeping
465    /// operations.
466    pub(crate) async fn record_or_replay<G, S>(
467        &self,
468        guest: &mut G,
469        syscall: S,
470    ) -> Result<i64, Errno>
471    where
472        G: Guest<Self>,
473        S: Into<Syscall>,
474    {
475        self.record_or_replay
476            .handle_syscall_event(&mut guest.into_guest(), syscall.into())
477            .await
478            // TODO: Get rid of this and make this whole function use the Error type.
479            .map_err(|err| err.into_errno().unwrap())
480    }
481}

483fn preserve_record_or_replay_error(
484    result: Result<i64, Error>,
485) -> Result<Result<i64, Errno>, Error> {
486    match result {
487        Ok(value) => Ok(Ok(value)),
488        Err(Error::Errno(error)) => Ok(Err(error)),
489        Err(error) => Err(error),
490    }
491}

493/// Linux reports a positive byte count when a later guest-visible write error
494/// follows partial progress. An error in the record/replay tool is different:
495/// replay itself failed, so it must remain fatal even after earlier bytes were
496/// accepted.
497pub(crate) fn finish_partial_record_or_replay_write(
498    written: i64,
499    error: Error,
500) -> Result<i64, Error> {
501    match error {
502        Error::Errno(_) if written > 0 => Ok(written),
503        error => Err(error),
504    }
505}

647    /// Fork the file table for `child`.
648    ///
649    /// FORGETTING IS DELIBERATELY *NOT* DONE HERE, and the reason is that this
650    /// fork does not produce a second copy of the cache. `DetFd` holds its
651    /// `OpenFileDescription` behind an `Arc<Mutex<..>>`, so `file_handles
652    /// .clone()` below hands the child the *same* description objects the
653    /// parent holds. That mirrors the kernel, where `fork(2)` duplicates the
654    /// descriptor table while both tables keep pointing at one open file
655    /// description, and an `flock(2)` lock belongs to that description. A
656    /// `flock` through either task therefore mutates the one shared record and
657    /// both sides observe it; there is no second copy that could go stale, so
658    /// invalidating here destroys correct information and buys nothing.
659    ///
660    /// The case that genuinely needs invalidation is a *separately hosted* tool
661    /// state — a real process clone under DBT, where the in-process runtime is
662    /// copied into a new address space and the `Arc` stops being shared. That
663    /// is handled at its own boundary, by
664    /// `reverie_dbt_runtime_process_clone_result`, which calls
665    /// `forget_flock_modes` only on a clone the kernel reported as successful.
666    ///
667    /// Forgetting here as well was not merely redundant, it was harmful: it
668    /// marked every descriptor unknown on any fork, and
669    /// `has_unsafe_vfork_flock_state` refuses `vfork` on unknown as well as on
670    /// held. So any guest that forked and later vforked was refused even when
671    /// it had never called `flock` at all.

749    /// Called only in the actual shared guest descriptor table, after the
750    /// global state confirms this is the container's first image. TCGETS is
751    /// read-only: neither terminal settings nor OFD flags are changed.
752    fn discover_inherited_terminals(&mut self, descriptors: impl IntoIterator<Item = RawFd>) {
753        for fd in descriptors {
754            let Some(detfd) = self.file_handles.get(&fd) else {
755                continue;
756            };
757            let Some(stat) = detfd.stat() else { continue };
758            if stat.mode & libc::S_IFMT != libc::S_IFCHR {
759                continue;
760            }
761            let mut settings: libc::termios = unsafe { std::mem::zeroed() };
762            // SAFETY: TCGETS writes only to this live, adequately sized buffer.
763            if unsafe { libc::ioctl(fd, libc::TCGETS, &raw mut settings) } != 0 {
764                continue;
765            }
766            detfd.set_inherited_external_read(crate::fd::InheritedExternalRead {
767                origin: detfd.open_file_id(),
768                dev: stat.dev,
769                inode: stat.inode,
770                rdev: stat.rdev,
771                file_type: libc::S_IFCHR,
772                access_mode: detfd.status_flags() & libc::O_ACCMODE,
773                resource: detfd.resource(),
774            });
775        }

778    /// Restore only external-read descriptions proven to survive the prepared exec. Unknown
779    /// descriptors and guest-created PTYs are not promoted by rediscovery.
780    fn restore_inherited_reads(
781        &mut self,
782        owner: DetTid,
783        inherited_reads: BTreeMap<RawFd, crate::fd::InheritedExternalRead>,
784    ) -> Result<(), Errno> {
785        let mut aliases: HashMap<OpenFileId, DetFd> = HashMap::new();
786        for (fd, terminal) in inherited_reads {
787            self.discover_fd_from_current_process(owner, fd)?;
788            let detfd = self.file_handles.get_mut(&fd).ok_or(Errno::EBADF)?;
789            let stat =
790                stat::fstat(unsafe { BorrowedFd::borrow_raw(fd) }).map_err(|_| Errno::last())?;
791            if stat.st_mode & libc::S_IFMT != terminal.file_type
792                || (stat.st_dev, stat.st_ino, stat.st_rdev)
793                    != (terminal.dev, terminal.inode, terminal.rdev)
794                || detfd.status_flags() & libc::O_ACCMODE != terminal.access_mode
795            {
796                return Err(Errno::EIO);
797            }
798            // Do not repeat TCGETS here: a hung-up terminal may return EIO
799            // from that query while still being the inherited open file.
800            if let Some(alias) = aliases.get(&terminal.origin) {
801                if detfd.status_flags() != alias.status_flags() {
802                    return Err(Errno::EIO);
803                }
804                detfd.share_open_file_with(alias);
805            } else {
806                detfd.set_resource(terminal.resource.clone());
807                detfd.set_inherited_external_read(terminal.clone());
808                aliases.insert(terminal.origin, detfd.clone());
809            }
810        }
811        Ok(())
812    }

860    pub(crate) fn open_files_closed_on_exec(&self, table_is_shared: bool) -> Vec<OpenFileId> {
861        if table_is_shared {
862            return Vec::new();
863        }
864
865        let mut open_files = HashMap::new();
866        for detfd in self.file_handles.values() {
867            let id = detfd.open_file_id();
868            let total_aliases = detfd.open_file_alias_count();
869            let entry = open_files.entry(id).or_insert((0, total_aliases, true));
870            debug_assert_eq!(entry.1, total_aliases);
871            entry.0 += 1;
872            entry.2 &= detfd.is_cloexec();
873        }
874
875        let mut closed: Vec<_> = open_files
876            .into_iter()
877            .filter_map(|(id, (table_aliases, total_aliases, all_cloexec))| {
878                (all_cloexec && table_aliases == total_aliases).then_some(id)
879            })
880            .collect();
881        closed.sort();
882        closed
883    }

942    // TODO-HUMAN-REVIEW(PR-845): Review SaBRe on-demand inherited descriptor discovery.
943    fn discover_fd_from_current_process(&mut self, owner: DetTid, fd: RawFd) -> Result<(), Errno> {
944        if self.file_handles.contains_key(&fd) {
945            return Ok(());
946        }
947
948        let fd_flags = unsafe { libc::fcntl(fd, libc::F_GETFD) };
949        let status_flags = unsafe { libc::fcntl(fd, libc::F_GETFL) };
950        if fd_flags == -1 || status_flags == -1 {
951            return Err(Errno::last());
952        }
953        let raw_stat =
954            stat::fstat(unsafe { BorrowedFd::borrow_raw(fd) }).map_err(|_| Errno::last())?;
955        let file_type = stat::SFlag::from_bits_truncate(raw_stat.st_mode);
956        let ty = if file_type.contains(stat::SFlag::S_IFIFO) {
957            FdType::Pipe
958        } else if file_type.contains(stat::SFlag::S_IFSOCK) {
959            FdType::Socket
960        } else {
961            FdType::Regular
962        };
963        let mut flags = OFlag::from_bits_truncate(status_flags);
964        let physically_nonblocking = flags.contains(OFlag::O_NONBLOCK);
965        // Discovered descriptors have unknown provenance, so an observed
966        // O_NONBLOCK bit must remain guest-visible. Detcore-created scheduler
967        // pipes are registered when created and do not reach this fallback.
968        if fd_flags & libc::FD_CLOEXEC != 0 {
969            flags.insert(OFlag::O_CLOEXEC);
970        }
971        self.add_fd(owner, fd, flags, ty, Some(raw_stat.into()))?;
972        // This newly discovered description predates our observation. Its
973        // kernel lock may be held, but no cached lock claim existed to go stale.
974        self.with_detfd(fd, |detfd| detfd.mark_flock_mode_unobserved())?;
975        // A placeholder for a procfs file Detcore renders, which the backend
976        // rediscovers after `exec`: the file again, at the position the old
977        // image left and with the snapshot it took, as a backend that keeps
978        // Detcore's state across `exec` has them (`syscalls::files::rendered_procfs`).
979        // The snapshot is the one its name carries (`procfs::placeholder_name`).
980        // Only an empty regular file with no link can be one.
981        let may_be_placeholder = raw_stat.st_mode & libc::S_IFMT == libc::S_IFREG
982            && raw_stat.st_nlink == 0
983            && raw_stat.st_size == 0;
984        if let Some((path, snapshot)) = may_be_placeholder.then(|| placeholder_behind(fd)).flatten()
985        {
986            crate::syscalls::rendered_procfs::register_placeholder(
987                raw_stat.st_dev,
988                raw_stat.st_ino,
989                path,
990            );
991            // Every open makes its own placeholder, so another descriptor on
992            // the same one is the same open file: one cursor, as before.
993            if let Some(alias) = self.rendered_alias(fd, raw_stat.st_dev, raw_stat.st_ino) {
994                let mut alias = alias.with_fd(fd);
995                alias.set_cloexec(fd_flags & libc::FD_CLOEXEC != 0);
996                self.add_detfd(alias);
997                return Ok(());
998            }
999            // SAFETY: lseek only reads the descriptor's position.
1000            let offset = unsafe {
1001                crate::util::raw_syscall(
1002                    libc::SYS_lseek,
1003                    [fd as u64, 0, libc::SEEK_CUR as u64, 0, 0, 0],
1004                )
1005            };
1006            let offset = usize::try_from(offset).unwrap_or(0);
1007            let procfs = crate::procfs::ProcfsFile::from_path(path)
1008                .expect("a rendered path is a procfs file")
1009                .into_rendered();
1010            let procfs = if snapshot.is_empty() {
1011                procfs
1012            } else {
1013                procfs.with_rendered_snapshot(snapshot)
1014            };
1015            self.with_detfd(fd, |detfd| {
1016                detfd.set_path(path);
1017                detfd.set_procfs(procfs.clone());
1018                detfd.set_procfs_offset(offset);
1019            })?;
1020        }
1021        // A fixed procfs file the kernel opened (through a symbolic link),
1022        // which the old image read from Detcore's rendering: the file again,
1023        // at the position Detcore kept in the kernel
1024        // (`DetFd::procfs_keeps_kernel_position`), so the new image does not
1025        // read the host's file. procfs files report size 0.
1026        else if raw_stat.st_mode & libc::S_IFMT == libc::S_IFREG
1027            && raw_stat.st_size == 0
1028            && let Some(procfs) = fixed_procfs_behind(fd)
1029        {
1030            // SAFETY: lseek only reads the descriptor's position.
1031            let offset = unsafe {
1032                crate::util::raw_syscall(
1033                    libc::SYS_lseek,
1034                    [fd as u64, 0, libc::SEEK_CUR as u64, 0, 0, 0],
1035                )
1036            };
1037            let offset = usize::try_from(offset).unwrap_or(0);
1038            let path = procfs
1039                .rendered_path()
1040                .expect("a fixed procfs file has a path");
1041            self.with_detfd(fd, |detfd| {
1042                detfd.set_path(path);
1043                detfd.set_procfs(procfs.clone());
1044                detfd.set_procfs_offset(offset);
1045            })?;
1046        }
1047        if let Some(resource) = stdio_resource(fd) {
1048            self.with_detfd(fd, |detfd| detfd.set_resource(resource.clone()))?;
1049        }
1050        if ty == FdType::Pipe && physically_nonblocking {
1051            self.with_detfd(fd, |detfd| detfd.set_physically_nonblocking())?;
1052        }
1053        Ok(())
1054    }

2246    Published,
2247    Invalidated,

2250impl ChildCpuPublication {

2253            Vec::new(),
2254        ))))

2257    fn finish(&self, published: bool) {
2258        let waiters = {
2259            let mut state = self.0.lock().expect("child CPU publication mutex poisoned");
2260            if matches!(*state, ChildCpuPublicationState::Invalidated)
2261                || published && matches!(*state, ChildCpuPublicationState::Published)
2262            {
2263                return;
2264            }
2265            let next = if published {
2266                ChildCpuPublicationState::Published
2267            } else {
2268                ChildCpuPublicationState::Invalidated
2269            };
2270            match std::mem::replace(&mut *state, next) {
2271                ChildCpuPublicationState::Pending(waiters) => waiters,
2272                _ => Vec::new(),
2273            }
2274        };
2275        // Do not call an arbitrary task waker under either accounting lock.
2276        for waiter in waiters.into_iter().filter_map(|waiter| waiter.upgrade()) {
2277            waiter.wake();
2278        }
2279    }

2281    async fn wait(&self) -> bool {
2282        let waiter = Arc::new(futures::task::AtomicWaker::new());
2283        {
2284            let mut state = self.0.lock().expect("child CPU publication mutex poisoned");
2285            match &mut *state {
2286                ChildCpuPublicationState::Pending(waiters) => {
2287                    waiters.retain(|waiter| waiter.strong_count() != 0);
2288                    waiters.push(Arc::downgrade(&waiter));
2289                }
2290                ChildCpuPublicationState::Published => return true,
2291                ChildCpuPublicationState::Invalidated => return false,
2292            }
2293        }
2294        futures::future::poll_fn(|cx| {
2295            waiter.register(cx.waker());
2296            let state = self.0.lock().expect("child CPU publication mutex poisoned");
2297            match *state {
2298                ChildCpuPublicationState::Pending(_) => std::task::Poll::Pending,
2299                ChildCpuPublicationState::Published => std::task::Poll::Ready(true),
2300                ChildCpuPublicationState::Invalidated => std::task::Poll::Ready(false),
2301            }
2302        })
2303        .await
2304    }

2327                previous.user = previous.user.max(child.user);
2328                previous.system = previous.system.max(child.system);
2329                previous.children_user = previous.children_user.max(child.children_user);
2330                previous.children_system = previous.children_system.max(child.children_system);
2331            })
2332            .or_insert(child);
2333    }
2334
2335    fn reap_child(&mut self, pid: DetPid) {
2336        let Some(child) = self.exited_children.remove(&pid) else {
2337            return;
2338        };
2339        self.snapshot.children_user =
2340            self.snapshot.children_user + child.user + child.children_user;
2341        self.snapshot.children_system =
2342            self.snapshot.children_system + child.system + child.children_system;
2343    }
2344
2345    fn prepare_child(&mut self, pid: DetPid) {
2346        self.exited_children.remove(&pid);
2347    }

2803/// RR-style stable per-thread slowdown factor for chaos scheduling.
2804///
2805/// Returns the multiplier applied to a thread's *mean* chaos timeslice length.
2806/// A factor `> 1.0` means the thread is preempted less often (runs "faster"
2807/// between preemptions — a slower relative wall-clock for its peers), `< 1.0`
2808/// means it is preempted more often. The factor is drawn log-uniformly from
2809/// `[1/max_factor, max_factor]`, so slow and fast are symmetric in log-space
2810/// and `1.0` is the geometric center.
2811///
2812/// The value is a **pure, deterministic function** of `(sched_seed, dettid)`:
2813/// it depends on no run-order, no wall-clock, and no shared PRNG state, so it
2814/// is stable for a thread across the whole run and reproducible under a fixed
2815/// seed (unlike the per-timeslice `chaos_prng` draw, which is redrawn every
2816/// slice and averages out over a long run). Threads are perturbed by a fixed
2817/// constant so the factor stream differs from other seed-derived streams (e.g.
2818/// `post_fork_prng`) that also start from `sched_seed`.
2819///
2820/// `max_factor <= 1.0` disables the spread and returns `1.0` (nominal) for
2821/// every thread; callers validate `max_factor >= 1.0`.
2822// AUTONOMOUS-BOT-IMPLEMENTED
2823// TODO-HUMAN-REVIEW(PR-1151)
2824/// `epoch` selects a deterministic chaos EPOCH: the factor is redrawn per
2825/// (thread, epoch) so a thread's bias changes in deterministic phases across a
2826/// long run instead of staying fixed. `epoch == 0` reproduces the epoch-less
2827/// per-thread-slowdown value EXACTLY (the epoch term is `0` and cancels out of
2828/// the mix), so enabling epochs never perturbs the first epoch's behavior.
2829pub(crate) fn chaos_per_thread_slowdown_factor(
2830    sched_seed: u64,
2831    dettid: DetTid,
2832    epoch: u64,
2833    max_factor: f64,
2834) -> RcbTimeMultiplier {
2835    // `<=` (rather than `!(max_factor > 1.0)`) keeps clippy's partial-ord lint
2836    // happy; validate_invariants already rejects non-finite factors upstream.
2837    if max_factor <= 1.0 {
2838        return RcbTimeMultiplier::ONE;
2839    }
2840    // Mix the seed with the (stable) deterministic thread id to give each
2841    // thread its own point in the factor distribution. The salt keeps this
2842    // stream distinct from other sched_seed-derived streams.
2843    const SLOWDOWN_SALT: u64 = 0x736c_6f77_646f_776e; // "slowdown"
2844    // AUTONOMOUS-BOT-IMPLEMENTED
2845    // TODO-HUMAN-REVIEW(PR-1151)
2846    // Fold the epoch into the mix with its own golden-ratio multiplier. At
2847    // `epoch == 0` this term is 0, leaving `mixed` identical to the epoch-less
2848    // factor; each successive epoch decorrelates the draw deterministically.
2849    const EPOCH_GOLDEN: u64 = 0xbf58_476d_1ce4_e5b9;
2850    let mixed = sched_seed
2851        ^ SLOWDOWN_SALT
2852        ^ ((dettid.as_raw() as u32 as u64).wrapping_mul(0x9e37_79b9_7f4a_7c15))
2853        ^ epoch.wrapping_mul(EPOCH_GOLDEN);
2854    let mut prng = Pcg64Mcg::seed_from_u64(mixed);
2855    // u in [0,1); map to exponent in [-1, 1) then factor = max_factor^exp,
2856    // i.e. a log-uniform draw over [1/max_factor, max_factor).
2857    let u: f64 = prng.random::<f64>();
2858    let exponent = 2.0 * u - 1.0;
2859    RcbTimeMultiplier::from_f64(max_factor.powf(exponent))
2860}

3765    /// End the current slice no later than `remaining` from now: the virtual
3766    /// time that was left before the earliest pending deadline when this
3767    /// thread was granted its turn.
3768    ///
3769    /// Global time advances exactly as this thread's clock does while it runs
3770    /// alone, so its slice then ends, and it takes a turn, once global time has
3771    /// passed that deadline.
3772    ///
3773    /// The slice keeps at least `min_slice`, one scheduler turn charge, and
3774    /// at least one RCB's worth of time. Every cap-induced slice end is a turn
3775    /// charged that much, so a thread runs at least as long as the turn it
3776    /// pays for: the virtual time the cap adds is at most the thread's own
3777    /// work. With a shorter floor, a sibling sleeping just over one turn
3778    /// charge made each slice about 1 us of work per 1-1.5 ms of virtual time.
3779    ///
3780    /// Only the software slice end moves, which takes effect at the thread's
3781    /// next check-in: a syscall, signal or timer event. The PMU maximum keeps
3782    /// its value, so a thread that makes no syscalls still runs to it: a
3783    /// precise PMU stop at every deadline costs about 70 ms of host time each,
3784    /// and a fast timer would make them constant
3785    /// (<https://github.com/rrnewton/hermit/issues/4121>). A schedule replay follows
3786    /// its recorded turns, and a preemption replay its recorded slice ends,
3787    /// neither capped. Past the last recorded point a replay's slice end is
3788    /// `LogicalTime::MAX` (its PMU maximum stays bounded), and the cap may
3789    /// lower it; that need not reproduce the record's own slice ends.
3790    pub(crate) fn cap_timeslice_at_deadline(
3791        &mut self,
3792        remaining: Duration,
3793        clock_multiplier: f64,
3794        min_slice: Duration,
3795        schedule_replay: bool,
3796    ) {
3797        let recorded_slice =
3798            self.preemption_points.is_some() && self.stats.last_recorded_slice.is_none();
3799        if recorded_slice || schedule_replay {
3800            return;
3801        }
3802        let Some(end) = self.end_of_timeslice else {
3803            return;
3804        };
3805        let one_rcb =
3806            (NANOS_PER_RCB * clock_multiplier * self.rcb_time_multiplier().as_f64()).ceil() as u64;
3807        let current = self.thread_logical_time.as_nanos();
3808        let cap = current
3809            + remaining
3810                .max(min_slice)
3811                .max(Duration::from_nanos(one_rcb + 1));
3812        if cap < end {
3813            debug!(
3814                "[dtid {}] end of slice moved from {} to {}, the earliest pending deadline (current {})",
3815                self.dettid, end, cap, current
3816            );
3817            self.end_of_timeslice = Some(cap);
3818        }
3819    }

5265/// new DetTid for some deterministic entropy. This ensures sequentially-spawned
5266/// threads get distinct PRNG states.

5268pub fn thread_rng_from_parent(msg: &str, parent: &Pcg64Mcg, child: DetTid) -> Pcg64Mcg {
5269    thread_rng_from_parent_entropy_labeled(msg, parent, child.as_raw() as u32 as u128, "tid")
5270}

5274    User,
5275    Chaos,
5276}
5277
5278impl ChildRngStream {
5279    fn domain(self) -> &'static [u8] {
5280        match self {
5281            Self::User => b"user",
5282            Self::Chaos => b"chaos",
5283        }
5284    }

5287// TODO-HUMAN-REVIEW(PR-1052): Review pedigree-derived child-thread PRNG seeding.
5288pub(crate) fn thread_rng_from_parent_pedigree(
5289    msg: &str,
5290    parent: &Pcg64Mcg,
5291    child: &Pedigree,
5292    stream: ChildRngStream,
5293) -> Pcg64Mcg {
5294    let bits = child.raw();
5295    let mut packed = Vec::with_capacity(bits.len().div_ceil(8));
5296    let mut byte = 0_u8;
5297    for (index, bit) in bits.iter().enumerate() {
5298        if *bit {
5299            byte |= 1 << (index % 8);
5300        }
5301        if index % 8 == 7 {
5302            packed.push(byte);
5303            byte = 0;
5304        }
5305    }
5306    if !bits.len().is_multiple_of(8) {
5307        packed.push(byte);
5308    }
5309
5310    // Pedigrees are unbounded, while Pcg64Mcg has a 128-bit state. Hash the
5311    // complete, length-delimited path instead of using the fallible compressed
5312    // virtual-PID encoding, so child RNG identity has no artificial tree-depth
5313    // boundary. The domain string prevents this digest from being confused
5314    // with another future use of the same pedigree serialization. Pcg64Mcg
5315    // forces its low state bit, so this retains 127 effective digest bits.
5316    let mut hasher = Sha256::new();
5317    hasher.update(b"hermit-child-rng-pedigree-v1\0");
5318    hasher.update(stream.domain());
5319    hasher.update([0]);
5320    hasher.update((bits.len() as u64).to_le_bytes());
5321    hasher.update(&packed);
5322    let digest = hasher.finalize();
5323
5324    let mut seed = <Pcg64Mcg as SeedableRng>::Seed::default();
5325    parent.clone().fill_bytes(seed.as_mut());
5326    for (seed_byte, pedigree_byte) in seed.iter_mut().zip(digest) {
5327        *seed_byte ^= pedigree_byte;
5328    }
5329    detlog!(
5330        "RNG {} seeding child {:?} pedigree {}: {:?} from parent {:?}",
5331        msg,
5332        stream,
5333        child,
5334        seed,
5335        parent
5336    );
5337    let mut rng = Pcg64Mcg::from_seed(seed);
5338    rng.next_u64();
5339    rng.next_u64();
5340    rng.next_u64();
5341    rng.next_u64();
5342    rng
5343}

5345fn thread_rng_from_parent_entropy(msg: &str, parent: &Pcg64Mcg, entropy: u128) -> Pcg64Mcg {
5346    thread_rng_from_parent_entropy_labeled(msg, parent, entropy, "entropy")

5348
5349fn thread_rng_from_parent_entropy_labeled(
5350    msg: &str,
5351    parent: &Pcg64Mcg,
5352    entropy: u128,
5353    identity_kind: &str,
5354) -> Pcg64Mcg {
5355    // Perform the default SeedableRng::from_seed procedure
5356    let mut seed = <Pcg64Mcg as SeedableRng>::Seed::default();
5357    // Generate a seed from the parent:
5358    parent.clone().fill_bytes(seed.as_mut());
5359    detlog!("RNG {} Generated new seed {:?}", msg, seed);
5360    // Pcg64Mcg forces its internal state odd, so seed bit zero carries no
5361    // entropy. DBT uses 96 bits for a stable process/thread sequence; mix those
5362    // bytes after the forced bit while retaining the existing DetTid layout.
5363    let entropy_bytes = entropy.to_le_bytes();
5364    for (seed_byte, entropy_byte) in seed[4..].iter_mut().zip(entropy_bytes) {
5365        *seed_byte ^= entropy_byte;
5366    }
5367    detlog!(
5368        "RNG {} seeding child {} {}: {:?} from parent {:?}",
5369        msg,
5370        identity_kind,
5371        entropy,
5372        seed,
5373        parent
5374    );
5375    let mut rng = Pcg64Mcg::from_seed(seed);
5376    // Pcg64Mcg integrates flipped bits across the state quickly. Some PRNGs don't.
5377    // Defensively ensure flipped bits "propagate":
5378    rng.next_u64();
5379    rng.next_u64();
5380    rng.next_u64();
5381    rng.next_u64();
5382    rng
5383}

5386/// placeholder for, which this image of the process has not seen: a memfd
5387/// named for it and sealed as Detcore seals its placeholders, which a guest
5388/// could imitate (a known limitation, see

5390/// system calls (under in-guest backends the tool shares the guest's C

5392/// The fixed procfs file the kernel opened at `fd`, if its descriptor link
5393/// names one (`ProcfsFile::snapshot_is_fixed`).
5394fn fixed_procfs_behind(fd: RawFd) -> Option<crate::procfs::ProcfsFile> {
5395    let link = std::ffi::CString::new(format!("/proc/self/fd/{fd}")).ok()?;
5396    let mut target = [0_u8; 256];
5397    // SAFETY: the path is NUL-terminated and `target` is writable.
5398    let length = unsafe {
5399        crate::util::raw_syscall(
5400            libc::SYS_readlinkat,
5401            [
5402                libc::AT_FDCWD as u64,
5403                link.as_ptr() as u64,
5404                target.as_mut_ptr() as u64,
5405                target.len() as u64,
5406                0,
5407                0,
5408            ],
5409        )
5410    };
5411    let length = usize::try_from(length)
5412        .ok()
5413        .filter(|length| *length < target.len())?;
5414    let path = std::path::Path::new(
5415        <std::ffi::OsStr as std::os::unix::ffi::OsStrExt>::from_bytes(&target[..length]),
5416    );
5417    crate::procfs::ProcfsFile::from_path(path).filter(crate::procfs::ProcfsFile::snapshot_is_fixed)
5418}

5421    // SAFETY: F_GET_SEALS only reads the descriptor's seals.
5422    let seals = unsafe {
5423        crate::util::raw_syscall(
5424            libc::SYS_fcntl,
5425            [fd as u64, libc::F_GET_SEALS as u64, 0, 0, 0, 0],
5426        )
5427    };
5428    // An `O_PATH` descriptor cannot show its seals (EBADF); its name decides.
5429    let wanted = i64::from(crate::syscalls::rendered_procfs::PLACEHOLDER_SEALS);
5430    if seals != -i64::from(libc::EBADF) && (seals < 0 || seals & wanted != wanted) {
5431        return None;
5432    }
5433    let link = std::ffi::CString::new(format!("/proc/self/fd/{fd}")).ok()?;
5434    let mut target = [0_u8; crate::procfs::PLACEHOLDER_LINK_CAPACITY];
5435    // SAFETY: the path is NUL-terminated and `target` is writable.
5436    let length = unsafe {
5437        crate::util::raw_syscall(
5438            libc::SYS_readlinkat,
5439            [
5440                libc::AT_FDCWD as u64,
5441                link.as_ptr() as u64,
5442                target.as_mut_ptr() as u64,
5443                target.len() as u64,
5444                0,
5445                0,
5446            ],
5447        )
